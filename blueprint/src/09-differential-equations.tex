\chapter{Linear Differential Equations}
\label{ch:differential-equations}

Differential equations belong to the same finite-estimate programme as the
integral and derivative chapters. A solution is an interval algorithm with
rational mesh, enclosure, and tail estimates giving nested boxes of shrinking
width, not a limit in a complete function space. The central target is
\begin{equation}\label{eq:linear-ode}
  x'(t)=A(t)x(t)+b(t),\qquad x(s)=x_s,
\end{equation}
where \(A(t)\) is a finite square matrix and \(x(t),b(t)\) are finite vectors.

\section{A familiar second-order model, then its vector form}

The intended first encounter with this chapter is a scalar second-order
equation, not an abstract matrix series.  For example, the forced harmonic
oscillator
\begin{equation}\label{eq:forced-oscillator}
 q''(t)+\omega^2q(t)=r(t),\qquad
 q(s)=q_s,\quad q'(s)=v_s
\end{equation}
has, in the homogeneous constant-coefficient case, the familiar solution
\[
 q(t)=q_s\cos\bigl(\omega(t-s)\bigr)
      +\frac{v_s}{\omega}\sin\bigl(\omega(t-s)\bigr)
 \qquad(\omega\ne0).
\]
The opposite sign \(q''-\lambda^2q=0\) has the corresponding exponential
modes \(e^{\lambda(t-s)}\) and \(e^{-\lambda(t-s)}\).  More generally,
constant-coefficient second-order equations are organized by exponential
modes, with conjugate imaginary modes rewritten as sine and cosine.  These
are the formulas scientists and engineers expect the later theory to recover.

They are not definitions of sine, cosine, or exponential in this chapter:
using them first would make the desired agreement theorem circular.  Instead,
put
\[
 X(t)=
 \begin{bmatrix}q(t)\\q'(t)\end{bmatrix},
 \qquad
 A_\omega=
 \begin{bmatrix}0&1\\-\omega^2&0\end{bmatrix},
 \qquad
 b_r(t)=
 \begin{bmatrix}0\\r(t)\end{bmatrix}.
\]
Then \eqref{eq:forced-oscillator} is precisely
\[
 X'(t)=A_\omega X(t)+b_r(t).
\]
The variable-coefficient second-order equation
\[
 q''(t)+a(t)q'(t)+c(t)q(t)=r(t)
\]
has the same first-order form with
\[
 A(t)=
 \begin{bmatrix}0&1\\-c(t)&-a(t)\end{bmatrix},
 \qquad
 b(t)=
 \begin{bmatrix}0\\r(t)\end{bmatrix}.
\]
Thus the general vector-valued inhomogeneous system \eqref{eq:linear-ode}
is not a detour: it is the constructive engine behind the familiar
second-order formulas.  Peano--Baker will construct the vector transition,
and the trigonometric and exponential displays above will later be recovered
by uniqueness.

\section{Finite time ordering before analysis}

\begin{definition}[Rational matrices and sampled systems]
\label{def:rational-linear-system}
\leanok
For a fixed dimension \(d\), a rational vector is a function
\(\{0,\ldots,d-1\}\to\Q\), and a rational matrix is a function
\(\{0,\ldots,d-1\}^2\to\Q\). Matrix multiplication is the literal finite
rational sum
\[
 (CD)_{ij}=\sum_{k=0}^{d-1}C_{ik}D_{kj}.
\]
The sampled recurrence is \(x_{k+1}=S_kx_k+g_k\). For the Euler sampling of
\eqref{eq:linear-ode} with rational step \(h\), take
\(S_k=I+hA(t_k)\) and \(g_k=hb(t_k)\). The implementation uses local
rational matrix definitions, not an imported matrix or analysis library.
\end{definition}

\begin{definition}[Finite Peano--Baker word sum]
\label{def:finite-peano-baker-word-sum}
\uses{def:rational-linear-system}
\leanok
For matrices \(B_0,\ldots,B_{N-1}\), the discrete Peano--Baker sum is
\[
 \sum_{N>i_1>\cdots>i_r\ge0}B_{i_1}\cdots B_{i_r},
\]
including the empty word \(I\). Newer sample times occur on the left, because
generally \(B_iB_j\ne B_jB_i\). Lean proves the finite algebra theorem
\[
 (I+B_{N-1})\cdots(I+B_0)
 =\sum_{N>i_1>\cdots>i_r\ge0}B_{i_1}\cdots B_{i_r}.
\]
by \code{discretePeanoBakerExpansion}; \code{DiscretePeanoBakerExpansion}
is its reusable proposition form. Separating this rational identity from
analytic convergence prevents a continuous ODE theorem from concealing either
issue.
\end{definition}

\begin{theorem}[Finite Peano--Baker enumeration]
\label{thm:finite-peano-baker-enumeration}
\uses{def:finite-peano-baker-word-sum}
\leanok
The checked enumerator contains exactly \(2^N\) words after \(N\) samples.
Its first three word lists are
\[
 [\,[]\,],\qquad [\,[],[0]\,],\qquad
 [\,[],[0],[1],[1,0]\,].
\]
Thus the two-step expression has the required ordered term \(B_1B_0\), not
the generally incorrect \(B_0B_1\). These are finite rational computations;
they do not yet assert convergence as the mesh tends to zero.
\end{theorem}

\section{Exact finite variation of constants}

\begin{theorem}[Discrete variation of constants]
\label{thm:discrete-variation-of-constants}
\uses{def:rational-linear-system}
\leanok
For the sampled system \(x_{k+1}=S_kx_k+g_k\), let the homogeneous
chronological action be
\[
 H_0(x)=x,\qquad H_{k+1}(x)=S_kH_k(x).
\]
It is useful to keep its transition matrix as a separate finite object:
\[
 T(m,0)=S_{m-1}\cdots S_1S_0,\qquad T(0,0)=I,
 \qquad H_m(x)=T(m,0)x.
\]
This is an exact recursive product, with no claim that a mesh has converged.
For Euler increments \(S_k=I+B_k\), the checked transition is exactly the
finite Peano--Baker chronological product
\[
 T(N,0)=(I+B_{N-1})\cdots(I+B_0).
\]
Lean proves the exact finite decomposition
\[
 x_N=H_N(x_0)+v_N,\qquad
 v_0=0,\quad v_{k+1}=S_kv_k+g_k.
\]
Thus \(v_N\) is the finite, time-ordered Duhamel contribution of the forcing
samples. The checked lemmas
for the transition action and variation-of-constants identity deliberately
keep the matrix products and the forcing response recursive, so they need only
the distributivity of finite rational matrix action; the future matrix-simplex
integral formula will use the same time order. The \code{ForcingZero} condition
and its two theorems prove that, when every \(g_k=0\), the second summand
vanishes and the trajectory is exactly the homogeneous action.
\end{theorem}

\begin{theorem}[Finite uniqueness for sampled linear systems]
\label{thm:sampled-linear-uniqueness}
\uses{def:rational-linear-system,thm:discrete-variation-of-constants}
\leanok
Fix rational matrices \(S_k\), rational forcing vectors \(g_k\), and an
initial vector \(x_0\).  A sampled solution is any vector sequence satisfying
\[
  x_{k+1}=S_kx_k+g_k,\qquad x_0\text{ as prescribed}.
\]
Lean proves that every sampled solution agrees at every natural index with
the recursive trajectory.  In particular, every homogeneous candidate
\[
  z_{k+1}=S_kz_k,\qquad z_0=0,
\]
is identically zero.

The proof is induction over the literal finite rational updates.  It uses
neither an inverse transition matrix nor a limiting argument.  Together with
the preceding transition formula, it says that the chronological product is
not merely one possible calculation: it is the unique sampled evolution from
its declared initial data.  This is the exact discrete uniqueness component of
the later continuous theorem, not a claim that continuous ODE uniqueness has
already been proved.
\end{theorem}

\section{The continuous Peano--Baker target}

\begin{definition}[Interval-certified linear-system input]
\label{def:interval-linear-system}
\uses{def:finite-peano-baker-word-sum}
\leanok
On a rational time interval \([s,t]\), each entry of \(A\) and \(b\) is an
existing \code{FunctionOnInterval} with the same domain. The regularity
obligation requires a supplied \code{IntervalRegularOn} witness for each
entry. The ODE layer therefore consumes interval algorithms and moduli, not a
classical assumption that coefficients live on a completed real interval.
\end{definition}

\begin{theorem}[Peano--Baker state transition: construction target]
\label{thm:peano-baker-state-transition}
\uses{def:interval-linear-system,thm:effective-ftc-certificate}
For interval-regular coefficients, construct a matrix of raw reals
\(\Phi(t,s)\) by the ordered-simplex series
\begin{align*}
 \Phi(t,s)={}&I+\int_s^t A(u_1)\,du_1\\
 &+\int_s^t A(u_1)\int_s^{u_1}A(u_2)\,du_2\,du_1+\cdots.
\end{align*}
The order of factors follows the nested integration order. The certificate
consists of finite interval boxes for truncated simplex integrals and a
rational submultiplicative matrix-norm bound for the tail. A bound
\(\lVert A(u)\rVert\le M\) on an interval of length \(T\) should yield the
factorial tail \(\sum_{r>N}(MT)^r/r!\). The proof must obtain nested,
shrinking raw-matrix boxes and verify
\[
 \partial_t\Phi(t,s)=A(t)\Phi(t,s),\qquad \Phi(s,s)=I,
\qquad \Phi(t,r)\Phi(r,s)=\Phi(t,s).
\]
This is a target, not a completed Lean theorem: componentwise matrix interval
arithmetic, iterated-simplex integral constructors, and the factorial-tail
certificate remain to be formalized.
\end{theorem}

\begin{theorem}[Variation of constants: construction target]
\label{thm:variation-of-constants}
\uses{thm:peano-baker-state-transition}
Once the state transition has the preceding certificate, the solution of
\eqref{eq:linear-ode} should be the interval-certified raw vector
\[
 x(t)=\Phi(t,s)x_s+\int_s^t\Phi(t,u)b(u)\,du.
\]
The derivative identity is checked by finite product and integral rules, and
the initial value follows from the zero-length integral certificate. This is
the general linear ODE result needed by most applications; nonlinear
existence theorems are deliberately outside this first layer.
\end{theorem}

\begin{theorem}[Linear Picard--Lindel\"of: construction target]
\label{thm:linear-picard-lindelof}
\uses{thm:sampled-linear-uniqueness,thm:peano-baker-state-transition,thm:variation-of-constants}
The preceding Peano--Baker transition and variation-of-constants formula are
also the constructive existence-and-uniqueness theorem for the affine linear
initial-value problem
\[
  x'(u)=A(u)x(u)+b(u),\qquad x(s)=x_s.
\]
The certified solution is the displayed variation-of-constants raw vector.
Its executable modulus is assembled from the coefficient enclosure, simplex
quadrature error, and the Peano--Baker factorial tail.  If two certified
solutions have the same initial box, their difference solves the homogeneous
system with zero initial data.  The uniqueness proof must be a universal
finite estimate, not merely a claim about the constructed solution.  Given a
rational bound \(B\) for a candidate difference \(z\), iterate its Volterra
identity \(z(t)=\int_s^tA(u)z(u)\,\dd u\) exactly \(N\) times.  A coefficient
bound \(\lVert A(u)\rVert\le M\) on an interval of length \(T\) then gives
\[
  \lVert z(t)\rVert\le B\,\frac{(MT)^N}{N!}.
\]
The same rational factorial-tail certificate used to construct the
Peano--Baker series makes this bound smaller than every requested positive
rational.  Hence \(z(t)=0\) as a raw real, and the two solutions agree,
without an ambient complete function space or a nonconstructive supremum.
Thus the continuous Peano--Baker layer is the linear Picard--Lindel\"of
theorem needed for science and engineering, including the scalar uniqueness
route for \(f'=f\).

This deliberately does not assert the general nonlinear theorem: that later
extension needs an interval-certified Lipschitz map and a separate Picard
contraction construction.
\end{theorem}

\begin{remark}[Why Peano--Baker existence also needs a separate uniqueness certificate]
\label{rem:peano-baker-uniqueness-certificate}
\uses{thm:linear-picard-lindelof}
The Peano--Baker construction itself supplies one interval algorithm satisfying
the initial-value problem.  That is existence, not yet uniqueness: another
candidate solution must be compared with it by a finite estimate that applies
to every certified candidate.  The required input is a rational coefficient
bound \(M\), an interval length \(T\), and a rational enclosure \(B\) for the
candidate difference \(z\).  A supplied solution certificate normally gives
such a bound directly; if it is local, a finite rational cover of the time
interval supplies one before the argument is iterated.

Subtracting the two Volterra identities gives a zero-initial homogeneous
identity.  Its \(N\)-fold ordered substitution is a finite simplex expression,
and interval multiplication bounds its norm by
\[
  B\frac{(MT)^N}{N!}.
\]
For a requested positive rational epsilon, the factorial-tail procedure
computes an \(N\) for which this rational bound is below epsilon.  The two
candidate output boxes consequently overlap at every requested precision,
which is the project definition of equality of raw vectors.  No supremum of
\(\lVert z\rVert\), limiting function sequence, or completeness theorem is
being invoked.  This is why the linear theorem is properly described as
constructive Picard--Lindel\"of, while a nonlinear Picard theorem still needs
additional contraction data.
\end{remark}

\section{Elementary functions recovered by linear uniqueness}

\begin{theorem}[Elementary-function agreement from linear uniqueness: construction target]
\label{thm:linear-ode-elementary-function-agreement}
\uses{thm:linear-picard-lindelof}
The continuous linear theorem has a deliberately important consequence beyond
the solution of a single engineering model: it supplies a common
identification principle for independently constructed elementary functions.
Every proposed representation must first carry its own interval algorithm,
derivative certificate, and initial-value certificate.  If two such
representations solve the same linear problem, the factorial-tail uniqueness
argument identifies them as raw functions.

For the scalar problem
\[
 E'(t)=E(t),\qquad E(0)=1,
\]
Peano--Baker constructs the canonical exponential directly from the
one-dimensional transition.  The power-series evaluator, the
repeated-multiplication (Euler/compound-interest) evaluator, and the
inverse-logarithmic-integral evaluator therefore agree once each has been
shown to satisfy this equation and initial condition:
\[
 E_{\mathrm{series}}
 \Equiv E_{\mathrm{repeated}}
 \Equiv E_{\mathrm{invlog}}.
\]
This is stronger and cleaner than a collection of pairwise limit comparisons:
the representations are all forced to be the same solution.  In particular,
the chosen Euler base is \(e=E(1)\), and the derivative statement
\(\dd(e^t)/\dd t=e^t\) is a theorem of the selected canonical
representation, not a convention.

The rotation system supplies the trigonometric counterpart:
\[
 \frac{\dd}{\dd t}
 \begin{bmatrix}C(t)\\S(t)\end{bmatrix}
 =
 \begin{bmatrix}0&-1\\1&0\end{bmatrix}
 \begin{bmatrix}C(t)\\S(t)\end{bmatrix},
 \qquad
 (C(0),S(0))=(1,0).
\]
The Peano--Baker solution is then identified with the sine and cosine
constructed from the inverse-arctangent circle chart, once their
derivative and initial-value certificates have been supplied.  Over rational
complex boxes this same agreement yields
\[
 \exp(it)=C(t)+iS(t).
\]
Thus Euler's identity is a long but natural ODE consequence, rather than an
additional definition of either exponential or trigonometry.

On the positive branch, the inverse of this canonical exponential is the
canonical logarithm.  The positive logarithmic-integral evaluator must be
proved to be that inverse by its own finite FTC and inverse-function
certificates; its already checked comparison at \(2\) is useful boundary
data, not a substitute for the function theorem.  Once this agreement is
available, the integration-by-parts target
\[
 \int_0^1\arctan(x)\,\dd x
 =\frac{\pi}{4}-\frac12\log 2
\]
has the canonical \(\log 2\) required for the derived pi computation
\[
 \pi=4\int_0^1\arctan(x)\,\dd x+2\log2.
\]
The global product derivative, FTC, and change-of-variables certificates
needed for that integral identity remain separate work.  Likewise,
\(\exp(i\pi/2)=i\) and \(\pi=-2i\log(i)\) remain future scoreboard
corroborations, not completed consequences of this theorem.
\end{theorem}

\section{Special cases and computational uses}

\begin{definition}[Constant and commuting samples]
\label{def:constant-commuting-samples}
\uses{def:rational-linear-system}
\leanok
The finite layer records when time ordering simplifies: all step matrices
equal, or all sampled steps commute pairwise. These are explicit propositions,
since a generic linear system has neither property.

A stronger finite nilpotence condition is also useful: a sampled family is
square-zero when \(B_iB_j=0\) for every pair of sample times. This is more
than commutation, but it makes the first nontrivial truncation of the
Peano--Baker series exact.
\end{definition}

\begin{theorem}[Finite Peano--Baker specializations]
\label{thm:peano-baker-specializations}
\uses{thm:peano-baker-state-transition,def:constant-commuting-samples}
\leanok
The finite layer already proves constant, zero, square-zero, and
pairwise-commuting specializations. For a constant Euler increment \(B\),
\[
 (I+B)\cdots(I+B)=(I+B)^N
   =\sum_{N>i_1>\cdots>i_r\ge0}B\cdots B,
\]
where the right side is the checked finite Peano--Baker word sum. For
\(B=0\), it is exactly \(I\) at every stage. These statements are the
constant and zero-coefficient regression cases for the eventual continuous
theorem, proved by \code{chronologicalProduct\_constant},
\code{peanoBakerDiscreteSum\_constant}, and
\code{peanoBakerDiscreteSum\_zeroCoefficient}.

For a square-zero sampled family, every selected word with two or more
factors is zero. Therefore Lean proves the exact finite collapse
\[
 (I+B_{N-1})\cdots(I+B_0)
 = I+\sum_{k=0}^{N-1}B_k.
\]
The right-hand side is the executable recursive matrix sum
\code{matrixSequenceSum}; the two checked theorems above establish the
formula both for the chronological product and for the ordered-word
Peano--Baker sum. This is a genuine nilpotent regression case, not a
continuous limiting argument.

For arbitrary pairwise-commuting sampled increments, Lean now proves the
finite algebra needed before an exponential collapse: every \(B_i\) commutes
with the accumulated chronological transition
\[
 B_i (I+B_{N-1})\cdots(I+B_0)
 =
 (I+B_{N-1})\cdots(I+B_0) B_i.
\]
The checked lemmas \code{PairwiseCommuting} and
\code{matrixMul\_commutes\_chronologicalProduct} establish precisely this
reordering fact, while \code{chronologicalProduct\_split} gives the
time-interval composition law with later increments on the left. Identifying
the continuous commuting case with
\(\exp(\int_s^t A(u)\,du)\) still requires the analytic construction and is
not asserted by these finite lemmas.

The continuous construction should yield the familiar special cases
\[
 \begin{array}{rcl}
 A(u)=A_0&:&\Phi(t,s)=\exp((t-s)A_0),\\[2mm]
 A(u)A(v)=A(v)A(u)&:&
 \Phi(t,s)=\exp\!\left(\int_s^tA(u)\,du\right),\\[2mm]
 x'=a(t)x+b(t)&:&
 x(t)=e^{\int_s^t a}\,x_s+\int_s^t e^{\int_u^t a}\,b(u)\,du.
 \end{array}
\]
For piecewise-constant coefficients, the direct finite answer is the
chronological product of constant-step exponentials. The remaining
specializations are the analytic commuting-exponential identification and
finite algebra for strictly triangular and higher-order nilpotent coefficient
algebras; sufficiently long ordered products should then vanish, making the
series a finite certified calculation.
\end{theorem}

\begin{remark}[Engineering-facing error certificates]
\label{rem:ode-error-certificates}
\uses{thm:peano-baker-state-transition,thm:variation-of-constants}
A practical result should return a rational enclosure for each component and
an error budget split between mesh error, coefficient-evaluation error, and
the Peano--Baker tail. Piecewise-constant dynamics, constant-coefficient
circuits and control models, triangular reaction chains, and small coupled
linearizations are natural first regression examples. Stiffness, adaptive
step choice, and nonlinear Picard--Lindel\"of theory should follow only after
these finite certificates are in place.
\end{remark}
