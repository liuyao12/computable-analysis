\chapter{Linear Differential Equations}
\label{ch:differential-equations}

The continuous target is the rational-box initial-value problem
\begin{equation}\label{eq:linear-ode}
 x'(t)=A(t)x(t)+b(t),\qquad x(s)=x_s.
\end{equation}
The chapter starts from the finite time-ordered calculation, then states the
still-separate continuous certificates: a Peano--Baker enclosure for vector
existence and a factorial estimate for vector uniqueness.  Scalar
\(E'=E,E(0)=1\) instead uses the direct finite-mesh route in
Chapter~\ref{ch:exponential-logarithm}.

\section{Model, vector form, and proof route}

\begin{equation}\label{eq:forced-oscillator}
 q''+\omega^2q=r,\qquad q(s)=q_s,\quad q'(s)=v_s.
\end{equation}
With \(X=(q,q')^T\), this is \eqref{eq:linear-ode} with
\[
 A_\omega=\begin{bmatrix}0&1\\-\omega^2&0\end{bmatrix},\qquad
 b_r=\begin{bmatrix}0\\r\end{bmatrix}.
\]
The desired familiar output is
\begin{equation}\label{eq:forced-oscillator-duhamel}
 q(t)=q_s\cos(\omega(t-s))+
 \frac{v_s}{\omega}\sin(\omega(t-s))+
 \int_s^t\frac{\sin(\omega(t-u))}{\omega}r(u)\,\dd u.
\end{equation}
It is a later identification theorem, not a definition of sine or cosine.

\section{Finite time ordering}
\label{sec:ode-construction-identification-route}

\begin{definition}[Rational sampled system]
\label{def:rational-linear-system}
\leanok
For rational matrices and vectors, the executable recurrence is
\[
 x_{k+1}=S_kx_k+g_k.
\]
All matrix products and sums are literal finite rational loops.
\end{definition}

\begin{theorem}[Forced oscillator: finite seed]
\label{thm:forced-oscillator-finite-seed}
\leanok
Euler sampling gives
\[
 q_{n+1}=q_n+hv_n,\qquad
 v_{n+1}=v_n-h\omega^2q_n+hr_n,
\]
and Lean checks the resulting finite second-difference identity.  No mesh
limit is claimed here.
\end{theorem}

\begin{definition}[Finite Peano--Baker sum]
\label{def:finite-peano-baker-word-sum}
\leanok
For increments \(B_0,\ldots,B_{N-1}\), the checked identity is
\[
 (I+B_{N-1})\cdots(I+B_0)
 =\sum_{N>i_1>\cdots>i_r\ge0}B_{i_1}\cdots B_{i_r}.
\]
Later times are on the left.
\end{definition}

\begin{theorem}[Finite word enumeration]
\label{thm:finite-peano-baker-enumeration}
\leanok
\begin{verbatim}
def words(n):
    if n == 0: return [[]]
    old = words(n - 1)
    return old + [[n - 1] + word for word in old]
\end{verbatim}

\end{figure}
\end{theorem}

\begin{theorem}[Discrete variation of constants and uniqueness]
\label{thm:discrete-variation-of-constants}
\label{thm:sampled-linear-uniqueness}
\leanok
Writing \(T(N,k)=S_{N-1}\cdots S_k\), Lean proves
\[
 x_N=T(N,0)x_0+\sum_{k<N}T(N,k+1)g_k.
\]
The recursive trajectory is the unique sampled solution with its declared
initial vector.  This is exact finite induction, including the zero-initial
homogeneous case.
\end{theorem}

\section{Continuous Peano--Baker target}

\begin{definition}[Interval-certified input]
\label{def:interval-linear-system}
\leanok
Each entry of \(A\) and \(b\) is an interval-regular function on the same
rational time interval.  The continuous construction consumes those
algorithms and their moduli, not a completed function space.
\end{definition}

\begin{theorem}[State transition: construction target]
\label{thm:peano-baker-state-transition}
\lean{ComputableAnalysis.LinearODE.peanoBakerFactorialTail_shifted_le_eps}
The required transition is
\[
 \Phi(t,s)=I+\int_s^tA(u_1)\,du_1+
 \int_s^tA(u_1)\int_s^{u_1}A(u_2)\,du_2\,du_1+\cdots.
\]
Finite simplex quadrature boxes plus a matrix bound \(\lVert A\rVert\le M\)
must provide the raw-matrix enclosure.  The scalar factorial-tail schedule
for \(\sum_{r>N}(MT)^r/r!\) is already checked; matrix/simplex packaging is
the remaining work.
\end{theorem}

\begin{theorem}[Variation of constants: construction target]
\label{thm:variation-of-constants}
\[
 x(t)=\Phi(t,s)x_s+\int_s^t\Phi(t,u)b(u)\,du.
\]
The same enclosure must certify the derivative and initial-value fields.
\end{theorem}

\begin{theorem}[Linear Picard--Lindel\"of: construction target]
\label{thm:linear-picard-lindelof}
\lean{ComputableAnalysis.LinearODE.zeroInitialVolterraIterationBound_eq_zero}
For two candidates, the zero-initial difference must satisfy the finite
Volterra estimate
\[
 \lVert z(t)\rVert\le B\frac{(MT)^N}{N!}\quad\text{for every }N.
\]
The checked factorial schedule makes this smaller than every positive
rational tolerance, hence proves raw-box equality.  Deriving this universal
bound from interval Volterra identities is still open; no general nonlinear
Picard theorem is claimed.
\end{theorem}

\begin{remark}[Why uniqueness is separate]
\label{rem:peano-baker-uniqueness-certificate}
Peano--Baker produces one candidate.  The iterated zero-initial bound is what
compares it with every other certified candidate; it replaces neither
existence nor a hidden completeness principle.
\end{remark}

\section{Elementary-function identification}

\begin{theorem}[Agreement from vector linear uniqueness: construction target]
\label{thm:linear-ode-elementary-function-agreement}
Once each named representation independently supplies its derivative and
initial-value certificate, vector linear uniqueness identifies it with the
Peano--Baker solution.  The intended application is
\[
 \binom{C}{S}'=\begin{bmatrix}0&-1\\1&0\end{bmatrix}\binom{C}{S},
 \quad(C(0),S(0))=(1,0).
\]
This is the route from rotation to \(\exp(it)=C(t)+iS(t)\).  The scalar
exponential/logarithm route is intentionally separate: it uses direct mesh
contraction, not the vector Peano--Baker theorem.  In both cases,
representative-specific derivative proofs are not supplied by uniqueness.
\end{theorem}

\begin{theorem}[Finite-mesh contraction: checked closure]
\label{thm:direct-scalar-mesh-uniqueness}
\lean{ComputableAnalysis.ScalarODE.FiniteMesh.sumUpTo}
\lean{ComputableAnalysis.ScalarODE.FiniteMesh.sumUpTo_increments}
\lean{ComputableAnalysis.ScalarODE.FiniteMeshDifferenceBound}

\lean{ComputableAnalysis.ScalarODE.FiniteMeshDifferenceBound.next_le_half}
\lean{ComputableAnalysis.ScalarODE.ShortBlockMeshSweep}
\lean{ComputableAnalysis.ScalarODE.ShortBlockMeshSweep.next_le_half}
\lean{ComputableAnalysis.ScalarODE.DirectMeshHalvingCertificate}
\lean{ComputableAnalysis.ScalarODE.DirectMeshHalvingCertificate.ofShortBlockSweeps}
\lean{ComputableAnalysis.ScalarODE.DirectMeshHalvingCertificate.bound_le_geometric}
\lean{ComputableAnalysis.ScalarODE.DirectMeshHalvingCertificate.error_le_eps}
\lean{ComputableAnalysis.ScalarODE.DirectMeshHalvingCertificate.error_eq_zero}
\lean{ComputableAnalysis.ScalarODE.SelfDerivativeDirectMeshComparison}
\lean{ComputableAnalysis.ScalarODE.SelfDerivativeDirectMeshComparison.equivalent}
\lean{ComputableAnalysis.ScalarODE.selfDerivativeInitialValueUnique_of_directMesh}
\lean{ComputableAnalysis.exp.powerSeries_equiv_logIntegralInverse_on_interval_of_directMesh}
\leanok
For the difference of two scalar candidates with common initial value, a
finite rational mesh on a sufficiently short block supplies a nonnegative
error envelope \(B_r\).  At refinement \(r\), sum the cell estimates into
\[
 B_{r+1}\le L B_r+R_r,\qquad
 L\le\frac14,\quad R_r\le\frac{B_r}{4}.
\]
This is the finite \code{ShortBlockMeshSweep} certificate, and Lean checks
that it gives
\[
  B_{r+1}\le \frac{B_r}{2},
\]
then Lean proves \(B_r\le B_0\,2^{-r}\), chooses an explicit dyadic refinement
count for any positive rational tolerance, and proves the error below that
tolerance. Applying this to the proposed error divided by two concludes that
the error is zero.
More concretely, \code{FiniteMeshDifferenceBound} records rational endpoint
envelopes \(d_0,\ldots,d_N\), with \(d_0=0\), and the individual cell
bounds
\[
 d_{j+1}-d_j\le hB_r+\rho_j.
\]
Lean proves the exact telescoping identity
\(\sum_{j<N}(d_{j+1}-d_j)=d_N-d_0\), then packages the result as the
short-block sweep above.  Thus the remaining analytic step is specifically a
finite derivative-to-cell estimate, not a hidden appeal to a supremum,
integral limit, or complete function space.
Longer intervals use a finite chain of short blocks.  The closure theorem is
checked; deriving the one-sweep envelope from arbitrary interval derivative
certificates is the remaining analytic construction.  This route is
deliberately neither Peano--Baker nor Picard iteration.
\end{theorem}

\begin{remark}[Engineering-facing output]
\label{rem:ode-error-certificates}
The intended result reports rational component boxes and a split error budget:
coefficient evaluation, simplex quadrature, and factorial tail.  Constant,
piecewise-constant, triangular, and small coupled systems are the first
regression examples.
\end{remark}
