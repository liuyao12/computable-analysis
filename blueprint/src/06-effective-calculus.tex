\chapter{Fundamental Theorem of Calculus}
\label{ch:effective-calculus}

\section{The revisionist order: observation, foundation, reconstruction}

Elementary calculus begins with a striking computational observation.  For
the elementary functions people could calculate with, differentiation and
integration undo one another:
\[
  F(b)-F(a)=\int_a^b F'(x)\,\dd x.
\]
One can understand this first by reasoning with finer and finer partitions:
each small change is approximately a derivative value times a small interval,
and the changes telescope.  In that setting, the Mean Value Theorem can look
like a rather dull corollary: the integral average of $F'$ must equal $F'(t)$
at some intermediate point.

The foundational history runs in the opposite direction.  Once calculus was
to cover arbitrary functions rather than a catalogue of formulas, the Mean
Value Theorem became the more revealing statement.  Rolle's theorem, extrema,
and the intermediate-value behavior of derivatives expose the role of order
and completeness (classically, the least-upper-bound principle).  Direct
proofs of the Mean Value Theorem then provide the estimates needed to justify
the FTC for general functions.  In this sense the two principles are closely
interderivable under matching regularity and integration hypotheses, although
neither theorem by itself is an unrestricted replacement for all of the
definitions used by the other.

This book keeps both perspectives visible.  The elementary FTC supplies the
motivation and the endpoint quantity to be computed; the MVT is treated as a
foundational target, not merely as a consequence of an already constructed
integral; and the general FTC is reconstructed from local secant/derivative
certificates, partition refinement, and telescoping.  In the computable
version, a least-upper-bound argument is replaced by explicit rational
brackets, width schedules, and effective bisection.  Thus the relevant
question is not whether a completed real number is silently present, but
which finite data locate the intermediate slope and control the partition
error.

The historical relationship is best represented in three layers.  First, the
Fundamental Theorem of Calculus gives the endpoint identity
\[
  f(b)-f(a)=\int_a^b f'(x)\,dx.
\]
Second, the integral mean-value theorem identifies the integral with the
interval length times an average derivative,
\[
  \int_a^b f'(x)\,dx=(b-a)\,\overline{f'}.
\]
Finally, Lagrange's Mean Value Theorem supplies a point $t\in(a,b)$ with
\(\overline{f'}=f'(t)\).  The project treats the first two layers as the
primary computable interfaces.  A pointwise witness is a legitimate next
target for an effectively continuous monotone derivative (or an effectively
differentiable convex function), where a certified bisection can locate the
needed value.  Convexity without differentiability should instead use
one-sided derivatives or subgradients; it does not generally provide an
exact pointwise $f'(t)$.
\end{remark}

\begin{definition}[Raw-real order]\label{def:raw-real-order}
\lean{ComputableAnalysis.RealRaw.Le}
\uses{def:raw-real}
\leanok
For raw reals \(X,Y\), define
\[
  X\le Y
\]
to mean that every rational lower approximation to \(X\) is at most every
rational upper approximation to \(Y\):
\[
  \forall n,m,\qquad X_n^- \le Y_m^+.
\]
This is the order relation used in mathematical statements.  For valid raw
representatives,
\[
  X\sim Y
  \quad\Longleftrightarrow\quad
  X\le Y\ \text{and}\ Y\le X.
\]
\end{definition}

\begin{definition}[Endpoint difference]\label{def:endpoint-diff}
\lean{ComputableAnalysis.endpointDifferenceInterval}
\uses{def:real-fun-raw,def:raw-real-arithmetic}
For a raw real-valued function \(F\), define
\[
  E(F;u,v)=F(v)-F(u).
\]
At stage \(p\), if \(F(u)_p=[A^-,A^+]\) and
\(F(v)_p=[B^-,B^+]\), then
\[
  E(F;u,v)_p=[B^- - A^+,B^+ - A^-].
\]
\end{definition}

\begin{definition}[Rational secant]\label{def:rational-secant}
\lean{ComputableAnalysis.secantRaw}
\uses{def:endpoint-diff,def:raw-real-order}
\leanok
For rational \(u<v\), the secant of \(F\) on \([u,v]\) is the raw real
\[
  \Sec_F(u,v)=\frac{F(v)-F(u)}{v-u}.
\]
Its finite stages are the rational interval slopes
\[
  \Sec_F(u,v)_p=E(F;u,v)_p/(v-u).
\]
\end{definition}

\begin{definition}[Convexity]\label{def:exact-convexity}
\lean{ComputableAnalysis.ExactConvexOn}
\uses{def:rational-secant,def:raw-real-order}
\leanok
A raw real-valued function \(F\) is convex on the rational interval
\([a,b]\) when it is defined and valid at every rational point of
\([a,b]\), and its rational secants are monotone:
\[
  \Sec_F(w,x)\le \Sec_F(y,z)
\]
whenever
\[
  a\le w<x\le y<z\le b.
\]
Concavity is the same statement with the secant order reversed.
\end{definition}

\section{Derivative of a Convex Function}

\begin{definition}[Right derivative of a convex function]
\label{def:convex-right-derivative}
\lean{ComputableAnalysis.RightDerivativeAt}
\uses{def:exact-convexity}
Let \(F\) be convex on \([a,b]\), and let \(q<b\) be rational.  The right
secants
\[
  \Sec_F(q,q+h),\qquad h>0,\quad q+h\le b,
\]
are monotone as \(h\downarrow 0\) and bounded below by any left secant ending
at \(q\).  Lean does not obtain a derivative by an infimum.  Instead,
\code{RightDerivativeAt} supplies an explicit valid raw interval algorithm
\(D_+F(q)\), and \code{IsRightDerivative} certifies two finite-order laws:
\[
  D_+F(q)\le\Sec_F(q,q+h)
\]
for every admissible rational \(h\), and every raw real with the same
lower-bound property is \(\le D_+F(q)\).  Thus ``greatest lower bound'' is a
specification checked against a supplied algorithm, not an invocation of a
completed-real infimum or a limit theorem.
\end{definition}

\begin{definition}[Left derivative of a convex function]
\label{def:convex-left-derivative}
\lean{ComputableAnalysis.LeftDerivativeAt}
\uses{def:exact-convexity}
For \(a<q\), \code{LeftDerivativeAt} likewise supplies a valid raw interval
algorithm \(D_-F(q)\).  \code{IsLeftDerivative} proves that every admissible
left secant is \(\le D_-F(q)\), and that \(D_-F(q)\) is below every raw upper
bound for those secants.  This least-upper-bound characterization is again a
certificate, not a supremum construction.  For convex \(F\),
\(D_-F(q)\le D_+F(q)\).  The ordinary two-sided derivative exists exactly at
the rational points where these two one-sided derivatives are equivalent.
\end{definition}

\begin{theorem}[Right derivative is monotone]
\label{thm:convex-right-derivative-increasing}
\lean{ComputableAnalysis.rightDerivativeAt_mono}
\uses{def:convex-right-derivative}
\leanok
If \(F\) is convex on \([a,b]\), then
\[
  q_1<q_2
  \quad\Longrightarrow\quad
  D_+F(q_1)\le D_+F(q_2)
\]
for rational \(q_1,q_2<b\).
\end{theorem}

\begin{proof}
Apply the greatest-lower-bound certificate for \(D_+F(q_2)\).  It remains to
show that \(D_+F(q_1)\) is below every admissible right secant at \(q_2\).
The lower-bound certificate for \(D_+F(q_1)\), together with rational secant
monotonicity from convexity, gives exactly that finite order statement.
\end{proof}

\begin{theorem}[Left derivative is monotone]
\label{thm:convex-left-derivative-increasing}
\lean{ComputableAnalysis.leftDerivativeAt_mono}
\uses{def:convex-left-derivative}
\leanok
If \(F\) is convex on \([a,b]\), then
\[
  q_1<q_2
  \quad\Longrightarrow\quad
  D_-F(q_1)\le D_-F(q_2)
\]
for rational \(q_1,q_2\) in the interval.  The proof is the dual finite
certificate: every left secant ending at \(q_1\) is below the bridge secant
from \(q_1\) to \(q_2\), which is itself bounded above by the supplied least
upper bound at \(q_2\).
\end{theorem}

\begin{proof}
Apply the least-upper-bound certificate for \(D_-F(q_1)\) with \(D_-F(q_2)\)
as the proposed upper bound.  Convex secant monotonicity transports each
left secant at \(q_1\) through the bridge \([q_1,q_2]\), and the upper-bound
certificate for \(D_-F(q_2)\) closes the finite order chain.
\end{proof}

\begin{theorem}[Left derivative is below the right derivative]
\label{thm:convex-left-le-right-derivative}
\lean{ComputableAnalysis.leftDerivativeAt_le_rightDerivativeAt}
\uses{def:convex-left-derivative,def:convex-right-derivative}
\leanok
For a convex function, the supplied one-sided derivative certificates satisfy
\[
  D_-F(q)\le D_+F(q).
\]
This is the finite corner inequality: every left secant is below every right
secant, so each left secant is a lower bound for the right-secant family.
The theorem therefore records the possible corner without constructing a
classical limit or asserting two-sided differentiability.
\end{theorem}

\begin{proof}
Apply the greatest-lower-bound certificate for \(D_+F(q)\) to an arbitrary
left secant.  Convex secant monotonicity places that left secant below every
admissible right secant, and the least-upper-bound certificate for \(D_-F(q)\)
then yields the stated order.
\end{proof}

\begin{definition}[Derivative formula identification]\label{def:formula-identification}
\uses{def:convex-right-derivative}
Let \(g\) be a proposed raw function formula.  On a rational interval,
\(g\) is identified with \(D_+F\) when it is equivalent to the right-secant
algorithm supplied by \code{RightDerivativeAt} and satisfies its secant-bound
certificate.  The formula is not a guess: it is proved from finite secant
inequalities or algebraic identities for the particular function.
\end{definition}

\begin{remark}[Corners]\label{rem:convex-corners}
\uses{def:convex-left-derivative,def:convex-right-derivative}
Convexity does not imply existence of the two-sided derivative.  The function
\(|x|\) is convex and has \(D_-F(0)<D_+F(0)\).  The universal FTC for convex
functions must therefore use a one-sided derivative, or else explicitly
restrict to the points where the left and right derivatives agree.
\end{remark}

\section{Fundamental Theorem of Calculus}

The integral used in this chapter is the constructive definite integral
already introduced in Chapter~\ref{ch:integrals}.  The present chapter proves
when such an integral is equal to an endpoint difference.  This is one reason
the pi integration suite is a useful regression suite: some existing integral
computations of \(\pi\) can be proved equivalent by finite comparison, while
others should eventually be proved equivalent through FTC-style endpoint
identities.  It is evidence of one integration capability, not an aggregate
measure of calculus readiness.

\begin{remark}[Effective FTC reconstruction]
\label{rem:effective-ftc-reconstruction}
The order is deliberate: the theorem below is not being used to manufacture
the Mean Value Theorem retroactively.  Instead, the local Mean Value and
secant-bracket principles established earlier are the estimates from which
the endpoint identity is rebuilt.  For convex functions, the right and left
derivatives provide monotone bounds on every secant; summing those bounds over
finer rational partitions produces the FTC certificate.  This is the
computable analogue of the classical route in which the direct MVT proof,
grounded in completeness, supports the general FTC.
\end{remark}

\begin{theorem}[Fundamental theorem of calculus]\label{thm:convex-ftc}
\lean{ComputableAnalysis.ConvexFTCCertificate}
\lean{ComputableAnalysis.ConvexFTCCertificate.toCurvatureFTCCertificate}
\lean{ComputableAnalysis.ConvexFTCCertificate.toDerivativeBoundFTC}
\lean{ComputableAnalysis.ConvexFTCCertificate.equiv_endpoint}
\lean{ComputableAnalysis.ConcaveFTCCertificate}
\lean{ComputableAnalysis.ConcaveFTCCertificate.toCurvatureFTCCertificate}
\lean{ComputableAnalysis.ConcaveFTCCertificate.toDerivativeBoundFTC}
\lean{ComputableAnalysis.ConcaveFTCCertificate.equiv_endpoint}
\lean{ComputableAnalysis.convexFTC}
\lean{ComputableAnalysis.concaveFTC}
\uses{def:exact-convexity,def:convex-right-derivative,thm:monotone-integrable,thm:effective-ftc-certificate,def:definite-integral-identity}
If \(F\) is convex on a rational interval \([a,b]\), then the right
derivative \(D_+F\) is integrable and
\[
  \int_a^b D_+F(x)\,\dd x \sim F(b)-F(a).
\]
Equivalently, the same conclusion holds with the left derivative \(D_-F\).
The intended Lean proof should manufacture the derivative-bound certificate
from the one-sided derivative inequalities below, then feed it through
\code{definiteIdentityFor\_of\_derivativeBoundFTC}.  In cases where the
chosen algorithm is literally a scheduled left-Riemann sum, the same endpoint
comparison can also be exposed through \code{EffectiveFTC}.
\end{theorem}

\begin{proof}
For a partition \(P:a=x_0<\cdots<x_N=b\), convexity gives
\[
  D_+F(x_i)
  \le
  \Sec_F(x_i,x_{i+1})
  \le
  D_-F(x_{i+1})
\]
on each cell.  Multiplying by \(x_{i+1}-x_i\) and summing yields Darboux
lower and upper sums enclosing
\[
  \sum_i \bigl(F(x_{i+1})-F(x_i)\bigr)=F(b)-F(a).
\]
The gap between the sums is controlled by monotonicity of the one-sided
derivatives, and tends to zero under rational refinement.  Hence the integral
and the endpoint difference represent the same raw real.
\end{proof}

\begin{theorem}[Formula identification closure]
\label{thm:formula-identification-ftc}
\uses{def:formula-identification,thm:convex-ftc}
If \(g\) is identified with \(D_+F\) on \([a,b]\), then
\[
  \int_a^b g(x)\,\dd x \sim F(b)-F(a).
\]
\end{theorem}

\begin{remark}[Piecewise convexity]\label{rem:no-piecewise-ftc}
\uses{thm:convex-ftc}
There is no separate named theorem for piecewise convex or concave functions.
In an example proof, split the interval at rational breakpoints where the
convexity proof changes, apply \cref{thm:convex-ftc} on each piece, and
combine the endpoint equalities by raw-real arithmetic and transitivity.
\end{remark}

    \frac{(-1)^{n+1}x^{2n+2}}{1+x^2}
  \right|
  \le |x|^{2n+2}.
\]
This is the finite algebra behind the arctangent Taylor remainder.
\end{theorem}
