\chapter{Exponential and Logarithm}
\label{ch:exponential-logarithm}

This chapter develops exponential and logarithm after the integral and series
chapters.  The intended order is important: the logarithmic-integral
representation of exponential depends on Chapter~\ref{ch:integrals}, while the
power-series representation depends on Chapter~\ref{ch:infinite-series}.
The guiding sanity check is the same as for \(\pi\): independent computations
of the same constant or function should be proved equivalent as raw reals.

\section{Positive-base powers and the Euler base}

\begin{definition}[Positive base and repeated multiplication]
\label{def:positive-base-natural-powers}
\lean{ComputableAnalysis.exp.PositiveRealRaw.natPow}
\uses{def:raw-real}
A positive computable real is represented by a valid raw real together with
one positive rational lower bound that holds for every interval stage.  This
is the constructive positivity condition needed to make powers meaningful;
it does not appeal to an unrepresented completed real line.

For any such base \(b\), natural powers are defined by repeated
multiplication:
\[
 b^0=1,\qquad b^{n+1}=b^n b.
\]
The raw interval multiplication is the same finite rational interval
arithmetic used throughout the foundation.  Thus this definition works for
an arbitrary positive computable real base, not merely for a rational base.
\end{definition}

\begin{definition}[Rational powers]
\label{def:rational-powers}
\lean{ComputableAnalysis.exp.RationalPowerExtension}
\uses{def:positive-base-natural-powers}
A rational-power extension for \(b\) supplies a valid positive raw real
\(b^q\) for every \(q\in\mathbb Q\), with a certified positive rational
lower bound at every interval stage; it agrees with repeated multiplication
at natural exponents and satisfies
\[
 b^{p+q}=b^p b^q.
\]
For \(p\in\mathbb N\) and \(q>0\), its denominator law is the certified
root equation
\[
 \left(b^{p/q}\right)^q=b^p.
\]
Negative rational exponents are then governed by the additive law.  This is
an interface for a computation and its proof obligations: constructing the
positive root algorithms and proving their validity remains explicit work,
rather than being hidden in a completeness axiom.
\end{definition}

\begin{remark}[Continuity and extension from rational names]
\label{rem:rational-power-continuity-extension}
The Lean evaluator is presented on rational names, but this is not meant to
restrict the mathematical domain of an exponential or power function.  The
separate predicate
\code{EpsilonDeltaContinuousOn} is the literal rational
\(\varepsilon\)-\(\delta\) obligation: for every positive rational
\(\varepsilon\), it supplies a positive rational \(\delta\) and an evaluation
precision such that rational inputs within \(\delta\) have output boxes within
\(\varepsilon\) of each other, each of width at most \(\varepsilon\).  The
boxes need not overlap: nearby inputs of a nonconstant function can have
distinct exact rational output values.  A rational-power exponential must
carry this certificate on every rational interval.

This is the data from which a later completion theorem may extend a continuous
function from rational names to all computable real inputs, or, for the
complex theory, to a specified open computable domain.  Discontinuous partial
functions are not entitled to such an extension.  The extension construction
itself remains later work; no topology or completeness theorem is used in the
finite certificate stated here.
\end{remark}

\begin{definition}[Euler base from an exponential]
\label{def:e-from-exponential-at-one}
\lean{ComputableAnalysis.exp.ExponentialFunction.eAtOne}
\uses{def:rational-powers}
An exponential representation is a computational presentation at rational
names that agrees with one rational-power extension and carries its
continuity-in-exponent certificate.  Its Euler base is, by definition, its
value at one:
\[
 e_E=E(1).
\]
This single definition applies to every eventual construction of the
exponential function.  The power-series, compound-interest, and inverse-log
constructions below are therefore intended to be compared as representatives
of the same \(e_E\), not as unrelated constants.
\end{definition}

\begin{theorem}[Derivative characterizations of the Euler base]
\label{thm:e-self-derivative-and-unit-slope}
\uses{def:rational-powers,def:e-from-exponential-at-one}
There are two complementary characterizations to prove constructively:
\[
 \frac{d}{dx}e^x=e^x,
 \qquad
 \left.\frac{d}{dx}b^x\right|_{x=0}=1
 \Longleftrightarrow b=e.
\]
The first statement is represented by an interval-valued derivative
certificate on each rational interval; the second uses rational difference
quotients at zero.  Lean already proves the formal coefficient identity
\(\exp' = \exp\).  The remaining analytic theorem is to turn the certified
power-series tail bounds into the corresponding derivative certificate, whose
finite-difference and derivative boxes are within the requested tolerance
rather than required to overlap exactly, and then establish the displayed
uniqueness statement by finite rational estimates.  More generally, this is
the scalar case of the linear Picard--Lindel\"of construction target in
\cref{thm:linear-picard-lindelof}: the Peano--Baker series for the
one-dimensional coefficient \(A=1\) provides the solution and its factorial
tail, while the zero-initial homogeneous case proves uniqueness.  Thus neither
the existence construction nor the uniqueness argument needs completeness of
an ambient real-number type.  A general nonlinear Picard--Lindel\"of theorem
is a later extension requiring interval-certified Lipschitz maps and a
separate contraction argument.
\end{theorem}

\begin{definition}[Logarithm]
\label{def:logarithm-integral}
\lean{ComputableAnalysis.exp.logFromIntegral}
\uses{def:e-log-integral,def:real-fun-raw}
The logarithm is first the positive-branch integral
\[
  \log x = \int_1^x \frac{dt}{t}.
\]
On any chosen positive branch this is a \code{FunctionOnInterval}, hence an
instance of \cref{def:real-fun-raw}: for rational \(x\) in the branch interval
and stage \(n\), the integral construction returns a rational interval, and for
fixed \(x\) these intervals form a raw real.  Its inverse branch is
exponential.  Later algebraic identities such as \(\log(xy)=\log x+\log y\)
and \(\exp(x+y)=\exp(x)\exp(y)\) should be proved from uniqueness of the
inverse and then transported between representations.
\end{definition}

\begin{proposition}[Positive reciprocal continuity at two]
\label{prop:one-over-x-one-two-continuity}
\lean{ComputableAnalysis.Logarithm.oneOverXOnOneTwo_intervalRegular}
\lean{ComputableAnalysis.Logarithm.oneOverXOnOneTwo_epsilonDeltaContinuous}
\uses{def:logarithm-integral}
The kernel \(x\mapsto1/x\) on \([1,2]\) has a concrete interval evaluator:
for a rational input interval \(I=[p,r]\subseteq[1,2]\), it returns
\[
 \left[\frac1r,\frac1p\right].
\]
It contains every exact rational point value in \(I\).  Its width is
\[
 \frac1p-\frac1r
   =(r-p)\frac1{pr}\le r-p,
\]
so the input modulus \(r-p\le1/(n+1)\) produces an output box of width at
most \(1/(n+1)\).  Lean therefore proves the literal rational
\(\varepsilon\)--\(\delta\) continuity predicate for this kernel, without
topology or real-number completeness.  This interval evaluator is a stronger
domain-local continuity certificate than is needed for the first literal
integral below.  Extending that integral construction to arbitrary
\code{IntervalRegularOn} kernels remains later work.
\end{proposition}

\begin{definition}[Finite reciprocal Darboux integral at two]
\label{def:log-two-reciprocal-darboux}
\lean{ComputableAnalysis.Logarithm.logTwoKernel_lipschitz_on_unit}

Put \(g(t)=1/(1+t)\) on \([0,1]\), the affine translate of \(1/x\) on

\[
 |g(s)-g(t)|\le |t-s|\qquad(0\le s,t\le1).
\]

For every rational kernel with a supplied natural Lipschitz constant \(L\),
the generic constructor forms the lower and upper rectangles on each cell of
the midpoint-refined dyadic mesh.  The resulting boxes are nested, valid raw
real data, and have exact width
\[
 w_n=\frac{2L}{2^n}.
\]
For \(g\), the constant is \(L=1\); hence
\code{logTwoReciprocalIntegral} is a genuine finite rectangle construction
with \(w_n=2/2^n\).  It is the raw integral associated with
\(\int_0^1dt/(1+t)=\int_1^2dx/x\), not a placeholder interface and not an
appeal to a completed integral.  The change-of-variable and series comparison
theorems are intentionally separate: this definition alone does not name the
result \(\log 2\).

\begin{proposition}[Finite logarithm-series/Riemann bridge]
\label{prop:log-two-series-right-riemann}

\begin{remark}[A deliberately long ODE--exponential--logarithm--pi route]
\label{rem:ode-exp-log-pi-route}
The project should make one pi equivalence exercise the elementary-function
and ODE layers rather than obtain every pi row from arctangent alone.  The
natural route is the integration-by-parts identity
\[
  \pi
   =4\int_0^1\arctan(x)\,\dd x+2\log 2.
\]
Its intended proof order is:
\begin{enumerate}
  \item construct the continuous Peano--Baker state transition and its
  factorial tail; \cref{thm:linear-picard-lindelof} then gives constructive
  existence and uniqueness for \(x'=A(t)x+b(t)\);
  \item specialize to \(f'=f,\ f(0)=1\).  Once the power-series,
  Euler-limit, and inverse-logarithmic-integral exponentials all satisfy this
  problem, uniqueness proves that the three exponential representations
  agree;
  \item use the inverse-function and FTC certificates to identify the
  positive logarithmic integral with the inverse of that canonical
  exponential.  The concrete finite comparison at two is already checked by
  \cref{thm:log-two-series-reciprocal-integral}; transport that equality to
  the resulting canonical \(\log 2\);
  \item prove the arctangent derivative and instantiate the common
  increasing/decreasing-piece integration-by-parts certificate of
  \cref{thm:integration-by-parts-certificate}.  The rational
  identity
  \[
    \int_0^1\arctan(x)\,\dd x
      =\frac{\pi}{4}-\frac12\log 2
  \]
  follows from \((x\arctan x)'=\arctan x+x/(1+x^2)\), its finite
  corner-rectangle schedule, and the logarithmic endpoint calculation;
  \item package the right-hand side as a valid raw pi evaluator and prove it
  equivalent to the geometric circle-area presentation.
\end{enumerate}
Thus this one final equality deliberately depends on exp/log agreement and
therefore on linear ODE uniqueness.  It is not currently a scoreboard cell:
every arrow above must be a checked finite certificate first.  A later,
strictly longer complex corroboration is
\(\pi=-2i\log(i)\); it additionally needs extension from rational names to
represented complex inputs and the agreement of the circular and exponential
solutions of the \(2\)-dimensional rotation system.
\end{remark}

\section{Euler's identity}

\begin{theorem}[Euler identity]
\label{thm:euler-identity}
\uses{def:geometric-sin-cos-functionraw,def:exp-representations}
The exponential power series over rational complex boxes should agree with the
geometric sine and cosine functions from Chapter~\ref{ch:rational-circle-trigonometry}:
\[
  \exp(i\theta)=\cos(\theta)+i\sin(\theta).
\]
With the normalized quarter-turn convention this is stated for rational
\(t\), where \(\theta=t\pi/2\).  The proof route is to first prove the formal
power-series identity, then use the already established agreement between
power-series and geometric trigonometry.
\end{theorem}
