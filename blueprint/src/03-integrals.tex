\chapter{Integrals}
\label{ch:integrals}

Each integral begins as a rational-box algorithm.  Its separate validity
certificate is what makes the output a \code{RealRaw}; there is no ambient
completed-real integral or table of antiderivatives.

\section{From \texorpdfstring{\code{arctan.geom}}{arctan.geom} to definite integrals}

\begin{remark}[The geometric prototype]
\label{rem:arctan-geom-general-integral}
\code{arctan.geom} already encloses area by rational cells.  Replace the
circle density by a particular function and certify that its boxes shrink.
\end{remark}

\begin{definition}[Monotone-function integral raw]
\label{def:increasing-integral-stage}
\lean{ComputableAnalysis.NondecreasingOnInterval}
\lean{ComputableAnalysis.IntervalRegularOn}
\lean{ComputableAnalysis.Integral.nondecreasingDarbouxRange}
\lean{ComputableAnalysis.Integral.nondecreasingDarbouxRange_width_nonneg}
\lean{ComputableAnalysis.Integral.nondecreasingDarbouxStage}
\lean{ComputableAnalysis.Integral.nondecreasingDarbouxDyadicStage}
\uses{rem:arctan-geom-general-integral,def:interval-regular}
For a nondecreasing \(F\) on \([a,b]\), use opposite endpoints of a dyadic
partition and retain the value boxes:
\begin{verbatim}
def increasing_integral_stage(F, a, b, n):
    P = uniform_partition(a, b, 2**n)
    q = evaluation_stage(F, n)
    lower = sum((P[i+1]-P[i]) * F(P[i], q).lo for i in range(len(P)-1))
    upper = sum((P[i+1]-P[i]) * F(P[i+1], q).hi for i in range(len(P)-1))
    return [lower, upper]
\end{verbatim}
For a decreasing piece, interchange the endpoints.  All samples are rational
intervals, including singleton values when those are available.
\end{definition}

 s(t)=\frac{\sin(\pi t)}{\pi t},\qquad s(0)=1,\qquad 0\le t\le2

def piecewise_monotone_stage(F, monotone_pieces, turn_brackets, n):
    tails = [monotone_stage(F, P.left(n), P.right(n), n, P.direction)

\end{verbatim}

def substitution_stage(phi, t0, t1, n):

    X = [phi(t) for t in T]
    return T, X                 # equal t divisions; unequal x divisions

\section{Integration by parts: paired subdivisions}

\begin{theorem}[Finite geometric integration by parts]
\label{thm:finite-geometric-integration-by-parts}
\lean{ComputableAnalysis.finiteIntegrationByParts}
\lean{ComputableAnalysis.finiteIntegrationByParts_withVariation}
\lean{ComputableAnalysis.RationalPartition.commonRefinementOfPartitions}
\lean{ComputableAnalysis.unitMeshPath_quadraticVariation_le_epsilon}
\uses{def:finite-substitution-stage}
\leanok
Fix \([t_0,t_1]\), set
\(t_i=t_0+i(t_1-t_0)/2^n\), and sample \(f(t_i)\), \(g(t_i)\).
The two finite sums use this same equal subdivision.  In the picture, affine
\(f\) gives equal horizontal divisions while quadratic \(g\) gives unequal
vertical divisions:
\begin{verbatim}
def parts_stage(f, g, t0, t1, n):
    T = uniform_partition(t0, t1, 2**n)
    S = [(f(t), g(t)) for t in T]
    return horizontal_then_vertical(S), vertical_then_horizontal(S)
\end{verbatim}
\begin{figure}[!htbp]
\centering
\integrationByPartsAnimation
\caption{A common equal subdivision of \([t_0,t_1]\): paired zigzags through
the same \((f(t_i),g(t_i))\), with their corner discrepancy shrinking.}
\label{fig:integration-by-parts-cell}
\end{figure}
The checked finite identity is
\[
 \sum_i f(t_i)\bigl(g(t_{i+1})-g(t_i)\bigr)+
 \sum_i g(t_{i+1})\bigl(f(t_{i+1})-f(t_i)\bigr)
 =f(t_1)g(t_1)-f(t_0)g(t_0).
\]
\end{theorem}

\begin{definition}[Integration-by-parts certificate on rational pieces]
\label{def:integration-by-parts-certificate}
\uses{thm:finite-geometric-integration-by-parts,def:definite-integral-identity}
For rational monotone pieces, provide:
\begin{verbatim}
piece: integral(f*g'), integral(f'*g), product(f,g)
piece: directions, common_refinement, increment_moduli
stage: rational error budget for boxes + quadrature + corners
\end{verbatim}
This is the general interface; it is not yet a universal theorem.
\end{definition}

\begin{theorem}[Integration by parts from the certificate: construction target]
\label{thm:integration-by-parts-certificate}
\uses{def:integration-by-parts-certificate}
The target is
\[
 \int_{t_0}^{t_1} f(t)g'(t)\,\dd t\ \Equiv\
 f(t_1)g(t_1)-f(t_0)g(t_0)-\int_{t_0}^{t_1} f'(t)g(t)\,\dd t.
\]
\end{theorem}
