  10 def segmentLogValue (x : PositiveInput) (t : Rat) : RealRaw :=
  11   if ht : 0≤t ∧ t≤1 then (log (positivePoint x ⟨t,ht⟩)).val else RealRaw.ofRat 0

  13 def reciprocalBounds (x : PositiveInput) (eps : QPos) : Integral.Bounds (segmentIntegrand x) :=
  14   lipschitzBounds (segmentIntegrand x) rfl rfl (segmentVariation x) (segmentVariation_nonnegative x)
  15     (segmentIntegrand_lipschitz x) eps

  17 theorem reciprocalBounds_gap (x : PositiveInput) (eps : QPos) :
  18     (reciprocalBounds x eps).upperSum-(reciprocalBounds x eps).lowerSum≤eps.val :=
  19   lipschitzBounds_gap (segmentIntegrand x) rfl rfl _ _ _ eps

  21 theorem segmentIntegrand_bound (x : PositiveInput) (t : RationalUnit) :
  22     Small (realAxis (segmentIntegrandValue x t)).val (2*segmentBound x*(1/(segmentLower x).val)) := by
  23   have hR : 0≤1/(segmentLower x).val := by
  24     rw [Rat.div_def,Rat.one_mul]; exact Rat.le_of_lt (Rat.inv_pos.mpr (segmentLower x).property)
  25   exact Small.congr
  26     (mul_valid (realAxis (segmentDisplacement x)).property (segmentInverse x t).property)
  27     (realAxis (segmentIntegrandValue x t)).property (segmentIntegrand_embedding x t)
  28     (Small.mul (realAxis (segmentDisplacement x)).property (segmentInverse x t).property
  29       (Rat.le_of_lt (segmentBound_pos x)) hR (scalar_small _) (segmentInverse_bound x t))

  31 theorem segmentLog_cellBounds (x : PositiveInput) :
  32     Integral.CellDerivativeBounds (segmentIntegrand x) (segmentLogValue x) := by
  33   apply cellBounds_of_quadratic (segmentIntegrand x) (segmentLogValue x) rfl rfl
  34     (fun t ht0 ht1 => by simp only [segmentLogValue,dif_pos (And.intro ht0 ht1)]; exact (log _).property)
  35     (segmentRemainderConstant x) (by
  36       dsimp [segmentRemainderConstant]
  37       exact Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) (Rat.pow_nonneg (Rat.le_of_lt (segmentBound_pos x))))
  38         (Rat.pow_nonneg (by rw [Rat.div_def,Rat.one_mul]; exact Rat.le_of_lt (Rat.inv_pos.mpr (segmentLower x).property))))
  39     (segmentMesh x) (2*segmentBound x*(1/(segmentLower x).val))
  40   · intro t ht n
  41     have h := segmentIntegrand_bound x ⟨t,ht⟩
  42     rw [segmentIntegrand_compute x ⟨t,ht⟩ n]
  43     exact ⟨h.1 0 n,h.2.1 n 0⟩
  44   · intro u v hu hu0 huv hv1 hmesh
  45     have hum : 0≤u ∧ u≤1 := ⟨hu0,by grind only⟩
  46     have hvm : 0≤v ∧ v≤1 := ⟨by grind only,hv1⟩
  47     have h := segment_log_remainder x ⟨u,hum⟩ ⟨v,hvm⟩ huv hmesh
  48     have hcompute : (segmentIntegrand x).compute u hu = (segmentIntegrandValue x ⟨u,hum⟩).val.compute := by
  49       funext n
  50       exact segmentIntegrand_compute x ⟨u,hum⟩ n
  51     simpa only [Small,RealRaw.Le,ofRealRaw,realPart,imagPart,RealRaw.sub,RealRaw.subCompute,
  52       RealRaw.scaleRat,RealRaw.scaleRatCompute,segmentLogValue,dif_pos hum,dif_pos hvm,hcompute] using h

  54 /-- A concrete existence proof for the integral; the endpoint law follows
  55 from finite subdivision and tight reciprocal bounds. -/
  56 theorem segmentLog_hasIntegral (x : PositiveInput) : Integral.HasIntegral (segmentIntegrand x)
  57     (RealRaw.sub (segmentLogValue x 1) (segmentLogValue x 0)) :=
  58   (segmentLog_cellBounds x).hasIntegral (by change (0:Rat)≤1; decide +kernel)
  59     (fun eps => ⟨reciprocalBounds x eps,reciprocalBounds_gap x eps⟩)

  61 def oneInput : PositiveInput := ⟨⟨RealRaw.ofRat 1,RealRaw.ofRat_valid _⟩,⟨0,by decide +kernel⟩⟩
  62 def zeroInput : RealInput := ⟨RealRaw.ofRat 0,RealRaw.ofRat_valid _⟩
  63 
  64 theorem exp_zero : (exp zeroInput).val.Equiv oneInput.val.val := realPart_equiv entireExponential_zero
  65 
  66 theorem log_one : (log oneInput).val.Equiv zeroInput.val :=
  67   RealRaw.equiv_symm (log_unique oneInput zeroInput exp_zero)

  69 theorem positiveSegment_zero (x : PositiveInput) :
  70     (positiveSegment x ⟨0,by decide +kernel⟩).val.Equiv oneInput.val.val := by
  71   intro n
  72   apply (RealRaw.compareAt_overlap_iff _ _ n n).mpr
  73   simp only [positiveSegment,RealRaw.add,RealRaw.addCompute,RealRaw.scaleRat,RealRaw.scaleRatCompute,
  74     RealRaw.ofRat,if_pos (show (0:Rat)≤0 by decide +kernel),Rat.zero_mul,Rat.add_zero,oneInput,show (1:Rat)-0=1 by decide +kernel]
  75   exact ⟨Rat.le_refl,Rat.le_refl⟩
  76 
  77 theorem positiveSegment_one (x : PositiveInput) :
  78     (positiveSegment x ⟨1,by decide +kernel⟩).val.Equiv x.val.val := by
  79   intro n
  80   apply (RealRaw.compareAt_overlap_iff _ _ n n).mpr
  81   have ho := RealRaw.interval_order_of_valid _ x.val.property n
  82   simp only [positiveSegment,RealRaw.add,RealRaw.addCompute,RealRaw.scaleRat,RealRaw.scaleRatCompute,
  83     RealRaw.ofRat,if_pos (show (0:Rat)≤1 by decide +kernel),Rat.one_mul,Rat.sub_self,Rat.zero_add]
  84   exact ⟨ho,ho⟩

  86 theorem segmentLog_endpoint (x : PositiveInput) :
  87     (RealRaw.sub (segmentLogValue x 1) (segmentLogValue x 0)).Equiv (log x).val := by
  88   have h0 := RealRaw.equiv_trans (log (positivePoint x ⟨0,by decide +kernel⟩)).property
  89     (log oneInput).property zeroInput.property
  90     (log_congr _ _ (positiveSegment_zero x)) log_one
  91   have h1 := log_congr (positivePoint x ⟨1,by decide +kernel⟩) x (positiveSegment_one x)
  92   have h0' : (segmentLogValue x 0).Equiv zeroInput.val := by
  93     change (log (positivePoint x ⟨0,by decide +kernel⟩)).val.Equiv zeroInput.val
  94     exact h0
  95   have h1' : (segmentLogValue x 1).Equiv (log x).val := by
  96     change (log (positivePoint x ⟨1,by decide +kernel⟩)).val.Equiv (log x).val
  97     exact h1
  98   have hv0 : (segmentLogValue x 0).Valid := (log (positivePoint x ⟨0,by decide +kernel⟩)).property
  99   have hv1 : (segmentLogValue x 1).Valid := (log (positivePoint x ⟨1,by decide +kernel⟩)).property
 100   have hh := RealRaw.sub_equiv (x := segmentLogValue x 1) (x' := (log x).val)
 101     (y := segmentLogValue x 0) (y' := zeroInput.val) hv1 (log x).property hv0 zeroInput.property h1' h0'
 102   have hz : (RealRaw.sub (log x).val zeroInput.val).Equiv (log x).val := by
 103     intro n
 104     apply (RealRaw.compareAt_overlap_iff _ _ n n).mpr
 105     have ho := RealRaw.interval_order_of_valid _ (log x).property n
 106     change ((log x).val.compute n).lo-(0:Rat)≤((log x).val.compute n).hi ∧
 107       ((log x).val.compute n).lo≤((log x).val.compute n).hi-(0:Rat)
 108     constructor <;> grind only
 109   exact RealRaw.equiv_trans (RealRaw.sub_valid hv1 hv0)
 110     (RealRaw.sub_valid (log x).property zeroInput.property) (log x).property hh hz

 112 /-- Independent rectangle-sum computation of the normalized reciprocal
 113 integral. The pullback is (x-1)/(1+t(x-1)), so x<1 is oriented correctly. -/
 114 def integralLog (x : PositiveInput) : RealInput :=
 115   ⟨rectangleValue (segmentIntegrand x) (reciprocalBounds x),
 116     rectangleValue_valid _ _ (reciprocalBounds_gap x) _ (segmentLog_hasIntegral x)⟩
 117 
 118 theorem integralLog_hasIntegral (x : PositiveInput) :
 119     Integral.HasIntegral (segmentIntegrand x) (integralLog x).val :=
 120   rectangleValue_hasIntegral _ _ (reciprocalBounds_gap x) _ (segmentLog_hasIntegral x)
 121 
 122 /-- Exact global identification of the reciprocal integral with logarithm. -/
 123 theorem integralLog_equiv_log (x : PositiveInput) : (integralLog x).val.Equiv (log x).val :=
 124   RealRaw.equiv_trans (integralLog x).property (segmentLog_hasIntegral x).valid (log x).property
 125     (rectangleValue_agrees _ _ _ (segmentLog_hasIntegral x)) (segmentLog_endpoint x)
 126 
 127 theorem integralLog_congr (x y : PositiveInput) (h : x.val.val.Equiv y.val.val) :
 128     (integralLog x).val.Equiv (integralLog y).val :=
 129   RealRaw.equiv_trans (integralLog x).property (log x).property (integralLog y).property
 130     (integralLog_equiv_log x) (RealRaw.equiv_trans (log x).property (log y).property
 131       (integralLog y).property (log_congr x y h) (RealRaw.equiv_symm (integralLog_equiv_log y)))
 132 
 133 /-- Exponential is an actual two-sided inverse of the constructed integral. -/
 134 theorem integralLog_exp (x : RealInput) :
 135     (integralLog ⟨exp x,exp_positive x⟩).val.Equiv x.val :=
 136   RealRaw.equiv_trans (integralLog _).property (log _).property x.property (integralLog_equiv_log _) (log_exp x)
 137 
 138 theorem exp_integralLog (x : PositiveInput) : (exp (integralLog x)).val.Equiv x.val.val :=
 139   RealRaw.equiv_trans (exp (integralLog x)).property (exp (log x)).property x.val.property
 140     (exp_congr _ _ (integralLog_equiv_log x)) (exp_log x)
 141 
 142 /-- Any real computation inverted by the reciprocal integral agrees with
 143 exponential; this law supports independently justified inversion methods. -/
 144 theorem exp_unique_of_integralLog (x : RealInput) (y : PositiveInput)
 145     (h : (integralLog y).val.Equiv x.val) : y.val.val.Equiv (exp x).val :=
 146   exp_unique_of_log x y (RealRaw.equiv_trans (log y).property (integralLog y).property x.property
 147     (RealRaw.equiv_symm (integralLog_equiv_log y)) h)
 148 end ComputableAnalysis.ExponentialComputations
