   6 /-! Compound-interest approximations at arbitrary represented complex inputs.
   7 The actual local logarithm supplies a quadratic error. Its inverse identity
   8 and the exponential addition law turn this into a quantitative limit proof. -/

  13 /-- A literal finite compound-interest power, including negative/complex inputs. -/
  14 def compoundApproximation (z : Scalar) (n : Nat) : Scalar :=
  15   ⟨power (add one (scaleRat (1/((n+1:Nat):Rat)) z.val)) (n+1),
  16     power_valid _ (add_valid (ofQComplex_valid _) (scaleRat_valid z.property)) _⟩
  17 
  18 def compoundRadius (z : Scalar) : QPos :=
  19   ⟨2*scalarBound z+1,by have h := scalarBound_pos z; grind only⟩
  20 
  21 def compoundErrorConstant (z : Scalar) : Rat :=
  22   256*exponentialBudget (exponentialRatio (compoundRadius z))*
  23     (exponentialRatio (compoundRadius z))^2*(scalarBound z)^2
  24 
  25 theorem compoundErrorConstant_nonnegative (z : Scalar) : 0 ≤ compoundErrorConstant z := by
  26   unfold compoundErrorConstant
  27   exact Rat.mul_nonneg (Rat.mul_nonneg
  28     (Rat.mul_nonneg (by decide +kernel)
  29       (exponentialBudget_nonnegative _ (exponentialRatio_positive _)))
  30     (Rat.pow_nonneg (Rat.le_of_lt (exponentialRatio_positive _))))
  31     (Rat.pow_nonneg (Rat.le_of_lt (scalarBound_pos z)))

  33 /-- Explicit error for every sufficiently fine compound-interest power.
  34 The hypothesis is just a computable lower bound on the finite mesh size. -/
  35 theorem compoundApproximation_error (z : Scalar) (n : Nat)
  36     (hmesh : scalarBound z/((n+1:Nat):Rat) ≤ (1:Rat)/128) :
  37     Small (sub (compoundApproximation z n).val (entireExponentialValue z).val)
  38       (compoundErrorConstant z/((n+1:Nat):Rat)) := by
  39   let C := scalarBound z
  40   let m : Rat := ((n+1:Nat):Rat)
  41   have hm : 0<m := Rat.natCast_pos.mpr (Nat.succ_pos n)
  42   have hi : 0≤1/m := Rat.le_of_lt (by rw [Rat.div_def,Rat.one_mul]; exact Rat.inv_pos.mpr hm)
  43   have hC : 0<C := scalarBound_pos z
  44   have hC0 := Rat.le_of_lt hC
  45   let w : Scalar := ⟨scaleRat (1/m) z.val,scaleRat_valid z.property⟩
  46   have hw : Small w.val (C/m) := by
  47     have h := small_scale hi (scalar_small z)
  48     change Small w.val ((1/m)*C) at h
  49     have he : (1/m)*C=C/m := by grind [Rat.div_def]
  50     rw [he] at h
  51     exact h
  52   have hs : 0≤C/m := Rat.mul_nonneg hC0 (Rat.le_of_lt (Rat.inv_pos.mpr hm))
  53   have hd : interior LocalLogarithm.radius.val w :=
  54     ⟨C/m,hs,by change C/m≤(1:Rat)/128 at hmesh; change C/m<(1:Rat)/32; grind only,hw⟩
  55   let l := LocalLogarithm.function.eval w hd
  56   let a : Scalar := ⟨scaleRat m l.val,scaleRat_valid l.property⟩
  57   let E := 16*C^2/m
  58   have hE : 0≤E := Rat.mul_nonneg
  59     (Rat.mul_nonneg (by decide +kernel) (Rat.pow_nonneg hC0)) (Rat.le_of_lt (Rat.inv_pos.mpr hm))
  60   have hlinear := localLogarithm_linear_error w hd (C/m) hs hw
  61   have hscale := small_scale (Rat.le_of_lt hm) hlinear
  62   have he : (scaleRat m (sub l.val w.val)).Equiv (sub a.val z.val) := by
  63     apply ComplexRawQuotient.equiv_of_ofRaw_eq
  64       (hleft := scaleRat_valid (sub_valid l.property w.property))
  65       (hright := sub_valid a.property z.property)
  66     let L := ComplexRawQuotient.ofRaw l.val l.property
  67     let Z := ComplexRawQuotient.ofRaw z.val z.property
  68     change ComplexRawQuotient.scaleRat m (L-ComplexRawQuotient.scaleRat (1/m) Z)=
  69       ComplexRawQuotient.scaleRat m L-Z
  70     change ComplexRawQuotient.scaleRat m (L+ -ComplexRawQuotient.scaleRat (1/m) Z)=
  71       ComplexRawQuotient.scaleRat m L+ -Z
  72     have hneg (Y : ComplexRawQuotient.Value) :
  73         ComplexRawQuotient.scaleRat m (-Y)= -(ComplexRawQuotient.scaleRat m Y) := by
  74       rw [ComplexRawQuotient.neg_eq_scaleRat_neg_one,ComplexRawQuotient.scaleRat_scaleRat,
  75         ComplexRawQuotient.neg_eq_scaleRat_neg_one,ComplexRawQuotient.scaleRat_scaleRat]
  76       congr 1
  77       grind only
  78     rw [ComplexRawQuotient.scaleRat_add,hneg,ComplexRawQuotient.scaleRat_scaleRat]
  79     rw [show m*(1/m)=1 by have h := Rat.mul_inv_cancel m (Rat.ne_of_gt hm); grind [Rat.div_def],
  80       ComplexRawQuotient.scaleRat_one]
  81   have herr : Small (sub a.val z.val) E := by
  82     have hh := Small.congr (scaleRat_valid (sub_valid l.property w.property))
  83       (sub_valid a.property z.property) he hscale
  84     have hiid := Rat.mul_inv_cancel m (Rat.ne_of_gt hm)
  85     have heB : m*(16*(C/m)^2)=E := by
  86       dsimp [E]
  87       rw [Rat.pow_succ,Rat.pow_succ,Rat.pow_zero,Rat.pow_succ,Rat.pow_succ,Rat.pow_zero]
  88       grind [Rat.div_def]
  89     rw [heB] at hh
  90     exact hh
  91   have heC : E≤C := by
  92     have h := Rat.mul_le_mul_of_nonneg_left hmesh (show 0≤16*C by grind only)
  93     change 16*C*(C/m)≤16*C*((1:Rat)/128) at h
  94     dsimp [E]; rw [Rat.pow_succ,Rat.pow_succ,Rat.pow_zero]; grind [Rat.div_def]
  95   have haSmall : Small a.val (2*C) := by
  96     have h := small_add herr (scalar_small z)
  97     have heq : (add (sub a.val z.val) z.val).Equiv a.val := by
  98       apply ComplexRawQuotient.equiv_of_ofRaw_eq
  99         (hleft := add_valid (sub_valid a.property z.property) z.property) (hright := a.property)
 100       change (ComplexRawQuotient.ofRaw a.val a.property-ComplexRawQuotient.ofRaw z.val z.property)+
 101         ComplexRawQuotient.ofRaw z.val z.property=ComplexRawQuotient.ofRaw a.val a.property
 102       grind only
 103     exact Small.mono (Small.congr
 104       (add_valid (sub_valid a.property z.property) z.property) a.property heq h) (by grind only)
 105   have hzR : (exponentialChart (compoundRadius z)).domain z :=
 106     ⟨C,hC0,by dsimp [compoundRadius,C]; grind only,scalar_small z⟩
 107   have haR : (exponentialChart (compoundRadius z)).domain a :=
 108     ⟨2*C,by grind only,by dsimp [compoundRadius,C]; grind only,haSmall⟩
 109   have hcontinuity := entireExponential_lipschitz (compoundRadius z) z a hzR haR E hE herr
 110   have hpow := power_congr (entireExponentialValue l).val (LocalLogarithm.onePlus w).val
 111     (entireExponentialValue l).property (LocalLogarithm.onePlus w).property
 112     (entireExponential_localLogarithm w hd) (n+1)
 113   have hnat := entireExponential_nat_multiple l (n+1)
 114   have happrox : (compoundApproximation z n).val.Equiv (entireExponentialValue a).val :=
 115     equiv_symm (equiv_trans (entireExponentialValue a).property
 116       (power_valid _ (entireExponentialValue l).property _) (compoundApproximation z n).property
 117       hnat hpow)
 118   have hh := Small.congr
 119     (sub_valid (entireExponentialValue a).property (entireExponentialValue z).property)
 120     (sub_valid (compoundApproximation z n).property (entireExponentialValue z).property)
 121     (FunctionTheory.sub_congr (equiv_symm happrox) (equiv_refl _ (entireExponentialValue z).property)) hcontinuity
 122   have heB : 16*exponentialBudget (exponentialRatio (compoundRadius z))*
 123       (exponentialRatio (compoundRadius z))^2*E=compoundErrorConstant z/((n+1:Nat):Rat) := by
 124     dsimp [compoundErrorConstant,E,m,C]
 125     grind [Rat.div_def]
 126   rw [heB] at hh
 127   exact hh

 129 /-- A finite initial mesh bound, computed from the input's rational box. -/
 130 def compoundStart (z : Scalar) : Nat := 128*((scalarBound z).num.natAbs+1)
 131 
 132 def compoundPrefix (z : Scalar) (k : Nat) : Scalar := compoundApproximation z (compoundStart z+k)
 133 def compoundError (z : Scalar) (k : Nat) : Rat := compoundErrorConstant z/((k+1:Nat):Rat)

 135 theorem reciprocal_antitone {a b : Rat} (ha : 0<a) (hab : a≤b) : 1/b≤1/a := by
 136   have hb : 0<b := by grind only
 137   apply Rat.le_of_mul_le_mul_right (c := a*b)
 138   · have heA := Rat.mul_inv_cancel a (Rat.ne_of_gt ha)
 139     have heB := Rat.mul_inv_cancel b (Rat.ne_of_gt hb)
 140     calc
 141       (1/b)*(a*b)=a := by rw [Rat.div_def]; grind [Rat.mul_assoc,Rat.mul_comm,Rat.mul_inv_cancel _ (Rat.ne_of_gt hb)]
 142       _≤b := hab
 143       _=(1/a)*(a*b) := by rw [Rat.div_def]; grind [Rat.mul_assoc,Rat.mul_comm,Rat.mul_inv_cancel _ (Rat.ne_of_gt ha)]
 144   · exact Rat.mul_pos ha hb

 146 theorem compoundPrefix_error (z : Scalar) (k : Nat) :
 147     Small (sub (compoundPrefix z k).val (entireExponentialValue z).val) (compoundError z k) := by
 148   let C := scalarBound z
 149   let B : Rat := (((scalarBound z).num.natAbs+1:Nat):Rat)
 150   let m : Rat := ((compoundStart z+k+1:Nat):Rat)
 151   have hC : C≤B := Rat.le_trans (self_le_qabs C) (qabs_le_numNatAbs_succ C)
 152   have hm : 0<m := Rat.natCast_pos.mpr (by omega)
 153   have hlarge : 128*B≤m := by
 154     have hn : compoundStart z≤compoundStart z+k+1 := by omega
 155     have hc : ((compoundStart z:Nat):Rat)≤m := by change ((compoundStart z):Rat)≤((compoundStart z+k+1:Nat):Rat); exact_mod_cast hn
 156     have he : ((compoundStart z):Rat)=128*B := by
 157       dsimp [compoundStart,B]
 158       rw [Rat.natCast_mul]
 159       rfl
 160     rw [he] at hc
 161     exact hc
 162   have him : 0≤m⁻¹ := Rat.le_of_lt (Rat.inv_pos.mpr hm)
 163   have hmul := Rat.mul_le_mul_of_nonneg_right (show 128*C≤m by grind only) him
 164   have he := Rat.mul_inv_cancel m (Rat.ne_of_gt hm)
 165   have hmesh : C/m≤(1:Rat)/128 := by grind [Rat.div_def]
 166   have herror := compoundApproximation_error z (compoundStart z+k) hmesh
 167   have hk : 0<((k+1:Nat):Rat) := Rat.natCast_pos.mpr (Nat.succ_pos k)
 168   have hkm : ((k+1:Nat):Rat)≤m := by change ((k+1:Nat):Rat)≤((compoundStart z+k+1:Nat):Rat); exact_mod_cast (show k+1≤compoundStart z+k+1 by omega)
 169   have hi := reciprocal_antitone hk hkm
 170   have hK := compoundErrorConstant_nonnegative z
 171   have hh := Rat.mul_le_mul_of_nonneg_left hi hK
 172   apply Small.mono herror
 173   simpa only [compoundError,Rat.div_def,Rat.one_mul] using hh

 175 theorem compoundError_antitone (z : Scalar) {k n : Nat} (hkn : k≤n) :
 176     compoundError z n≤compoundError z k := by
 177   have hk : 0<((k+1:Nat):Rat) := Rat.natCast_pos.mpr (Nat.succ_pos k)
 178   have hn : ((k+1:Nat):Rat)≤((n+1:Nat):Rat) := by exact_mod_cast (show k+1≤n+1 by omega)
 179   have h := Rat.mul_le_mul_of_nonneg_left (reciprocal_antitone hk hn)
 180     (compoundErrorConstant_nonnegative z)
 181   simpa only [compoundError,Rat.div_def,Rat.one_mul] using h

 183 theorem compoundError_shrinks (z : Scalar) : ShrinksToZero (compoundError z) :=
 184   shrinksToZero_of_ratOverSuccBound (fun _ => Rat.le_refl)

 186 theorem compoundPrefix_cauchy (z : Scalar) (k n : Nat) (hkn : k≤n) :
 187     Small (sub (compoundPrefix z n).val (compoundPrefix z k).val) (2*compoundError z k) := by
 188   have h := SeriesLimitLaws.small_sub (compoundPrefix_error z n) (compoundPrefix_error z k)
 189   have he : (sub (sub (compoundPrefix z n).val (entireExponentialValue z).val)
 190       (sub (compoundPrefix z k).val (entireExponentialValue z).val)).Equiv
 191       (sub (compoundPrefix z n).val (compoundPrefix z k).val) := by
 192     apply ComplexRawQuotient.equiv_of_ofRaw_eq
 193       (hleft := sub_valid (sub_valid (compoundPrefix z n).property (entireExponentialValue z).property)
 194         (sub_valid (compoundPrefix z k).property (entireExponentialValue z).property))
 195       (hright := sub_valid (compoundPrefix z n).property (compoundPrefix z k).property)
 196     change (ComplexRawQuotient.ofRaw (compoundPrefix z n).val (compoundPrefix z n).property-
 197       ComplexRawQuotient.ofRaw (entireExponentialValue z).val (entireExponentialValue z).property)-
 198       (ComplexRawQuotient.ofRaw (compoundPrefix z k).val (compoundPrefix z k).property-
 199       ComplexRawQuotient.ofRaw (entireExponentialValue z).val (entireExponentialValue z).property)=
 200       ComplexRawQuotient.ofRaw (compoundPrefix z n).val (compoundPrefix z n).property-
 201       ComplexRawQuotient.ofRaw (compoundPrefix z k).val (compoundPrefix z k).property
 202     grind only
 203   have hb := Small.congr
 204     (sub_valid (sub_valid (compoundPrefix z n).property (entireExponentialValue z).property)
 205       (sub_valid (compoundPrefix z k).property (entireExponentialValue z).property))
 206     (sub_valid (compoundPrefix z n).property (compoundPrefix z k).property) he h
 207   exact Small.mono hb (by have h := compoundError_antitone z hkn; grind only)

 209 /-- Executable limit from finite powers and shrinking error bounds. The
 210 runtime samples the compound powers; it does not evaluate the power series. -/
 211 def compoundValue (z : Scalar) : Scalar :=
 212   ⟨RepresentedCauchySum.value (fun k => (compoundPrefix z k).val)
 213       (fun k => (compoundPrefix z k).property) (fun k => 2*compoundError z k),
 214     RepresentedCauchySum.value_valid _ _ _
 215       (SeriesLimitLaws.shrinks_scale _ (compoundError_shrinks z) 2 (by decide +kernel))
 216       (compoundPrefix_cauchy z)⟩
 217 
 218 /-- Exact agreement of the compound-interest and power-series constructions
 219 on every valid represented complex input. -/
 220 theorem compoundValue_equiv_powerSeries (z : Scalar) :
 221     (compoundValue z).val.Equiv (entireExponentialValue z).val := by
 222   apply RepresentedCauchySum.unique (fun k => (compoundPrefix z k).val)
 223     (fun k => (compoundPrefix z k).property) (fun k => 2*compoundError z k)
 224     (SeriesLimitLaws.shrinks_scale _ (compoundError_shrinks z) 2 (by decide +kernel))
 225     _ _ (compoundValue z).property (entireExponentialValue z).property
 226   · exact RepresentedCauchySum.value_close_prefix _ _ _ (compoundPrefix_cauchy z)
 227   · intro k
 228     have hs := RepresentedCauchySum.small_sub_symm _ _ _ (compoundPrefix_error z k)
 229     have hnonneg : 0≤compoundError z k := Rat.mul_nonneg (compoundErrorConstant_nonnegative z)
 230       (Rat.le_of_lt (Rat.inv_pos.mpr (Rat.natCast_pos.mpr (Nat.succ_pos k))))
 231     exact Small.mono hs (by grind only)
 232 
 233 theorem compoundValue_congr (z w : Scalar) (hzw : z.val.Equiv w.val) :
 234     (compoundValue z).val.Equiv (compoundValue w).val :=
 235   equiv_trans (compoundValue z).property (entireExponentialValue z).property (compoundValue w).property
 236     (compoundValue_equiv_powerSeries z)
 237     (equiv_trans (entireExponentialValue z).property (entireExponentialValue w).property (compoundValue w).property
 238       (entireExponentialValue_congr z w hzw) (equiv_symm (compoundValue_equiv_powerSeries w)))
 239 end ComputableAnalysis.ExponentialComputations
