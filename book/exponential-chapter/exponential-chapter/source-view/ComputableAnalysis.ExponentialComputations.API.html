   6 /-! One function-level API for proved exponential computations.
   7 An implementation may be registered only after its exact agreement theorem
   8 has been proved. This is a comparison package, not an assumed characterization
   9 or an existence theorem. -/
  10 namespace ComputableAnalysis.ExponentialComputations
  11 open ComplexRaw FunctionTheory RiemannHilbert ModularForms
  12 
  13 structure Implementation where
  14   eval : Scalar → Scalar
  15   agrees : ∀ z, (eval z).val.Equiv (entireExponentialValue z).val
  16 
  17 namespace Implementation
  18 /-- Agreement is with a common reference, so callers need no pairwise table. -/
  19 theorem equivalent (f g : Implementation) (z : Scalar) :
  20     (f.eval z).val.Equiv (g.eval z).val :=
  21   equiv_trans (f.eval z).property (entireExponentialValue z).property (g.eval z).property
  22     (f.agrees z) (equiv_symm (g.agrees z))
  23 
  24 theorem congr (f : Implementation) (z w : Scalar) (h : z.val.Equiv w.val) :
  25     (f.eval z).val.Equiv (f.eval w).val :=
  26   equiv_trans (f.eval z).property (entireExponentialValue z).property (f.eval w).property
  27     (f.agrees z) (equiv_trans (entireExponentialValue z).property
  28       (entireExponentialValue w).property (f.eval w).property
  29       (entireExponentialValue_congr z w h) (equiv_symm (f.agrees w)))
  30 
  31 def realEval (f : Implementation) (x : RealInput) : RealInput :=
  32   ⟨(f.eval (realAxis x)).val.realPart,realPart_valid (f.eval (realAxis x)).property⟩
  33 
  34 theorem realEval_agrees (f : Implementation) (x : RealInput) :
  35     (f.realEval x).val.Equiv (exp x).val := realPart_equiv (f.agrees (realAxis x))
  36 end Implementation
  37 
  38 def powerSeries : Implementation :=
  39   ⟨entireExponentialValue,fun z => equiv_refl _ (entireExponentialValue z).property⟩
  40 def compoundInterest : Implementation := ⟨compoundValue,compoundValue_equiv_powerSeries⟩
  41 def linearODE : Implementation := ⟨MatrixExponential.scalarExponential,scalarExponential_agreement⟩

  43 /-- The bounded-input Taylor construction is covered by the same agreement
  44 network, without narrowing its inputs to rational constants. -/
  45 theorem representedTaylor_agrees {C : Rat} (A : ComplexExponentialLift.BoundedInput C) :
  46     (ComplexExponentialLift.BoundedInput.exponential A).Equiv
  47       (powerSeries.eval ⟨A.raw,A.valid⟩).val := equiv_symm (entireExponential_legacy_lift A)

  49 /-- Logarithm computations register one proved agreement, rather than a
  50 pairwise comparison with every existing algorithm. -/
  51 structure LogImplementation where
  52   eval : PositiveInput → RealInput
  53   agrees : ∀ x, (eval x).val.Equiv (log x).val
  54 
  55 namespace LogImplementation
  56 theorem equivalent (f g : LogImplementation) (x : PositiveInput) :
  57     (f.eval x).val.Equiv (g.eval x).val :=
  58   RealRaw.equiv_trans (f.eval x).property (log x).property (g.eval x).property
  59     (f.agrees x) (RealRaw.equiv_symm (g.agrees x))
  60 
  61 theorem congr (f : LogImplementation) (x y : PositiveInput)
  62     (h : x.val.val.Equiv y.val.val) : (f.eval x).val.Equiv (f.eval y).val :=
  63   RealRaw.equiv_trans (f.eval x).property (log x).property (f.eval y).property
  64     (f.agrees x) (RealRaw.equiv_trans (log x).property (log y).property (f.eval y).property
  65       (log_congr x y h) (RealRaw.equiv_symm (f.agrees y)))
  66 
  67 theorem exp_eval (f : LogImplementation) (x : PositiveInput) :
  68     (exp (f.eval x)).val.Equiv x.val.val :=
  69   RealRaw.equiv_trans (exp (f.eval x)).property (exp (log x)).property x.val.property
  70     (exp_congr _ _ (f.agrees x)) (exp_log x)
  71 end LogImplementation
  72 
  73 def logarithmContinuation : LogImplementation :=
  74   ⟨log,fun x => RealRaw.equiv_refl _ (log x).property⟩
  75 def reciprocalIntegral : LogImplementation := ⟨integralLog,integralLog_equiv_log⟩
  76 
  77 /-- An independently justified inversion computation joins the same real
  78 exponential API by proving its integral inverse law. -/
  79 theorem real_computation_agrees_of_integral_inverse (f : RealInput → PositiveInput)
  80     (hinverse : ∀ x, (integralLog (f x)).val.Equiv x.val) (x : RealInput) :
  81     (f x).val.val.Equiv (exp x).val := exp_unique_of_integralLog x (f x) (hinverse x)
  82 end ComputableAnalysis.ExponentialComputations
