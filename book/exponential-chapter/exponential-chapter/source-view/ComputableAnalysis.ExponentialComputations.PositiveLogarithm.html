   5 /-! A global real logarithm on positive represented inputs. The complex
   6 continuation algorithm constructs a logarithm; its real coordinate is
   7 independent of the route because exponential fibers have equal real parts.
   8 This module proves the inverse laws. Integral identification is separate. -/
   9 namespace ComputableAnalysis.ExponentialComputations

  12 private def scalarClass (z : Scalar) := ComplexRawQuotient.ofRaw z.val z.property

  14 abbrev PositiveInput := {x : RealInput // x.val.Pos}
  15 def complexLog (x : PositiveInput) : Scalar :=
  16   NonzeroLogarithmConstruction.value (realAxis x.val) (positive_real_nonzero x.val x.property)
  17 def log (x : PositiveInput) : RealInput :=
  18   ⟨(complexLog x).val.realPart,realPart_valid (complexLog x).property⟩
  19 
  20 theorem complexLog_exponential (x : PositiveInput) :
  21     (entireExponentialValue (complexLog x)).val.Equiv (realAxis x.val).val :=
  22   equiv_trans (entireExponentialValue (complexLog x)).property
  23     (MatrixExponential.scalarExponential (complexLog x)).property (realAxis x.val).property
  24     (equiv_symm (scalarExponential_agreement _))
  25     (NonzeroLogarithmConstruction.exponential _ _)
  26 
  27 theorem log_congr (x y : PositiveInput) (h : x.val.val.Equiv y.val.val) :
  28     (log x).val.Equiv (log y).val := by
  29   apply entireExponential_fiber_realPart (complexLog x) (complexLog y)
  30   exact equiv_trans (entireExponentialValue (complexLog x)).property (realAxis x.val).property
  31     (entireExponentialValue (complexLog y)).property (complexLog_exponential x)
  32     (equiv_trans (realAxis x.val).property (realAxis y.val).property
  33       (entireExponentialValue (complexLog y)).property (real_embedding_congr x.val.property y.val.property h)
  34       (equiv_symm (complexLog_exponential y)))

  36 /-- One inverse law on every valid represented real input. -/
  37 theorem log_exp (x : RealInput) :
  38     (log ⟨exp x,exp_positive x⟩).val.Equiv x.val := by
  39   apply entireExponential_fiber_realPart (complexLog ⟨exp x,exp_positive x⟩) (realAxis x)
  40   exact equiv_trans (entireExponentialValue _).property (realAxis (exp x)).property
  41     (entireExponentialValue (realAxis x)).property
  42     (complexLog_exponential _) (equiv_symm (exp_real_embedding x))

  44 theorem realAxis_conjugate (x : RealInput) : (conj (realAxis x).val).Equiv (realAxis x).val := by
  45   intro n
  46   apply (compareAt_overlap_iff _ _ n n).mpr
  47   have ho := RealRaw.interval_order_of_valid x.val x.property n
  48   simp only [realAxis,ofRealRaw,conj,QBox.conj,QComplex.conj]
  49   constructor <;> constructor <;> simp only [Rat.neg_zero] <;> first | exact ho | exact Rat.le_refl

  51 private theorem positive_sum (x y : RealInput) (hx : x.val.Pos) (hy : y.val.Pos) :
  52     (RealRaw.add x.val y.val).Pos := by
  53   obtain ⟨i,hi⟩ := hx
  54   obtain ⟨j,hj⟩ := hy
  55   let k := max i j
  56   have hxi := (x.property.2.1 i k (Nat.le_max_left _ _)).1
  57   have hyj := (y.property.2.1 j k (Nat.le_max_right _ _)).1
  58   refine ⟨k,?_⟩
  59   change 0<(x.val.compute k).lo+(y.val.compute k).lo
  60   change 0<(x.val.compute i).lo at hi
  61   change 0<(y.val.compute j).lo at hj
  62   grind only
  63 
  64 /-- The second inverse law retains only the genuine positivity domain. -/
  65 theorem exp_log (x : PositiveInput) : (exp (log x)).val.Equiv x.val.val := by
  66   let l := complexLog x
  67   let a := exp (log x)
  68   let A := realAxis a
  69   let X := realAxis x.val
  70   have hprod := mul_equiv (entireExponentialValue l).property X.property
  71     (conj_valid _ (entireExponentialValue l).property) (conj_valid _ X.property)
  72     (complexLog_exponential x) (conj_equiv (complexLog_exponential x))
  73   have hxx := mul_equiv X.property X.property (conj_valid _ X.property) X.property
  74     (equiv_refl _ X.property) (realAxis_conjugate x.val)
  75   have hmag := entireExponential_squared_magnitude l
  76   let r : Scalar := ⟨ofRealRaw (RealRaw.scaleRat 2 l.val.realPart),
  77     ofRealRaw_valid _ (RealRaw.scaleRat_valid (realPart_valid l.property))⟩
  78   have hadd : (add (realAxis (log x)).val (realAxis (log x)).val).Equiv r.val := by
  79     intro n
  80     apply (compareAt_overlap_iff _ _ n n).mpr
  81     have ho := RealRaw.interval_order_of_valid (log x).val (log x).property n
  82     change ((complexLog x).val.compute n).lo.re ≤ ((complexLog x).val.compute n).hi.re at ho
  83     simp only [r,l,realAxis,log,ofRealRaw,realPart,add,QBox.add,QComplex.add,RealRaw.scaleRat,RealRaw.scaleRatCompute,
  84       if_pos (show (0:Rat)≤2 by decide +kernel)]
  85     constructor <;> constructor <;> grind only
  86   have heA := exp_real_embedding (log x)
  87   have hAA := equiv_trans (mul_valid A.property A.property)
  88     (mul_valid (entireExponentialValue (realAxis (log x))).property (entireExponentialValue (realAxis (log x))).property)
  89     (entireExponentialValue r).property
  90     (mul_equiv A.property (entireExponentialValue (realAxis (log x))).property
  91       A.property (entireExponentialValue (realAxis (log x))).property (equiv_symm heA) (equiv_symm heA))
  92     (equiv_trans (mul_valid (entireExponentialValue (realAxis (log x))).property (entireExponentialValue (realAxis (log x))).property)
  93       (entireExponentialValue ⟨add (realAxis (log x)).val (realAxis (log x)).val,add_valid (realAxis (log x)).property (realAxis (log x)).property⟩).property
  94       (entireExponentialValue r).property (entireExponential_addition _ _)
  95       (entireExponentialValue_congr _ _ hadd))
  96   have heq := equiv_trans (mul_valid A.property A.property) (entireExponentialValue r).property
  97     (mul_valid X.property X.property) hAA
  98     (equiv_trans (entireExponentialValue r).property
  99       (mul_valid (entireExponentialValue l).property (conj_valid _ (entireExponentialValue l).property))
 100       (mul_valid X.property X.property) (equiv_symm hmag)
 101       (equiv_trans (mul_valid (entireExponentialValue l).property (conj_valid _ (entireExponentialValue l).property))
 102         (mul_valid X.property (conj_valid _ X.property)) (mul_valid X.property X.property) hprod hxx))
 103   let S : Scalar := scalarSum A X
 104   have hS : NonzeroBoxSearch.Nonzero S := by
 105     have hp := positive_sum a x.val (exp_positive (log x)) x.property
 106     have hs : S.val.Equiv (realAxis ⟨RealRaw.add a.val x.val.val,RealRaw.add_valid a.property x.val.property⟩).val := by
 107       intro n
 108       apply (compareAt_overlap_iff _ _ n n).mpr
 109       have ho := RealRaw.interval_order_of_valid (RealRaw.add a.val x.val.val)
 110         (RealRaw.add_valid a.property x.val.property) n
 111       simp only [S,scalarSum,A,X,realAxis,ofRealRaw,add,QBox.add,QComplex.add,RealRaw.add,RealRaw.addCompute,Rat.zero_add]
 112       constructor <;> constructor <;> first | exact ho | exact Rat.le_refl
 113     intro hz
 114     exact positive_real_nonzero _ hp (equiv_trans
 115       (realAxis ⟨RealRaw.add a.val x.val.val,RealRaw.add_valid a.property x.val.property⟩).property
 116       S.property (ofQComplex_valid _)
 117       (equiv_symm hs) hz)
 118   let I := RepresentedReciprocal.inverse S hS
 119   have hi := ComplexRawQuotient.ofRaw_eq_ofRaw
 120     (hleft := mul_valid S.property I.property) (hright := ofQComplex_valid _)
 121     (RepresentedReciprocal.mul_inverse S hS)
 122   have hh := ComplexRawQuotient.ofRaw_eq_ofRaw
 123     (hleft := mul_valid A.property A.property) (hright := mul_valid X.property X.property) heq
 124   have hc : A.val.Equiv X.val := by
 125     apply ComplexRawQuotient.equiv_of_ofRaw_eq (hleft := A.property) (hright := X.property)
 126     let AA := scalarClass A
 127     let XX := scalarClass X
 128     let II := scalarClass I
 129     change (AA+XX)*II=1 at hi
 130     change AA*AA=XX*XX at hh
 131     change AA=XX
 132     grind only
 133   exact realPart_equiv hc

 135 /-- A constructed local analytic extension of the real logarithm. -/
 136 def logChart (x : PositiveInput) : DomainFunctions.Map :=
 137   NonzeroLogarithmConstruction.localFunction (realAxis x.val) (positive_real_nonzero x.val x.property)
 138 def logChartHolomorphic (x : PositiveInput) : DomainFunctions.Holomorphic (logChart x) :=
 139   NonzeroLogarithmConstruction.localHolomorphic _ _

 141 theorem logChart_real_agreement (x y : PositiveInput)
 142     (hy : (logChart x).domain (realAxis y.val)) :
 143     ((logChart x).eval (realAxis y.val) hy).val.realPart.Equiv (log y).val := by
 144   apply entireExponential_fiber_realPart _ (complexLog y)
 145   have he := NonzeroLogarithmConstruction.local_exponential
 146     (realAxis x.val) (positive_real_nonzero x.val x.property) (realAxis y.val) hy
 147   exact equiv_trans (entireExponentialValue _).property
 148     (MatrixExponential.scalarExponential _).property (entireExponentialValue (complexLog y)).property
 149     (equiv_symm (scalarExponential_agreement _))
 150     (equiv_trans (MatrixExponential.scalarExponential _).property (realAxis y.val).property
 151       (entireExponentialValue (complexLog y)).property he (equiv_symm (complexLog_exponential y)))

 153 /-- The actual derivative of the locally agreeing logarithm is the reciprocal,

 161 /-- Any real computation whose exponential is the target agrees with log. -/
 162 theorem log_unique (x : PositiveInput) (a : RealInput)
 163     (h : (exp a).val.Equiv x.val.val) : a.val.Equiv (log x).val :=
 164   exp_injective a (log x) (RealRaw.equiv_trans (exp a).property x.val.property
 165     (exp (log x)).property h (RealRaw.equiv_symm (exp_log x)))

 167 /-- Any supplied computation inverted by log must be the exponential.
 168 No conclusion about an unproved integral interpretation is inferred here. -/
 169 theorem exp_unique_of_log (x : RealInput) (y : PositiveInput)
 170     (h : (log y).val.Equiv x.val) : y.val.val.Equiv (exp x).val :=
 171   RealRaw.equiv_trans y.val.property (exp (log y)).property (exp x).property
 172     (RealRaw.equiv_symm (exp_log y)) (exp_congr (log y) x h)
