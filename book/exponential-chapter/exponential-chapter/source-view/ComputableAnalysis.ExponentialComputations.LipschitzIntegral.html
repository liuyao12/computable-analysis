  11 private def pointRaw (f : FunctionOnInterval) (x : Rat)
  12     (hx : inDomainInterval f.lower f.upper x) : RealRaw := {compute := f.compute x hx}
  13 private theorem point_valid (f : FunctionOnInterval) (x : Rat)
  14     (hx : inDomainInterval f.lower f.upper x) : (pointRaw f x hx).Valid :=
  15   f.valid_on x (f.defined_on x hx)

  17 private theorem gridEventually (L : Rat) (eps : QPos) : ∃ N : Nat, ∀ n : Nat, N≤n →
  18     decide (4*qabs L*(1/2:Rat)^n≤eps.val)=true := by
  19   refine ⟨RationalMajorant.halfDecayShift (4*qabs L) eps,fun n hn => ?_⟩
  20   simp only [decide_eq_true_eq]
  21   exact RationalMajorant.halfDecayShift_spec_of_le
  22     (Rat.mul_nonneg (by decide +kernel) (qabs_nonneg L)) eps hn

  24 /-- Find the first adequate binary mesh, rather than exponentiating a loose
  25 existence bound. The latter is used only in the termination proof. -/
  26 private def gridShift (L : Rat) (eps : QPos) : Nat := PrecisionSearch.firstFrom
  27   (fun n => decide (4*qabs L*(1/2:Rat)^n≤eps.val)) (gridEventually L eps) 0
  28 
  29 private theorem gridShift_spec (L : Rat) (eps : QPos) :
  30     4*qabs L*(1/2:Rat)^gridShift L eps≤eps.val := by
  31   have h := (PrecisionSearch.firstFrom_spec
  32     (fun n => decide (4*qabs L*(1/2:Rat)^n≤eps.val)) (gridEventually L eps) 0).2
  33   simpa only [gridShift,decide_eq_true_eq] using h

  35 private def grid (f : FunctionOnInterval) (h0 : f.lower=0) (h1 : f.upper=1)
  36     (L : Rat) (eps : QPos) : RationalPartition f.lower f.upper :=
  37   RationalPartition.uniform f.lower f.upper (2^gridShift L eps)
  38     (Nat.pow_pos (by decide)) (by rw [h0,h1]; decide +kernel)
  39 private def mesh (L : Rat) (eps : QPos) : Rat := (1/2:Rat)^gridShift L eps
  40 private theorem grid_width (f : FunctionOnInterval) (h0 : f.lower=0) (h1 : f.upper=1)
  41     (L : Rat) (eps : QPos) (k : Nat) :
  42     (grid f h0 h1 L eps).point (k+1)-(grid f h0 h1 L eps).point k = mesh L eps := by
  43   change leftPoint f.lower f.upper (2^gridShift L eps) (k+1)-
  44     leftPoint f.lower f.upper (2^gridShift L eps) k = _
  45   rw [leftPoint_step,h0,h1]
  46   simp only [ComputableAnalysis.mesh, if_neg (Nat.ne_of_gt (Nat.pow_pos (by decide : 0 < (2:Nat)) (n := gridShift L eps))),
  47     show (1:Rat)-0=1 by decide +kernel]
  48   exact (RationalMajorant.half_pow_eq_one_div_nat_two_pow _).symm
  49 private def tolerance (eps : QPos) : QPos := ⟨eps.val/2,by
  50   rw [Rat.div_def]; exact Rat.mul_pos eps.property (by decide +kernel)⟩
  51 
  52 private def sampleBox (f : FunctionOnInterval) (P : RationalPartition f.lower f.upper)
  53     (eps : QPos) (k : Nat) : QInterval :=
  54   if hk : k<P.pieces then
  55     let X := Real.ofRaw (pointRaw f (P.point k) (P.point_in_bounds (Nat.le_of_lt hk)))
  56       (point_valid f _ _)
  57     X.compute (ComputableCoefficient.widthStage X (tolerance eps))
  58   else {lo := 0,hi := 0}

  60 /-- Actual point boxes widened by a justified variation bound on the cell. -/
  61 def lipschitzBounds (f : FunctionOnInterval) (h0 : f.lower=0) (h1 : f.upper=1)
  62     (L : Rat) (hL : 0 ≤ L)
  63     (hlip : ∀ x y (hx : inDomainInterval f.lower f.upper x)
  64       (hy : inDomainInterval f.lower f.upper y),
  65       Small (ofRealRaw (RealRaw.sub (pointRaw f x hx) (pointRaw f y hy))) (L*qabs (x-y)))
  66     (eps : QPos) : Integral.Bounds f := by
  67   let P := grid f h0 h1 L eps
  68   refine { partition := P
  69            lower := fun k => (sampleBox f P eps k).lo - L * mesh L eps
  70            upper := fun k => (sampleBox f P eps k).hi + L * mesh L eps
  71            lower_le := ?_
  72            upper_ge := ?_ }
  73   all_goals
  74     intro k hk x hx n
  75     let c := P.point k
  76     have hc : inDomainInterval f.lower f.upper c := P.point_in_bounds (Nat.le_of_lt hk)
  77     have hxm := (P.cell k hk).contains_inDomain hx
  78     let X := Real.ofRaw (pointRaw f c hc) (point_valid f c hc)
  79     let s := ComputableCoefficient.widthStage X (tolerance eps)
  80     have hd : qabs (x-c) ≤ mesh L eps := by
  81       have hg := grid_width f h0 h1 L eps k
  82       have hxlo := hx.1
  83       have hxhi := hx.2
  84       change P.point k≤x at hxlo
  85       change x≤P.point (k+1) at hxhi
  86       rw [qabs_eq_self_of_nonneg (show 0 ≤ x-c by dsimp [c]; grind only)]
  87       change P.point (k+1)-P.point k=mesh L eps at hg
  88       dsimp [c]; grind only
  89     have hm : 0≤mesh L eps := Rat.pow_nonneg (by decide +kernel)
  90     have hE : 0 ≤ L*mesh L eps := Rat.mul_nonneg hL hm
  91     have hh := (hlip x c hxm hc).mono (Rat.mul_le_mul_of_nonneg_left hd hL)
  92     have hh' : Small (ofRealRaw (RealRaw.sub (pointRaw f x hxm)
  93         (RealRaw.scaleRat 1 (pointRaw f c hc)))) (L*mesh L eps) := by
  94       simpa only [Small,RealRaw.Le,ofRealRaw,realPart,imagPart,RealRaw.sub,RealRaw.subCompute,
  95         RealRaw.scaleRat,RealRaw.scaleRatCompute,if_pos (show (0:Rat)≤1 by decide +kernel),Rat.one_mul] using hh
  96     have hlo : (RealRaw.ofRat (X.compute s).lo).Le (pointRaw f c hc) :=
  97       fun _ m => RealRaw.le_refl _ (point_valid f c hc) s m
  98     have hhi : (pointRaw f c hc).Le (RealRaw.ofRat (X.compute s).hi) :=
  99       fun m _ => RealRaw.le_refl _ (point_valid f c hc) m s
 100     have h := remainder_order (pointRaw f x hxm) (pointRaw f c hc)
 101       (point_valid f x hxm) (point_valid f c hc) 1 (L*mesh L eps)
 102       (X.compute s).lo (X.compute s).hi (by decide +kernel) hh' hlo hhi
 103     simp only [Rat.one_mul] at h
 104     first
 105     | simpa only [sampleBox,dif_pos hk,X,s,c,RealRaw.ofRat,pointRaw] using h.1 0 n
 106     | simpa only [sampleBox,dif_pos hk,X,s,c,RealRaw.ofRat,pointRaw] using h.2 n 0

 108 private theorem sum_sub (f g : Nat → Rat) (n : Nat) :
 109     Integral.rectangleSum f n-Integral.rectangleSum g n = Integral.rectangleSum (fun k => f k-g k) n := by
 110   induction n with
 111   | zero => simp only [Integral.rectangleSum]; grind only
 112   | succ n ih => simp only [Integral.rectangleSum]; grind only

 114 /-- The computed finite bounds become arbitrarily tight. -/
 115 theorem lipschitzBounds_gap (f : FunctionOnInterval) (h0 : f.lower=0) (h1 : f.upper=1)
 116     (L : Rat) (hL : 0≤L) (hlip) (eps : QPos) :
 117     (lipschitzBounds f h0 h1 L hL hlip eps).upperSum-
 118       (lipschitzBounds f h0 h1 L hL hlip eps).lowerSum ≤ eps.val := by
 119   let B := lipschitzBounds f h0 h1 L hL hlip eps
 120   let P := B.partition
 121   have hg : ∀ k, k<P.pieces → B.upper k-B.lower k ≤ eps.val/2+2*L*mesh L eps := by
 122     intro k hk
 123     have hs := ComputableCoefficient.widthStage_spec
 124       (Real.ofRaw (pointRaw f (P.point k) (P.point_in_bounds (Nat.le_of_lt hk))) (point_valid f _ _))
 125       (tolerance eps)
 126     change (sampleBox f P eps k).hi+L*mesh L eps-
 127       ((sampleBox f P eps k).lo-L*mesh L eps) ≤ _
 128     have hbox : sampleBox f P eps k =
 129         (Real.ofRaw (pointRaw f (P.point k) (P.point_in_bounds (Nat.le_of_lt hk))) (point_valid f _ _)).compute
 130           (ComputableCoefficient.widthStage
 131             (Real.ofRaw (pointRaw f (P.point k) (P.point_in_bounds (Nat.le_of_lt hk))) (point_valid f _ _)) (tolerance eps)) := by
 132       simp only [sampleBox,dif_pos hk]
 133     rw [hbox]
 134     change _- _ ≤ eps.val/2 at hs
 135     grind only
 136   have hm := gridShift_spec L eps
 137   rw [qabs_eq_self_of_nonneg hL] at hm
 138   change 4*L*mesh L eps ≤ eps.val at hm
 139   have hh : eps.val/2+2*L*mesh L eps ≤ eps.val := by grind only
 140   change Integral.rectangleSum (fun k => (P.point (k+1)-P.point k)*B.upper k) P.pieces-
 141     Integral.rectangleSum (fun k => (P.point (k+1)-P.point k)*B.lower k) P.pieces≤_
 142   rw [sum_sub]
 143   calc
 144     _ ≤ Integral.rectangleSum (fun k => (P.point (k+1)-P.point k)*eps.val) P.pieces := by
 145       apply Integral.rectangleSum_mono
 146       intro k hk
 147       have hc := P.monotone k (k+1) (by omega) (by omega)
 148       have hs := Rat.mul_le_mul_of_nonneg_left (Rat.le_trans (hg k hk) hh)
 149         (show 0≤P.point (k+1)-P.point k by grind only)
 150       grind only
 151     _ = eps.val := by
 152       rw [Integral.rectangleSum_scale,Integral.rectangleSum_telescope,P.right_endpoint,P.left_endpoint,h0,h1]
 153       grind only
