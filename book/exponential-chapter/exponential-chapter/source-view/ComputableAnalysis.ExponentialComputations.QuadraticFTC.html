  10 theorem remainder_order (p d : RealRaw) (hp : p.Valid) (hd : d.Valid)
  11     (h C l r : Rat) (hh : 0 ≤ h)
  12     (he : Small (ofRealRaw (RealRaw.sub p (RealRaw.scaleRat h d))) C)
  13     (hl : (RealRaw.ofRat l).Le d) (hr : d.Le (RealRaw.ofRat r)) :
  14     (RealRaw.ofRat (h*l-C)).Le p ∧ p.Le (RealRaw.ofRat (h*r+C)) := by
  15   have hsd := RealRaw.scaleRat_valid hd (r := h)
  16   have hlo := RealRaw.le_add_le_add he.1 (RealRaw.le_scaleRat_le_scaleRat hh hl)
  17   have hhi := RealRaw.le_add_le_add he.2.1 (RealRaw.le_scaleRat_le_scaleRat hh hr)
  18   have hv := RealRaw.add_valid (RealRaw.sub_valid hp hsd) hsd
  19   have hc := RealRaw.sub_add_cancel_equiv hp hsd
  20   have htlo := RealRaw.le_trans hv hlo (RealRaw.le_of_equiv hv hp hc)
  21   have hthi := RealRaw.le_trans hv (RealRaw.le_of_equiv hp hv (RealRaw.equiv_symm hc)) hhi
  22   constructor
  23   · intro n m
  24     have h := htlo n m
  25     simpa only [RealRaw.add,RealRaw.addCompute,RealRaw.ofRat,RealRaw.scaleRat,
  26       RealRaw.scaleRatCompute,if_pos hh,realPart,ofRealRaw,Rat.sub_eq_add_neg,Rat.add_comm] using h
  27   · intro n m
  28     have h := hthi n m
  29     simpa only [RealRaw.add,RealRaw.addCompute,RealRaw.ofRat,RealRaw.scaleRat,
  30       RealRaw.scaleRatCompute,if_pos hh,realPart,ofRealRaw,Rat.add_comm] using h

  32 /-- Local quadratic estimates imply the cell-order laws used by the existing
  33 FTC. The mesh restriction is eliminated by finite dyadic subdivision. -/
  34 theorem cellBounds_of_quadratic (f : FunctionOnInterval) (P : Rat → RealRaw)
  35     (h0 : f.lower = 0) (h1 : f.upper = 1)
  36     (hv : ∀ x, 0 ≤ x → x ≤ 1 → (P x).Valid)
  37     (C : Rat) (hC : 0 ≤ C) (delta : QPos) (B : Rat)
  38     (hB : ∀ x (hx : inDomainInterval f.lower f.upper x) n,
  39       -B ≤ (f.compute x hx n).hi ∧ (f.compute x hx n).lo ≤ B)
  40     (he : ∀ u v (hu : inDomainInterval f.lower f.upper u),
  41       0 ≤ u → u ≤ v → v ≤ 1 → v-u ≤ delta.val →
  42       Small (ofRealRaw (RealRaw.sub (RealRaw.sub (P v) (P u))
  43         (RealRaw.scaleRat (v-u) {compute := f.compute u hu}))) (C*(v-u)^2)) :
  44     Integral.CellDerivativeBounds f P := by
  45   have localBounds (u v l r : Rat) (hu : 0 ≤ u) (huv : u ≤ v) (hv1 : v ≤ 1)
  46       (hs : v-u ≤ delta.val)
  47       (hl : ∀ x (hx : inDomainInterval f.lower f.upper x), u ≤ x → x ≤ v →
  48         ∀ n, l ≤ (f.compute x hx n).hi)
  49       (hr : ∀ x (hx : inDomainInterval f.lower f.upper x), u ≤ x → x ≤ v →
  50         ∀ n, (f.compute x hx n).lo ≤ r) :
  51       (RealRaw.ofRat ((v-u)*l-C*(v-u)^2)).Le (P v-P u) ∧
  52         (P v-P u).Le (RealRaw.ofRat ((v-u)*r+C*(v-u)^2)) := by
  53     have hum : inDomainInterval f.lower f.upper u := by rw [h0,h1]; exact ⟨hu,by grind only⟩
  54     exact remainder_order _ _ (RealRaw.sub_valid (hv v (by grind only) hv1) (hv u hu (by grind only)))
  55       (show ({compute := f.compute u hum} : RealRaw).Valid from f.valid_on u (f.defined_on u hum)) (v-u) (C*(v-u)^2) l r (by grind only)
  56       (he u v hum hu huv hv1 hs) (fun _ n => hl u hum (Rat.le_refl) huv n)
  57       (fun n _ => hr u hum (Rat.le_refl) huv n)
  58   have dyadic : ∀ (n : Nat) (u v l r : Rat), 0 ≤ u → u ≤ v → v ≤ 1 →
  59       (v-u)*((1:Rat)/2)^n ≤ delta.val →
  60       (∀ x (hx : inDomainInterval f.lower f.upper x), u ≤ x → x ≤ v →
  61         ∀ k, l ≤ (f.compute x hx k).hi) →
  62       (∀ x (hx : inDomainInterval f.lower f.upper x), u ≤ x → x ≤ v →
  63         ∀ k, (f.compute x hx k).lo ≤ r) →
  64       (RealRaw.ofRat ((v-u)*l-C*(v-u)^2*((1:Rat)/2)^n)).Le (P v-P u) ∧
  65         (P v-P u).Le (RealRaw.ofRat ((v-u)*r+C*(v-u)^2*((1:Rat)/2)^n)) := by
  66     intro n
  67     induction n with
  68     | zero =>
  69       intro u v l r hu huv hv1 hs hl hr
  70       simpa only [Rat.pow_zero,Rat.mul_one] using localBounds u v l r hu huv hv1
  71         (by simpa only [Rat.pow_zero,Rat.mul_one] using hs) hl hr
  72     | succ n ih =>
  73       intro u v l r hu huv hv1 hs hl hr
  74       have hum : u ≤ (u+v)/2 := by grind only
  75       have hmv : (u+v)/2 ≤ v := by grind only
  76       have hm0 : 0 ≤ (u+v)/2 := by grind only
  77       have hm1 : (u+v)/2 ≤ 1 := by grind only
  78       have hleft := ih u ((u+v)/2) l r hu hum hm1
  79         (by rw [Rat.pow_succ] at hs; change _ ≤ delta.val; grind only)
  80         (fun x hx hxu hxm k => hl x hx hxu (Rat.le_trans hxm hmv) k)
  81         (fun x hx hxu hxm k => hr x hx hxu (Rat.le_trans hxm hmv) k)
  82       have hright := ih ((u+v)/2) v l r hm0 hmv hv1
  83         (by rw [Rat.pow_succ] at hs; change _ ≤ delta.val; grind only)
  84         (fun x hx hxm hxv k => hl x hx (Rat.le_trans hum hxm) hxv k)
  85         (fun x hx hxm hxv k => hr x hx (Rat.le_trans hum hxm) hxv k)
  86       have hsum := RealRaw.add_valid
  87         (RealRaw.sub_valid (hv _ hm0 hm1) (hv u hu (by grind only)))
  88         (RealRaw.sub_valid (hv v (by grind only) hv1) (hv _ hm0 hm1))
  89       have hend := RealRaw.sub_valid (hv v (by grind only) hv1) (hv u hu (by grind only))
  90       have ht := RealRaw.sub_add_sub_cancel_middle_equiv
  91         (hv u hu (by grind only)) (hv _ hm0 hm1) (hv v (by grind only) hv1)
  92       have hlo := RealRaw.le_trans hsum (RealRaw.le_add_le_add hleft.1 hright.1)
  93         (RealRaw.le_of_equiv hsum hend ht)
  94       have hhi := RealRaw.le_trans hsum (RealRaw.le_of_equiv hend hsum (RealRaw.equiv_symm ht))
  95         (RealRaw.le_add_le_add hleft.2 hright.2)
  96       constructor
  97       · intro i j
  98         have h := hlo i j
  99         change _ ≤ ((P v-P u).compute j).hi at h ⊢
 100         simp only [RealRaw.add,RealRaw.addCompute,RealRaw.ofRat] at h
 101         rw [Rat.pow_succ]
 102         simp only [Rat.pow_succ,Rat.pow_zero,Rat.mul_one] at h ⊢
 103         simp only [RealRaw.ofRat]
 104         grind only
 105       · intro i j
 106         have h := hhi i j
 107         change ((P v-P u).compute i).lo ≤ _ at h ⊢
 108         simp only [RealRaw.add,RealRaw.addCompute,RealRaw.ofRat] at h
 109         rw [Rat.pow_succ]
 110         simp only [Rat.pow_succ,Rat.pow_zero,Rat.mul_one] at h ⊢
 111         simp only [RealRaw.ofRat]
 112         grind only
 113   refine ⟨?_,?_,?_⟩
 114   · intro x hx; rw [h0,h1] at hx; exact hv x hx.1 hx.2
 115   · intro u v l hu huv hv1 hl i j
 116     rw [h0] at hu; rw [h1] at hv1
 117     apply Classical.byContradiction
 118     intro hnot
 119     have hg : 0 < (((v-u)*l)-((P v-P u).compute j).hi)/2 := by
 120       change ¬(v-u)*l ≤ ((P v-P u).compute j).hi at hnot
 121       grind only
 122     let eps : QPos := ⟨_,hg⟩
 123     let n := max (RationalMajorant.halfDecayShift (v-u) delta)
 124       (RationalMajorant.halfDecayShift (C*(v-u)^2) eps)
 125     have hs := RationalMajorant.halfDecayShift_spec_of_le (show 0 ≤ v-u by grind only) delta
 126       (show RationalMajorant.halfDecayShift (v-u) delta ≤ n from Nat.le_max_left _ _)
 127     have herr := RationalMajorant.halfDecayShift_spec_of_le
 128       (Rat.mul_nonneg hC (Rat.pow_nonneg (show 0 ≤ v-u by grind only))) eps
 129       (show RationalMajorant.halfDecayShift (C*(v-u)^2) eps ≤ n from Nat.le_max_right _ _)
 130     have hdlo := (dyadic n u v l B hu huv hv1 hs hl
 131       (fun x hx _ _ k => (hB x hx k).2)).1
 132     have hb := hdlo 0 j
 133     change (v-u)*l-C*(v-u)^2*((1:Rat)/2)^n ≤ ((P v-P u).compute j).hi at hb
 134     change C*(v-u)^2*((1:Rat)/2)^n ≤ (((v-u)*l)-((P v-P u).compute j).hi)/2 at herr
 135     grind only
 136   · intro u v r hu huv hv1 hr i j
 137     rw [h0] at hu; rw [h1] at hv1
 138     apply Classical.byContradiction
 139     intro hnot
 140     have hg : 0 < (((P v-P u).compute i).lo-(v-u)*r)/2 := by
 141       change ¬((P v-P u).compute i).lo ≤ (v-u)*r at hnot
 142       grind only
 143     let eps : QPos := ⟨_,hg⟩
 144     let n := max (RationalMajorant.halfDecayShift (v-u) delta)
 145       (RationalMajorant.halfDecayShift (C*(v-u)^2) eps)
 146     have hs := RationalMajorant.halfDecayShift_spec_of_le (show 0 ≤ v-u by grind only) delta
 147       (show RationalMajorant.halfDecayShift (v-u) delta ≤ n from Nat.le_max_left _ _)
 148     have herr := RationalMajorant.halfDecayShift_spec_of_le
 149       (Rat.mul_nonneg hC (Rat.pow_nonneg (show 0 ≤ v-u by grind only))) eps
 150       (show RationalMajorant.halfDecayShift (C*(v-u)^2) eps ≤ n from Nat.le_max_right _ _)
 151     have hdhi := (dyadic n u v (-B) r hu huv hv1 hs
 152       (fun x hx _ _ k => (hB x hx k).1) hr).2
 153     have hb := hdhi i 0
 154     change ((P v-P u).compute i).lo ≤ (v-u)*r+C*(v-u)^2*((1:Rat)/2)^n at hb
 155     change C*(v-u)^2*((1:Rat)/2)^n ≤ (((P v-P u).compute i).lo-(v-u)*r)/2 at herr
 156     grind only
 157 end ComputableAnalysis.ExponentialComputations
