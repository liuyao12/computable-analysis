  11 private theorem real_embedding_bound {z : ComplexRaw} {R : Rat}
  12     (hR : 0 ≤ R) (h : Small z R) : Small (ofRealRaw z.realPart) R := by
  13   refine ⟨h.1,h.2.1,?_,?_⟩
  14   · intro n m; change -R ≤ 0; grind only
  15   · intro n m; exact hR

  17 /-- The normalized coordinate is the reciprocal times the displacement. -/
  18 theorem relativePoint_displacement (c z : Scalar) (hc : NonzeroBoxSearch.Nonzero c) :
  19     (RelativeLogarithm.point c hc z).val.Equiv
  20       (mul (RepresentedReciprocal.inverse c hc).val (sub z.val c.val)) := by
  21   have hi := ComplexRawQuotient.ofRaw_eq_ofRaw
  22     (hleft := mul_valid (RepresentedReciprocal.inverse c hc).property c.property)
  23     (hright := ofQComplex_valid _) (RepresentedReciprocal.inverse_mul c hc)
  24   apply ComplexRawQuotient.equiv_of_ofRaw_eq
  25     (hleft := (RelativeLogarithm.point c hc z).property)
  26     (hright := mul_valid (RepresentedReciprocal.inverse c hc).property (sub_valid z.property c.property))
  27   let C := ComplexRawQuotient.ofRaw c.val c.property
  28   let Z := ComplexRawQuotient.ofRaw z.val z.property
  29   let I := ComplexRawQuotient.ofRaw (RepresentedReciprocal.inverse c hc).val
  30     (RepresentedReciprocal.inverse c hc).property
  31   change I*C=1 at hi
  32   change -1+I*Z=I*(Z-C)
  33   grind only

  35 /-- The global real logarithm increment agrees with its computed local chart. -/
  36 theorem log_increment (c z : PositiveInput)
  37     (hz : RelativeLogarithm.domain (realAxis c.val) (positive_real_nonzero c.val c.property) (realAxis z.val)) :
  38     (RealRaw.sub (log z).val (log c).val).Equiv
  39       (((RelativeLogarithm.function (realAxis c.val) (positive_real_nonzero c.val c.property)).eval
  40         (realAxis z.val) hz).val.realPart) := by
  41   let C := realAxis c.val
  42   let hc := positive_real_nonzero c.val c.property
  43   let L := (RelativeLogarithm.function C hc).eval (realAxis z.val) hz
  44   let A := MatrixExponential.propagatedBranch C hc (complexLog c) (realAxis z.val) hz
  45   have hexp := entireExponential_propagatedLogarithm C hc (complexLog c) (complexLog_exponential c)
  46     (realAxis z.val) hz
  47   have h := entireExponential_fiber_realPart A (complexLog z)
  48     (equiv_trans (entireExponentialValue A).property (realAxis z.val).property
  49       (entireExponentialValue (complexLog z)).property hexp (equiv_symm (complexLog_exponential z)))
  50   have hsum : (RealRaw.add (log c).val L.val.realPart).Equiv (log z).val := h
  51   exact RealRaw.equiv_trans
  52     (RealRaw.sub_valid (log z).property (log c).property)
  53     (RealRaw.sub_valid (RealRaw.add_valid (log c).property (realPart_valid L.property)) (log c).property)
  54     (realPart_valid L.property)
  55     (RealRaw.sub_equiv (log z).property (RealRaw.add_valid (log c).property (realPart_valid L.property))
  56       (log c).property (log c).property (RealRaw.equiv_symm hsum) (RealRaw.equiv_refl _ (log c).property))
  57     (RealRaw.add_sub_cancel_left_equiv (log c).property (realPart_valid L.property))

  59 /-- A quadratic logarithm remainder with any justified bound on the actual
  60 reciprocal. The conclusion is about the global logarithm on positive inputs. -/
  61 theorem log_increment_error (c z : PositiveInput) (B H : Rat) (hB : 0 ≤ B) (hH : 0 ≤ H)
  62     (hi : Small (RepresentedReciprocal.inverse (realAxis c.val)
  63       (positive_real_nonzero c.val c.property)).val B)
  64     (hstep : Small (sub (realAxis z.val).val (realAxis c.val).val) H)
  65     (hmesh : 2*B*H ≤ (1:Rat)/128) :
  66     Small (ofRealRaw (RealRaw.sub (RealRaw.sub (log z).val (log c).val)
  67       (mul (RepresentedReciprocal.inverse (realAxis c.val)
  68         (positive_real_nonzero c.val c.property)).val
  69         (sub (realAxis z.val).val (realAxis c.val).val)).realPart)) (16*(2*B*H)^2) := by
  70   let C := realAxis c.val
  71   let Z := realAxis z.val
  72   let hc := positive_real_nonzero c.val c.property
  73   let w := RelativeLogarithm.point C hc Z
  74   let d : Scalar := ⟨mul (RepresentedReciprocal.inverse C hc).val (sub Z.val C.val),
  75     mul_valid (RepresentedReciprocal.inverse C hc).property (sub_valid Z.property C.property)⟩
  76   have hwd := relativePoint_displacement C Z hc
  77   have hb := Small.mul (RepresentedReciprocal.inverse C hc).property (sub_valid Z.property C.property)
  78     hB hH hi hstep
  79   have hw : Small w.val (2*B*H) :=
  80     Small.congr d.property w.property (equiv_symm hwd) hb
  81   have hsize : 0 ≤ 2*B*H := Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hB) hH
  82   have hz : RelativeLogarithm.domain C hc Z :=
  83     ⟨2*B*H,hsize,by change 2*B*H<(1:Rat)/32; grind only,hw⟩
  84   let L := LocalLogarithm.function.eval w hz
  85   have he := localLogarithm_linear_error w hz (2*B*H) hsize hw
  86   have hinc := log_increment c z hz
  87   have hdiff := RealRaw.sub_equiv
  88     (RealRaw.sub_valid (log z).property (log c).property) (realPart_valid L.property)
  89     (realPart_valid d.property) (realPart_valid w.property) hinc (RealRaw.equiv_symm (realPart_equiv hwd))
  90   have hdiff' : (RealRaw.sub (RealRaw.sub (log z).val (log c).val) d.val.realPart).Equiv
  91       (sub L.val w.val).realPart := by
  92     intro n
  93     have hh := (RealRaw.compareAt_overlap_iff _ _ n n).mp (hdiff n)
  94     apply (RealRaw.compareAt_overlap_iff _ _ n n).mpr
  95     change (((log z).val.compute n).lo-((log c).val.compute n).hi)-(d.val.compute n).hi.re ≤
  96         (L.val.compute n).hi.re + -(w.val.compute n).lo.re ∧
  97       (L.val.compute n).lo.re + -(w.val.compute n).hi.re ≤
  98         (((log z).val.compute n).hi-((log c).val.compute n).lo)-(d.val.compute n).lo.re
  99     change (((log z).val.compute n).lo-((log c).val.compute n).hi)-(d.val.compute n).hi.re ≤
 100         (L.val.compute n).hi.re-(w.val.compute n).lo.re ∧
 101       (L.val.compute n).lo.re-(w.val.compute n).hi.re ≤
 102         (((log z).val.compute n).hi-((log c).val.compute n).lo)-(d.val.compute n).lo.re at hh
 103     simpa only [Rat.sub_eq_add_neg] using hh
 104   have hemb := ofRealRaw_equiv_of_equiv
 105     (RealRaw.sub_valid (RealRaw.sub_valid (log z).property (log c).property) (realPart_valid d.property))
 106     (realPart_valid (sub_valid L.property w.property)) hdiff'
 107   exact Small.congr (ofRealRaw_valid _ (realPart_valid (sub_valid L.property w.property)))
 108     (ofRealRaw_valid _ (RealRaw.sub_valid (RealRaw.sub_valid (log z).property (log c).property)
 109       (realPart_valid d.property))) (equiv_symm hemb)
 110     (real_embedding_bound (Rat.mul_nonneg (by decide +kernel) (Rat.pow_nonneg hsize)) he)
 111 end ComputableAnalysis.ExponentialComputations
