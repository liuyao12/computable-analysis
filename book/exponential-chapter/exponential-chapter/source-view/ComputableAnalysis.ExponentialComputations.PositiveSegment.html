   4 /-! A literal positive segment from one to an arbitrary positive represented
   5 real. A rational lower bound is found by a terminating box search. -/
   6 namespace ComputableAnalysis.ExponentialComputations
   7 open ComplexRaw FunctionTheory RiemannHilbert LocalODE DomainFunctions ModularForms
   8 set_option maxHeartbeats 1000000
   9 abbrev RationalUnit := {t : Rat // 0 ≤ t ∧ t ≤ 1}

  11 def eventually_positive (x : PositiveInput) : ∃ N, ∀ n, N≤n →
  12     decide (0<(x.val.val.compute n).lo)=true := by
  13   obtain ⟨N,hN⟩ := x.property
  14   refine ⟨N,fun n hn => ?_⟩
  15   have hh := (x.val.property.2.1 N n hn).1
  16   simp only [decide_eq_true_eq]
  17   change 0<(x.val.val.compute N).lo at hN
  18   grind only
  19 
  20 def positiveStage (x : PositiveInput) : Nat := PrecisionSearch.firstFrom
  21   (fun n => decide (0<(x.val.val.compute n).lo)) (eventually_positive x) 0
  22 
  23 theorem positiveStage_spec (x : PositiveInput) : 0<(x.val.val.compute (positiveStage x)).lo := by
  24   have h := (PrecisionSearch.firstFrom_spec
  25     (fun n => decide (0<(x.val.val.compute n).lo)) (eventually_positive x) 0).2
  26   simpa only [positiveStage,decide_eq_true_eq] using h
  27 
  28 def segmentLower (x : PositiveInput) : QPos :=
  29   ⟨minRat 1 (x.val.val.compute (positiveStage x)).lo,by
  30     have h := positiveStage_spec x
  31     unfold minRat; split <;> grind only⟩
  32 
  33 theorem segmentLower_le_one (x : PositiveInput) : (segmentLower x).val ≤ 1 := by
  34   dsimp [segmentLower]; unfold minRat; split <;> grind only
  35 
  36 theorem segmentLower_le_input (x : PositiveInput) : (RealRaw.ofRat (segmentLower x).val).Le x.val.val := by
  37   intro n m
  38   have hh := RealRaw.le_refl _ x.val.property (positiveStage x) m
  39   change (segmentLower x).val ≤ (x.val.val.compute m).hi
  40   dsimp [segmentLower]; unfold minRat; split <;> grind only
  41 
  42 theorem positive_of_lower (x : RealInput) (l : QPos) (hl : (RealRaw.ofRat l.val).Le x.val) : x.val.Pos := by
  43   let eps : QPos := ⟨l.val/2,by rw [Rat.div_def]; exact Rat.mul_pos l.property (by decide +kernel)⟩
  44   obtain ⟨N,hN⟩ := x.property.2.2 eps
  45   have hw := hN N (Nat.le_refl _)
  46   have hb := hl 0 N
  47   have hp := l.property
  48   refine ⟨N,?_⟩
  49   change l.val ≤ (x.val.compute N).hi at hb
  50   change (x.val.compute N).width ≤ l.val/2 at hw
  51   change 0 < (x.val.compute N).lo
  52   grind [QInterval.width]
  53 
  54 def positiveSegment (x : PositiveInput) (t : RationalUnit) : RealInput :=
  55   ⟨RealRaw.add (RealRaw.ofRat (1-t.val)) (RealRaw.scaleRat t.val x.val.val),
  56     RealRaw.add_valid (RealRaw.ofRat_valid _) (RealRaw.scaleRat_valid x.val.property)⟩
  57 
  58 theorem positiveSegment_lower (x : PositiveInput) (t : RationalUnit) :
  59     (RealRaw.ofRat (segmentLower x).val).Le (positiveSegment x t).val := by
  60   intro n m
  61   have hl := segmentLower_le_input x 0 m
  62   have h1 := segmentLower_le_one x
  63   have ht := t.property
  64   simp only [positiveSegment,RealRaw.add,RealRaw.addCompute,RealRaw.ofRat,RealRaw.scaleRat,
  65     RealRaw.scaleRatCompute,if_pos t.property.1]
  66   change (segmentLower x).val ≤ (1-t.val)+t.val*(x.val.val.compute m).hi
  67   change (segmentLower x).val ≤ (x.val.val.compute m).hi at hl
  68   have hprod := Rat.mul_le_mul_of_nonneg_left hl ht.1
  69   have hprod1 := Rat.mul_le_mul_of_nonneg_left h1 (show 0≤1-t.val by grind only)
  70   grind only
  71 
  72 def positivePoint (x : PositiveInput) (t : RationalUnit) : PositiveInput :=
  73   ⟨positiveSegment x t,positive_of_lower _ (segmentLower x) (positiveSegment_lower x t)⟩
  74 
  75 def segmentDisplacement (x : PositiveInput) : RealInput :=
  76   ⟨RealRaw.sub x.val.val (RealRaw.ofRat 1),RealRaw.sub_valid x.val.property (RealRaw.ofRat_valid _)⟩
  77 def segmentBound (x : PositiveInput) : Rat := scalarBound (realAxis (segmentDisplacement x))
  78 theorem segmentBound_pos (x : PositiveInput) : 0<segmentBound x := scalarBound_pos _
  79 
  80 /-- The actual segment difference agrees with its affine displacement. -/
  81 theorem positiveSegment_difference (x : PositiveInput) (u v : RationalUnit) :
  82     (sub (realAxis (positiveSegment x v)).val (realAxis (positiveSegment x u)).val).Equiv
  83       (scaleRat (v.val-u.val) (realAxis (segmentDisplacement x)).val) := by
  84   intro n
  85   apply (compareAt_overlap_iff _ _ n n).mpr
  86   have ho := RealRaw.interval_order_of_valid _ x.val.property n
  87   have hu := u.property
  88   have hv := v.property
  89   have hup := Rat.mul_le_mul_of_nonneg_left ho hu.1
  90   have hvp := Rat.mul_le_mul_of_nonneg_left ho hv.1
  91   simp only [realAxis,positiveSegment,segmentDisplacement,ofRealRaw,sub,add,neg,QBox.add,QBox.neg,
  92     QComplex.add,QComplex.neg,scaleRat,QBox.scaleRat,RealRaw.add,RealRaw.addCompute,RealRaw.ofRat,
  93     RealRaw.scaleRat,RealRaw.scaleRatCompute,if_pos hu.1,if_pos hv.1,RealRaw.sub,RealRaw.subCompute]
  94   split <;> constructor <;> constructor <;> grind only

  96 /-- Exact reciprocal comparison with the independently computed positive
  97 real interval reciprocal. -/
  98 theorem positive_inverse_agreement (c : PositiveInput) (l : QPos)
  99     (hl : (RealRaw.ofRat l.val).Le c.val.val) :
 100     (RepresentedReciprocal.inverse (realAxis c.val) (positive_real_nonzero c.val c.property)).val.Equiv
 101       (ofRealRaw (RealRaw.positiveReciprocal c.val.val l.val)) := by
 102   let r : Scalar := ⟨ofRealRaw (RealRaw.positiveReciprocal c.val.val l.val),
 103     ofRealRaw_valid _ (RealRaw.positiveReciprocal_valid _ _ c.val.property l.property (fun n => hl 0 n))⟩
 104   apply RepresentedReciprocal.inverse_unique (realAxis c.val) (positive_real_nonzero c.val c.property) r
 105   have hm := real_embedding_mul c.val.val (RealRaw.positiveReciprocal c.val.val l.val)
 106     c.val.property (RealRaw.positiveReciprocal_valid _ _ c.val.property l.property (fun n => hl 0 n))
 107   have hh := ofRealRaw_equiv_of_equiv (y := RealRaw.one)
 108     (RealRaw.mul_valid c.val.property (RealRaw.positiveReciprocal_valid _ _ c.val.property l.property (fun n => hl 0 n)))
 109     (RealRaw.ofRat_valid 1) (positiveReciprocal_mul_identity _ c.val.property l.val l.property hl)
 110   exact equiv_trans (mul_valid (realAxis c.val).property r.property)
 111     (ofRealRaw_valid _ (RealRaw.mul_valid c.val.property
 112       (RealRaw.positiveReciprocal_valid _ _ c.val.property l.property (fun n => hl 0 n))))
 113     (ofQComplex_valid _) hm hh

 115 def segmentReciprocal (x : PositiveInput) (t : RationalUnit) : RealInput :=
 116   ⟨RealRaw.positiveReciprocal (positiveSegment x t).val (segmentLower x).val,
 117     RealRaw.positiveReciprocal_valid _ _ (positiveSegment x t).property (segmentLower x).property
 118       (fun n => positiveSegment_lower x t 0 n)⟩
 119 def segmentIntegrandValue (x : PositiveInput) (t : RationalUnit) : RealInput :=
 120   ⟨RealRaw.mul (segmentDisplacement x).val (segmentReciprocal x t).val,
 121     RealRaw.mul_valid (segmentDisplacement x).property (segmentReciprocal x t).property⟩
