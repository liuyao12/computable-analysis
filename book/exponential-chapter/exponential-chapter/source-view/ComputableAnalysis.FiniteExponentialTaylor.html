   3 /-!
   4 # Finite exponential Taylor enclosures
   5 
   6 This module gives a potential-infinity remainder certificate for the rational
   7 exponential Taylor prefixes.  Every object below is a finite rational sum.
   8 The executable schedule chooses a first omitted degree; the main theorem says
   9 that every later finite prefix stays in one explicit rational interval around
  10 the scheduled prefix.
  11 
  12 There is no completed infinite sum, classical limit, completeness argument,
  13 or Mathlib dependency in this construction.
  14 -/
  15 
  16 namespace ComputableAnalysis

  20 /-- A finite block of exponential Taylor monomials, beginning at `start`.
  21 
  22 `remainderPartial x start terms` is the rational sum of exactly `terms`
  23 monomials, with degrees `start, ..., start + terms - 1`. -/
  24 def remainderPartial (x : Rat) (start : Nat) : Nat -> Rat
  25   | 0 => 0
  26   | terms + 1 =>
  27       remainderPartial x start terms +
  28         FormalPowerSeries.expCoeff (start + terms) * x ^ (start + terms)
  29 
  30 /-- Rewriting the integrated-prefix recurrence in the usual factorial
  31 monomial form.  This is an identity in rational arithmetic. -/
  32 theorem expCoeff_succ_monomial (x : Rat) (n : Nat) :
  33     FormalPowerSeries.expCoeff (n + 1) * x ^ (n + 1) =
  34       FormalPowerSeries.expCoeff n *
  35         (x ^ (n + 1) / ((n + 1 : Nat) : Rat)) := by
  36   unfold FormalPowerSeries.expCoeff
  37   rw [FormalPowerSeries.factorialRat_succ,
  38     Rat.div_def, Rat.div_def, Rat.div_def, Rat.inv_mul_rev]
  39   grind [Rat.mul_assoc, Rat.mul_comm]
  40 
  41 /-- The difference between two finite exponential Taylor prefixes is exactly
  42 the intervening finite block of factorial monomials. -/
  43 theorem remainderPartial_eq_expTaylorPrefix_sub
  44     (x : Rat) (n terms : Nat) :
  45     remainderPartial x (n + 1) terms =
  46       FinitePolynomial.expTaylorPrefix (n + terms) x -
  47         FinitePolynomial.expTaylorPrefix n x := by
  48   induction terms with
  49   | zero =>
  50       rw [Nat.add_zero, remainderPartial]
  51       grind [Rat.sub_eq_add_neg]
  52   | succ terms ih =>
  53       rw [remainderPartial, ih]
  54       rw [show n + (terms + 1) = (n + terms) + 1 by omega,
  55         FinitePolynomial.expTaylorPrefix_succ]
  56       have hindex : n + 1 + terms = n + terms + 1 := by omega
  57       rw [hindex]
  58       rw [expCoeff_succ_monomial]
  59       grind [Rat.sub_eq_add_neg, Rat.add_assoc, Rat.add_comm]
  60 
  61 /-- On the rational interval `|x| <= C`, a finite remainder block is bounded
  62 by the corresponding finite factorial majorant. -/
  63 theorem remainderPartial_abs_le_factorialTailPartial
  64     {C x : Rat} (hC : 0 <= C) (hx : qabs x <= C)
  65     (start terms : Nat) :
  66     qabs (remainderPartial x start terms) <=
  67       RationalMajorant.factorialTailPartial C start terms := by
  68   induction terms with
  69   | zero =>
  70       rw [remainderPartial, RationalMajorant.factorialTailPartial]
  71       exact Rat.le_refl
  72   | succ terms ih =>
  73       rw [remainderPartial, RationalMajorant.factorialTailPartial]
  74       have hterm :=
  75         FinitePolynomial.qabs_expCoeff_monomial_le_factorialTailTerm
  76           hC hx (start + terms)
  77       calc
  78         qabs
  79             (remainderPartial x start terms +
  80               FormalPowerSeries.expCoeff (start + terms) *
  81                 x ^ (start + terms)) <=
  82             qabs (remainderPartial x start terms) +
  83               qabs
  84                 (FormalPowerSeries.expCoeff (start + terms) *
  85                   x ^ (start + terms)) :=
  86           qabs_add_le _ _
  87         _ <= RationalMajorant.factorialTailPartial C start terms +
  88             RationalMajorant.factorialTailTerm C (start + terms) :=
  89           rat_add_le_add ih hterm
  90 
  91 /-- Explicit finite Taylor-prefix remainder bound.  If the first omitted
  92 degree is far enough into the factorial tail, every later finite prefix is
  93 within twice that first omitted majorant term. -/
  94 theorem expTaylorPrefix_remainder_le_two_first_omitted
  95     {C x : Rat} (hC : 0 <= C) (hx : qabs x <= C)
  96     (n terms : Nat)
  97     (hstart : C <= (((n + 2 : Nat) : Rat) / 2)) :
  98     qabs
  99         (FinitePolynomial.expTaylorPrefix (n + terms) x -
 100           FinitePolynomial.expTaylorPrefix n x) <=
 101       2 * RationalMajorant.factorialTailTerm C (n + 1) := by
 102   rw [← remainderPartial_eq_expTaylorPrefix_sub]
 103   exact Rat.le_trans
 104     (remainderPartial_abs_le_factorialTailPartial hC hx (n + 1) terms)
 105     (RationalMajorant.factorialTailPartial_bound hC hstart terms)
 106 
 107 /-- Executable first omitted degree for a bounded interval and a positive
 108 rational error request. -/
 109 def scheduledTailStart (C : Rat) (eps : QPos) : Nat :=
 110   RationalMajorant.factorialTailStart C +
 111     RationalMajorant.halfDecayShift
 112       (2 * RationalMajorant.factorialTailTerm C
 113         (RationalMajorant.factorialTailStart C)) eps
 114 
 115 /-- Degree of the finite prefix immediately before `scheduledTailStart`. -/
 116 def scheduledPrefixDegree (C : Rat) (eps : QPos) : Nat :=
 117   scheduledTailStart C eps - 1
 118 
 119 theorem scheduledTailStart_pos (C : Rat) (eps : QPos) :
 120     0 < scheduledTailStart C eps := by
 121   unfold scheduledTailStart RationalMajorant.factorialTailStart
 122   omega
 123 
 124 theorem scheduledPrefixDegree_succ (C : Rat) (eps : QPos) :
 125     scheduledPrefixDegree C eps + 1 = scheduledTailStart C eps := by
 126   unfold scheduledPrefixDegree
 127   have hpos := scheduledTailStart_pos C eps
 128   omega

 130 /-- Potential-infinity form of the exponential Taylor remainder theorem.
 131 
 132 For every rational `x` in `[-C,C]`, the scheduled finite prefix is within
 133 `eps` of every extension by any finite number of terms.  The quantification
 134 is solely over later finite stages. -/
 135 theorem scheduled_expTaylorPrefix_remainder_le
 136     {C x : Rat} (hC : 0 <= C) (hx : qabs x <= C)
 137     (eps : QPos) (extraTerms : Nat) :
 138     qabs
 139         (FinitePolynomial.expTaylorPrefix
 140             (scheduledPrefixDegree C eps + extraTerms) x -
 141           FinitePolynomial.expTaylorPrefix
 142             (scheduledPrefixDegree C eps) x) <=
 143       eps.val := by
 144   have hfinite := remainderPartial_abs_le_factorialTailPartial
 145     hC hx (scheduledTailStart C eps) extraTerms
 146   have heps := RationalMajorant.factorialTailPartial_shifted_le_eps
 147     hC eps extraTerms
 148   have hbound :
 149       qabs (remainderPartial x (scheduledTailStart C eps) extraTerms) <=
 150         eps.val := by
 151     exact Rat.le_trans hfinite (by simpa [scheduledTailStart] using heps)
 152   rw [← scheduledPrefixDegree_succ C eps] at hbound
 153   rw [remainderPartial_eq_expTaylorPrefix_sub] at hbound
 154   exact hbound
 155 
 156 /-- Explicit rational enclosure for every later finite exponential Taylor
 157 prefix at every point of the bounded interval `|x| <= C`. -/
 158 theorem scheduled_expTaylorPrefix_enclosure
 159     {C x : Rat} (hC : 0 <= C) (hx : qabs x <= C)
 160     (eps : QPos) (extraTerms : Nat) :
 161     let center := FinitePolynomial.expTaylorPrefix
 162       (scheduledPrefixDegree C eps) x
 163     let later := FinitePolynomial.expTaylorPrefix
 164       (scheduledPrefixDegree C eps + extraTerms) x
 165     center - eps.val <= later /\ later <= center + eps.val := by
 166   dsimp
 167   let difference : Rat :=
 168     FinitePolynomial.expTaylorPrefix
 169         (scheduledPrefixDegree C eps + extraTerms) x -
 170       FinitePolynomial.expTaylorPrefix (scheduledPrefixDegree C eps) x
 171   have hrem : qabs difference <= eps.val := by
 172     simpa [difference] using
 173       scheduled_expTaylorPrefix_remainder_le hC hx eps extraTerms
 174   have hlo : -eps.val <= difference :=
 175     Rat.le_trans (Rat.neg_le_neg hrem) (neg_qabs_le_self difference)
 176   have hhi : difference <= eps.val :=
 177     Rat.le_trans (self_le_qabs difference) hrem
 178   dsimp [difference] at hlo hhi
 179   constructor <;> grind [Rat.sub_eq_add_neg]
 180 
 181 end FiniteExponentialTaylor
