Proof bundle

ComputableAnalysis.ModularForms.EntireExponential

   1 import ComputableAnalysis.ModularForms.ExponentialCharts
   2 import ComputableAnalysis.RiemannHilbert.DomainHolomorphicLocality
   3 
   4 /-! Entire represented exponential obtained from agreeing factorial-series
   5 charts. Every point has an executable evaluator; internal chart radii are hidden. -/
   6 namespace ComputableAnalysis.ModularForms
   7 open ComplexRaw FunctionTheory RiemannHilbert
   8 
   9 /-- A chart large enough for the actual input's initial rational box. -/
  10 def exponentialInputRadius (z : Scalar) : QPos :=
  11   ⟨LocalODE.boxCoordinateBound (z.val.compute 0)+1,
  12     by have h := LocalODE.boxCoordinateBound_nonneg (z.val.compute 0); grind only⟩
  13 
  14 theorem exponentialInputRadius_mem (z : Scalar) :
  15     (exponentialChart (exponentialInputRadius z)).domain z := by
  16   refine ⟨LocalODE.boxCoordinateBound (z.val.compute 0),
  17     LocalODE.boxCoordinateBound_nonneg _, ?_, LocalODE.small_from_box z.val z.property 0⟩
  18   change LocalODE.boxCoordinateBound (z.val.compute 0) <
  19     LocalODE.boxCoordinateBound (z.val.compute 0)+1
  20   grind only
  21 
  22 /-- The value is computed using a certified chart chosen from the input box. -/
  23 def entireExponentialValue (z : Scalar) : Scalar :=
  24   ⟨(exponentialChart (exponentialInputRadius z)).eval z,
  25     (exponentialChart (exponentialInputRadius z)).valid z (exponentialInputRadius_mem z)⟩
  26 
  27 theorem entireExponentialValue_congr (z w : Scalar) (hzw : z.val.Equiv w.val) :
  28     (entireExponentialValue z).val.Equiv (entireExponentialValue w).val :=
  29   exponentialChart_agreement _ _ z w (exponentialInputRadius_mem z) (exponentialInputRadius_mem w) hzw
  30 
  31 def entireExponential : DomainFunctions.Map where
  32   domain _ := True
  33   eval z _ := entireExponentialValue z
  34   domain_congr _ _ _ := Iff.rfl
  35   eval_congr z w _ _ hzw := entireExponentialValue_congr z w hzw
  36 
  37 /-- Every supplied exponential chart agrees with the global evaluator. -/
  38 theorem entireExponential_chart (R : QPos) (z : Scalar) (hz : (exponentialChart R).domain z) :
  39     ((exponentialChart R).eval z).Equiv (entireExponentialValue z).val :=
  40   exponentialChart_agreement R (exponentialInputRadius z) z z hz
  41     (exponentialInputRadius_mem z) (equiv_refl _ z.property)
  42 
  43 /-- Actual holomorphicity on all valid represented complex inputs, derived
  44 from local derivatives and proved overlap agreement. -/
  45 def entireExponential_holomorphic : DomainFunctions.Holomorphic entireExponential :=
  46   DomainFunctions.holomorphic_of_local entireExponential
  47     (fun a _ => DomainFunctions.ofCertified (exponentialChart (exponentialInputRadius a)))
  48     (fun a _ => DomainFunctions.ofCertifiedHolomorphic (exponentialChart_holomorphic (exponentialInputRadius a)))
  49     (fun a _ => exponentialInputRadius_mem a)
  50     (fun _ _ _ _ => trivial)
  51     (fun a _ z hz => entireExponential_chart (exponentialInputRadius a) z hz)
  52 
  53 end ComputableAnalysis.ModularForms
