  10 theorem real_embedding_scale (r : Rat) (x : RealInput) :
  11     (scaleRat r (realAxis x).val).Equiv (realAxis ⟨RealRaw.scaleRat r x.val,RealRaw.scaleRat_valid x.property⟩).val := by
  12   intro n
  13   apply (compareAt_overlap_iff _ _ n n).mpr
  14   have ho := RealRaw.interval_order_of_valid (RealRaw.scaleRat r x.val) (RealRaw.scaleRat_valid x.property) n
  15   simp only [realAxis,ofRealRaw,scaleRat,QBox.scaleRat,RealRaw.scaleRat,RealRaw.scaleRatCompute,Rat.mul_zero]
  16   simp only [RealRaw.scaleRat,RealRaw.scaleRatCompute] at ho
  17   by_cases hr : 0≤r
  18   all_goals
  19     simp only [hr,if_true,if_false] at ho ⊢
  20     constructor <;> constructor <;> first | exact ho | exact Rat.le_refl

  22 theorem small_signed_scale (z : Scalar) (r B : Rat) (h : Small z.val B) :
  23     Small (scaleRat r z.val) (qabs r*B) := by
  24   by_cases hr : 0≤r
  25   · rw [qabs_eq_self_of_nonneg hr]; exact small_scale hr h
  26   · have hnr : 0≤ -r := by grind only
  27     have hs := SeriesLimitLaws.small_neg (small_scale hnr h)
  28     have he : (neg (scaleRat (-r) z.val)).Equiv (scaleRat r z.val) := by
  29       apply ComplexRawQuotient.equiv_of_ofRaw_eq
  30         (hleft := neg_valid (scaleRat_valid z.property)) (hright := scaleRat_valid z.property)
  31       rw [ComplexRawQuotient.ofRaw_neg _ (scaleRat_valid z.property),
  32         ComplexRawQuotient.ofRaw_scaleRat _ _ z.property,
  33         ComplexRawQuotient.ofRaw_scaleRat _ _ z.property,ComplexRawQuotient.neg_scaleRat]
  34       rw [Rat.neg_neg]
  35     have hq : qabs r= -r := by unfold qabs; rw [if_pos (show r<0 by grind only)]
  36     rw [hq]
  37     exact Small.congr (neg_valid (scaleRat_valid z.property)) (scaleRat_valid z.property) he hs

  39 theorem segment_difference_small (x : PositiveInput) (u v : RationalUnit) :
  40     Small (sub (realAxis (positiveSegment x v)).val (realAxis (positiveSegment x u)).val)
  41       (segmentBound x*qabs (v.val-u.val)) := by
  42   have h := small_signed_scale (realAxis (segmentDisplacement x)) (v.val-u.val)
  43     (segmentBound x) (scalar_small _)
  44   have hs := Small.congr (scaleRat_valid (realAxis (segmentDisplacement x)).property)
  45     (sub_valid (realAxis (positiveSegment x v)).property (realAxis (positiveSegment x u)).property)
  46     (equiv_symm (positiveSegment_difference x u v)) h
  47   simpa only [Rat.mul_comm] using hs

  53 theorem segmentInverse_real (x : PositiveInput) (t : RationalUnit) :
  54     (segmentInverse x t).val.Equiv (realAxis (segmentReciprocal x t)).val :=
  55   positive_inverse_agreement (positivePoint x t) (segmentLower x) (positiveSegment_lower x t)

  57 theorem segmentInverse_bound (x : PositiveInput) (t : RationalUnit) :
  58     Small (segmentInverse x t).val (1/(segmentLower x).val) :=
  59   Small.congr (realAxis (segmentReciprocal x t)).property (segmentInverse x t).property
  60     (equiv_symm (segmentInverse_real x t))
  61     (positiveReciprocal_embedding_small _ (positiveSegment x t).property (segmentLower x).val
  62       (segmentLower x).property (positiveSegment_lower x t))

  64 theorem segmentIntegrand_embedding (x : PositiveInput) (t : RationalUnit) :
  65     (mul (realAxis (segmentDisplacement x)).val (segmentInverse x t).val).Equiv
  66       (realAxis (segmentIntegrandValue x t)).val :=
  67   equiv_trans (mul_valid (realAxis (segmentDisplacement x)).property (segmentInverse x t).property)
  68     (mul_valid (realAxis (segmentDisplacement x)).property (realAxis (segmentReciprocal x t)).property)
  69     (realAxis (segmentIntegrandValue x t)).property
  70     (mul_equiv (realAxis (segmentDisplacement x)).property (realAxis (segmentDisplacement x)).property
  71       (segmentInverse x t).property (realAxis (segmentReciprocal x t)).property
  72       (equiv_refl _ (realAxis (segmentDisplacement x)).property) (segmentInverse_real x t))
  73     (real_embedding_mul _ _ (segmentDisplacement x).property (segmentReciprocal x t).property)

  75 def segmentVariation (x : PositiveInput) : Rat :=
  76   8*(segmentBound x)^2*(1/(segmentLower x).val)^2

  78 theorem segmentIntegrand_variation (x : PositiveInput) (u v : RationalUnit) :
  79     Small (sub (realAxis (segmentIntegrandValue x v)).val (realAxis (segmentIntegrandValue x u)).val)
  80       (segmentVariation x*qabs (v.val-u.val)) := by
  81   let U := realAxis (positiveSegment x u)
  82   let V := realAxis (positiveSegment x v)
  83   let A := segmentInverse x u
  84   let B := segmentInverse x v
  85   let D := realAxis (segmentDisplacement x)
  86   let R := 1/(segmentLower x).val
  87   let H := qabs (v.val-u.val)
  88   have hR : 0≤R := by dsimp [R]; rw [Rat.div_def,Rat.one_mul]; exact Rat.le_of_lt (Rat.inv_pos.mpr (segmentLower x).property)
  89   have hD := Rat.le_of_lt (segmentBound_pos x)
  90   have hH := qabs_nonneg (v.val-u.val)
  91   have hdiff := segment_difference_small x v u
  92   rw [show qabs (u.val-v.val)=qabs (v.val-u.val) by rw [show u.val-v.val= -(v.val-u.val) by grind only,qabs_neg]] at hdiff
  93   have hprod := Small.mul B.property (sub_valid U.property V.property) hR (Rat.mul_nonneg hD hH)
  94     (segmentInverse_bound x v) hdiff
  95   have hprod2 := Small.mul (mul_valid B.property (sub_valid U.property V.property)) A.property
  96     (Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hR) (Rat.mul_nonneg hD hH)) hR
  97     hprod (segmentInverse_bound x u)
  98   have hall := Small.mul D.property (mul_valid (mul_valid B.property (sub_valid U.property V.property)) A.property)
  99     hD (Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel)
 100       (Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hR) (Rat.mul_nonneg hD hH))) hR)
 101     (scalar_small D) hprod2
 102   have ha := ComplexRawQuotient.ofRaw_eq_ofRaw
 103     (hleft := mul_valid U.property A.property) (hright := ofQComplex_valid _)
 104     (RepresentedReciprocal.mul_inverse U (positive_real_nonzero (positiveSegment x u) (positivePoint x u).property))
 105   have hb := ComplexRawQuotient.ofRaw_eq_ofRaw
 106     (hleft := mul_valid V.property B.property) (hright := ofQComplex_valid _)
 107     (RepresentedReciprocal.mul_inverse V (positive_real_nonzero (positiveSegment x v) (positivePoint x v).property))
 108   have hprodEq : (mul D.val (mul (mul B.val (sub U.val V.val)) A.val)).Equiv
 109       (sub (mul D.val B.val) (mul D.val A.val)) := by
 110     apply ComplexRawQuotient.equiv_of_ofRaw_eq
 111       (hleft := mul_valid D.property (mul_valid (mul_valid B.property (sub_valid U.property V.property)) A.property))
 112       (hright := sub_valid (mul_valid D.property B.property) (mul_valid D.property A.property))
 113     change valueClass U*valueClass A=1 at ha
 114     change valueClass V*valueClass B=1 at hb
 115     change valueClass D*((valueClass B*(valueClass U-valueClass V))*valueClass A)=
 116       valueClass D*valueClass B-valueClass D*valueClass A
 117     grind only
 118   have hnames := FunctionTheory.sub_congr (segmentIntegrand_embedding x v) (segmentIntegrand_embedding x u)
 119   have he := equiv_trans
 120     (mul_valid D.property (mul_valid (mul_valid B.property (sub_valid U.property V.property)) A.property))
 121     (sub_valid (mul_valid D.property B.property) (mul_valid D.property A.property))
 122     (sub_valid (realAxis (segmentIntegrandValue x v)).property (realAxis (segmentIntegrandValue x u)).property)
 123     hprodEq hnames
 124   have hh := Small.congr
 125     (mul_valid D.property (mul_valid (mul_valid B.property (sub_valid U.property V.property)) A.property))
 126     (sub_valid (realAxis (segmentIntegrandValue x v)).property (realAxis (segmentIntegrandValue x u)).property) he hall
 127   have hc : 2*segmentBound x*(2*(2*R*(segmentBound x*H))*R)=segmentVariation x*H := by
 128     dsimp [segmentVariation]; simp only [Rat.pow_succ,Rat.pow_zero]; grind only
 129   change Small _ (2*segmentBound x*(2*(2*R*(segmentBound x*H))*R)) at hh
 130   rw [hc] at hh
 131   exact hh
