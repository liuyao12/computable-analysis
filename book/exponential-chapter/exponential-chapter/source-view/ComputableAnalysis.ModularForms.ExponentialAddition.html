   9 theorem exponential_series_addition (x y : ComplexRaw) (hx : x.Valid) (hy : y.Valid)
  10     (R : QPos) (hsx : Small x R.val) (hsy : Small y R.val) (hs : Small (add x y) R.val) :
  11     (mul
  12       (BoundedSeries.sumValue exponentialCoefficients x exponentialCoefficients_valid hx
  13         (exponentialBudget (exponentialRatio R)) (exponentialRatio R) R.val)
  14       (BoundedSeries.sumValue exponentialCoefficients y exponentialCoefficients_valid hy
  15         (exponentialBudget (exponentialRatio R)) (exponentialRatio R) R.val)).Equiv
  16     (BoundedSeries.sumValue exponentialCoefficients (add x y) exponentialCoefficients_valid
  17       (add_valid hx hy) (exponentialBudget (exponentialRatio R)) (exponentialRatio R) R.val) := by
  18   let C := exponentialBudget (exponentialRatio R)
  19   let q := 2*exponentialRatio R*R.val
  20   let t := LocalODE.seriesTerm exponentialCoefficients x
  21   let u := LocalODE.seriesTerm exponentialCoefficients y
  22   have ht := LocalODE.seriesTerm_valid _ _ exponentialCoefficients_valid hx
  23   have hu := LocalODE.seriesTerm_valid _ _ exponentialCoefficients_valid hy
  24   have hd := exponentialDiagonal_valid x y hx hy
  25   have hC : 0 ≤ C := exponentialBudget_nonnegative _ (exponentialRatio_positive R)
  26   have hK := Rat.le_of_lt (exponentialRatio_positive R)
  27   have hR := Rat.le_of_lt R.property
  28   have hq : 0 ≤ q := Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hK) hR
  29   have hlocal : q ≤ (1 : Rat)/2 := by
  30     have h := exponentialRatio_local R
  31     dsimp [q]
  32     grind
  33   have htB : ∀ n, Small (t n) (2*C*q^n) :=
  34     LocalODE.seriesTerm_bound _ _ exponentialCoefficients_valid hx C _ _ hC hK hR
  35       (exponentialCoefficients_small _ (exponentialRatio_positive R)) hsx
  36   have huB : ∀ n, Small (u n) (2*C*q^n) :=
  37     LocalODE.seriesTerm_bound _ _ exponentialCoefficients_valid hy C _ _ hC hK hR
  38       (exponentialCoefficients_small _ (exponentialRatio_positive R)) hsy
  39   have hdB : ∀ n, Small (exponentialDiagonal x y n) (2*C*q^n) :=
  40     exponentialDiagonal_bound x y hx hy R hs
  41   have hvT := ScalarSeries.value_valid t ht C q hC hq hlocal htB
  42   have hvU := ScalarSeries.value_valid u hu C q hC hq hlocal huB
  43   have hvD := exponentialDiagonalSum_valid x y hx hy R hs
  44   have he := LocalODE.tail_bound_shrinks C q hC hq hlocal
  45   have he0 (N : Nat) : 0 ≤ 4*C*q^N :=
  46     Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hC) (Rat.pow_nonneg hq)
  47   have hprod : (mul (ScalarSeries.value t ht C q) (ScalarSeries.value u hu C q)).Equiv
  48       (exponentialDiagonalSum x y hx hy R) := by
  49     apply product_limit_comparison _ _ _ hvT hvU hvD
  50       (fun N => ScalarSeries.block t 0 N) (fun N => ScalarSeries.block u 0 N)
  51       (fun N => ScalarSeries.block (exponentialDiagonal x y) 0 N)
  52       (ScalarSeries.block_valid t ht 0) (ScalarSeries.block_valid u hu 0)
  53       (ScalarSeries.block_valid _ hd 0)
  54       (fun N => 4*C*q^N) (fun N => 4*C*q^N) (fun N => 4*C*q^N)
  55       (fun N => 32*C*C*(8*exponentialRatio R*R.val)^N)
  56       he he he (exponentialProductRemainder_shrinks R) he0 he0
  57       (4*C) (4*C) (Rat.mul_nonneg (by decide +kernel) hC)
  58       (Rat.mul_nonneg (by decide +kernel) hC)
  59       ?_ (ScalarSeries.value_bound u hu C q hC hq hlocal huB)
  60       (ScalarSeries.value_close t ht C q hC hq hlocal htB)
  61       (ScalarSeries.value_close u hu C q hC hq hlocal huB)
  62       (ScalarSeries.value_close _ hd C q hC hq hlocal hdB)
  63       (exponential_finite_product_bound x y hx hy R hsx hsy)
  64     intro N
  65     simpa only [Rat.pow_zero, Rat.mul_one] using ScalarSeries.block_bound t C q hC hq hlocal htB 0 N
  66   exact equiv_trans (mul_valid hvT hvU) hvD
  67     (BoundedSeries.sumValue_valid _ _ exponentialCoefficients_valid (add_valid hx hy)
  68       C _ _ hC hK hR (exponentialCoefficients_small _ (exponentialRatio_positive R)) hs hlocal)
  69     hprod (exponentialDiagonalSum_eq_series x y hx hy R hs)

  71 /-- Exact addition for every pair of valid represented complex inputs;
  72 the internal rational chart choice is hidden from callers. -/
  73 theorem entireExponential_addition (x y : Scalar) :
  74     (mul (entireExponentialValue x).val (entireExponentialValue y).val).Equiv
  75       (entireExponentialValue ⟨add x.val y.val, add_valid x.property y.property⟩).val := by
  76   let A := boxCoordinateBound (x.val.compute 0)
  77   let B := boxCoordinateBound (y.val.compute 0)
  78   have hA : 0 ≤ A := boxCoordinateBound_nonneg _
  79   have hB : 0 ≤ B := boxCoordinateBound_nonneg _
  80   let R : QPos := ⟨A+B+1, by grind⟩
  81   have hxA : Small x.val A := small_from_box x.val x.property 0
  82   have hyB : Small y.val B := small_from_box y.val y.property 0
  83   have hxR : Small x.val R.val := hxA.mono (by dsimp [R]; grind)
  84   have hyR : Small y.val R.val := hyB.mono (by dsimp [R]; grind)
  85   have hsR : Small (add x.val y.val) R.val :=
  86     (small_add hxA hyB).mono (by dsimp [R]; grind)
  87   have hxD : (exponentialChart R).domain x := ⟨A,hA,by dsimp [R]; grind,hxA⟩
  88   have hyD : (exponentialChart R).domain y := ⟨B,hB,by dsimp [R]; grind,hyB⟩
  89   let z : Scalar := ⟨add x.val y.val, add_valid x.property y.property⟩
  90   have hzD : (exponentialChart R).domain z :=
  91     ⟨A+B,Rat.add_nonneg hA hB,by dsimp [R]; grind,small_add hxA hyB⟩
  92   have hxy := exponential_series_addition x.val y.val x.property y.property R hxR hyR hsR
  93   have hvalidX := (exponentialChart R).valid x hxD
  94   have hvalidY := (exponentialChart R).valid y hyD
  95   have hvalidZ := (exponentialChart R).valid z hzD
  96   exact equiv_trans
  97     (mul_valid (entireExponentialValue x).property (entireExponentialValue y).property)
  98     (mul_valid hvalidX hvalidY) (entireExponentialValue z).property
  99     (mul_equiv (entireExponentialValue x).property hvalidX
 100       (entireExponentialValue y).property hvalidY (equiv_symm (entireExponential_chart R x hxD))
 101       (equiv_symm (entireExponential_chart R y hyD)))
 102     (equiv_trans (mul_valid hvalidX hvalidY) hvalidZ (entireExponentialValue z).property
 103       hxy (entireExponential_chart R z hzD))
