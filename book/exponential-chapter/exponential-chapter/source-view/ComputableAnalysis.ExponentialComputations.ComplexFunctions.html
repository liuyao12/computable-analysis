  27 /-- A normalization is a logarithm of the center. Analytic and overlap laws
  28 are proved below from the actual series and are not fields of this record. -/
  29 structure LogSeed where
  30   center : Scalar
  31   nonzero : Nonzero center
  32   value : Scalar
  33   exponential : (exp value).val.Equiv center.val

  35 namespace LogSeed
  36 def function (s : LogSeed) : DomainFunctions.Map :=
  37   LogarithmBranchCharts.function s.center s.nonzero s.value
  38 def holomorphic (s : LogSeed) : DomainFunctions.Holomorphic s.function :=
  39   LogarithmBranchCharts.holomorphic s.center s.nonzero s.value
  40 theorem exponential_eval (s : LogSeed) (z : Scalar) (hz : s.function.domain z) :
  41     (exp (s.function.eval z hz)).val.Equiv z.val := by
  42   have ha := equiv_trans (MatrixExponential.scalarExponential s.value).property
  43     (exp s.value).property s.center.property (scalarExponential_agreement s.value) s.exponential
  44   exact equiv_trans (exp (s.function.eval z hz)).property
  45     (MatrixExponential.scalarExponential (s.function.eval z hz)).property z.property
  46     (equiv_symm (scalarExponential_agreement _))
  47     (LogarithmBranchCharts.exponential _ _ _ ha z hz)
  48 theorem initial (s : LogSeed) :
  49     (s.function.eval s.center (RelativeLogarithm.center_mem _ s.nonzero)).val.Equiv s.value.val :=
  50   LogarithmBranchCharts.initial _ _ _
  51 theorem derivative (s : LogSeed) (z : Scalar) (hz : s.function.domain z) :
  52     (s.holomorphic.derivative z hz).val.Equiv
  53       (RepresentedReciprocal.inverse z (RelativeLogarithm.nonzero _ s.nonzero z hz)).val :=
  54   LogarithmBranchCharts.derivative_reciprocal _ _ _ s.holomorphic z hz
  55 theorem congr (s t : LogSeed) (hc : s.center.val.Equiv t.center.val)

  61 def continueAlong (s : LogSeed) {d : Scalar} {hd : Nonzero d}
  62     (p : LogarithmContinuation.Chain s.center s.nonzero d hd) : LogSeed where
  63   center := d
  64   nonzero := hd
  65   value := LogarithmContinuation.value p s.value
  66   exponential := by

  70       (equiv_symm (scalarExponential_agreement _)) (LogarithmContinuation.value_exponential p s.value ha)

  80 /-- Constancy on the entire actual chart intersection, proved by finite
  81 telescoping with uniform remainders and convex segment coverage. -/
  82 theorem overlap_difference (s t : LogSeed) (p z : Scalar)
  83     (hp : s.function.domain p ∧ t.function.domain p)
  84     (hz : s.function.domain z ∧ t.function.domain z) :
  85     (sub (s.function.eval z hz.1).val (t.function.eval z hz.2).val).Equiv
  86       (sub (s.function.eval p hp.1).val (t.function.eval p hp.2).val) := by
  87   have hr := ComplexRawQuotient.ofRaw_eq_ofRaw
  88     (hleft := sub_valid ((RelativeLogarithm.function s.center s.nonzero).eval z hz.1).property
  89       ((RelativeLogarithm.function t.center t.nonzero).eval z hz.2).property)
  90     (hright := sub_valid ((RelativeLogarithm.function s.center s.nonzero).eval p hp.1).property
  91       ((RelativeLogarithm.function t.center t.nonzero).eval p hp.2).property)
  92     (relativeLogarithm_overlap_difference_constant s.center t.center s.nonzero t.nonzero p z hp hz)
  93   apply ComplexRawQuotient.equiv_of_ofRaw_eq
  94     (hleft := sub_valid (s.function.eval z hz.1).property (t.function.eval z hz.2).property)
  95     (hright := sub_valid (s.function.eval p hp.1).property (t.function.eval p hp.2).property)
  96   let A := scalarClass s.value
  97   let B := scalarClass t.value
  98   let CZ := scalarClass ((RelativeLogarithm.function s.center s.nonzero).eval z hz.1)
  99   let DZ := scalarClass ((RelativeLogarithm.function t.center t.nonzero).eval z hz.2)
 100   let CP := scalarClass ((RelativeLogarithm.function s.center s.nonzero).eval p hp.1)
 101   let DP := scalarClass ((RelativeLogarithm.function t.center t.nonzero).eval p hp.2)
 102   change CZ-DZ=CP-DP at hr
 103   change (A+CZ)-(B+DZ)=(A+CP)-(B+DP)
 104   grind only
 105 theorem overlap_agrees (s t : LogSeed) (p z : Scalar)
 106     (hp : s.function.domain p ∧ t.function.domain p)
 107     (hz : s.function.domain z ∧ t.function.domain z)
 108     (h : (s.function.eval p hp.1).val.Equiv (t.function.eval p hp.2).val) :
 109     (s.function.eval z hz.1).val.Equiv (t.function.eval z hz.2).val := by
 110   have hd := ComplexRawQuotient.ofRaw_eq_ofRaw
 111     (hleft := sub_valid (s.function.eval z hz.1).property (t.function.eval z hz.2).property)
 112     (hright := sub_valid (s.function.eval p hp.1).property (t.function.eval p hp.2).property)
 113     (s.overlap_difference t p z hp hz)
 114   have he := ComplexRawQuotient.ofRaw_eq_ofRaw
 115     (hleft := (s.function.eval p hp.1).property) (hright := (t.function.eval p hp.2).property) h
 116   apply ComplexRawQuotient.equiv_of_ofRaw_eq
 117     (hleft := (s.function.eval z hz.1).property) (hright := (t.function.eval z hz.2).property)
 118   let SZ := scalarClass (s.function.eval z hz.1)
 119   let TZ := scalarClass (t.function.eval z hz.2)
 120   let SP := scalarClass (s.function.eval p hp.1)
 121   let TP := scalarClass (t.function.eval p hp.2)
 122   change SZ-TZ=SP-TP at hd
 123   change SP=TP at he
 124   change SZ=TZ
 125   grind only
 126 end LogSeed

 128 /-- Executable selection of a normalized germ at every nonzero input. The
 129 selected pointwise values are not claimed to form a global analytic log. -/
 130 def logarithmAt (z : Scalar) (hz : Nonzero z) : LogSeed where
 131   center := z
 132   nonzero := hz
 133   value := NonzeroLogarithmConstruction.value z hz
 134   exponential := equiv_trans (exp _).property (MatrixExponential.scalarExponential _).property z.property
 135     (equiv_symm (scalarExponential_agreement _)) (NonzeroLogarithmConstruction.exponential z hz)

 137 /-- Choose the real normalization before extending into a complex chart. -/
 138 def realLogarithmAt (x : PositiveInput) : LogSeed where
 139   center := realAxis x.val
 140   nonzero := positive_real_nonzero x.val x.property
 141   value := realAxis (log x)
 142   exponential := equiv_trans (exp _).property (realAxis (ExponentialComputations.exp (log x))).property
 143     (realAxis x.val).property (exp_real (log x))
 144     (real_embedding_congr (ExponentialComputations.exp (log x)).property x.val.property (exp_log x))

 146 theorem realLogarithmAt_initial (x : PositiveInput) :
