   3 /-! Finite continuation of actual logarithm germs. A chain records only
   4 local domain membership; its continued values, exponential identity and
   5 open overlap agreement are proved from the constructed Taylor charts. -/
   6 namespace ComputableAnalysis.RiemannHilbert.LogarithmContinuation
   7 open ComplexRaw FunctionTheory LocalODE DomainFunctions NonzeroBoxSearch
   8 set_option maxHeartbeats 1000000

  10 inductive Chain : (c : Scalar) → Nonzero c → (d : Scalar) → Nonzero d → Type where
  11   | nil (c : Scalar) (hc : Nonzero c) : Chain c hc c hc
  12   | step {c d e : Scalar} {hc : Nonzero c} {hd : Nonzero d} {he : Nonzero e}
  13       (hcd : RelativeLogarithm.domain c hc d) (tail : Chain d hd e he) : Chain c hc e he
  14 
  15 def value {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d} : Chain c hc d hd → Scalar → Scalar
  16   | .nil _ _, a => a
  17   | .step hcd tail, a => value tail ((LogarithmBranchCharts.function _ _ a).eval _ hcd)
  18 
  19 def terminalFunction {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  20     (p : Chain c hc d hd) (a : Scalar) : DomainFunctions.Map :=
  21   LogarithmBranchCharts.function d hd (value p a)
  22 
  23 def terminalHolomorphic {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  24     (p : Chain c hc d hd) (a : Scalar) : DomainFunctions.Holomorphic (terminalFunction p a) :=
  25   LogarithmBranchCharts.holomorphic d hd (value p a)
  26 
  27 theorem value_congr {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  28     (p : Chain c hc d hd) (a b : Scalar) (hab : a.val.Equiv b.val) :
  29     (value p a).val.Equiv (value p b).val := by
  30   induction p generalizing a b with
  31   | nil => exact hab
  32   | @step c d e hc hd he hcd tail ih =>
  33       exact ih _ _ (LogarithmBranchCharts.value_congr c c hc hc a b
  34         (equiv_refl _ c.property) hab d d hcd hcd (equiv_refl _ d.property))
  35 
  36 theorem value_exponential {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  37     (p : Chain c hc d hd) (a : Scalar)
  38     (ha : (MatrixExponential.scalarExponential a).val.Equiv c.val) :
  39     (MatrixExponential.scalarExponential (value p a)).val.Equiv d.val := by
  40   induction p generalizing a with
  41   | nil => exact ha
  42   | step hcd tail ih => exact ih _ (LogarithmBranchCharts.exponential _ _ a ha _ hcd)
  43 
  44 theorem terminal_exponential {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  45     (p : Chain c hc d hd) (a : Scalar)
  46     (ha : (MatrixExponential.scalarExponential a).val.Equiv c.val)
  47     (z : Scalar) (hz : (terminalFunction p a).domain z) :
  48     (MatrixExponential.scalarExponential ((terminalFunction p a).eval z hz)).val.Equiv z.val :=
  49   LogarithmBranchCharts.exponential d hd (value p a) (value_exponential p a ha) z hz
  50 
  51 theorem terminal_initial {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  52     (p : Chain c hc d hd) (a : Scalar) :
  53     ((terminalFunction p a).eval d (RelativeLogarithm.center_mem d hd)).val.Equiv (value p a).val :=
  54   LogarithmBranchCharts.initial d hd (value p a)
  55 
  56 theorem terminal_derivative {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  57     (p : Chain c hc d hd) (a : Scalar) (hF : DomainFunctions.Holomorphic (terminalFunction p a))
  58     (z : Scalar) (hz : (terminalFunction p a).domain z) :
  59     (hF.derivative z hz).val.Equiv (RepresentedReciprocal.inverse z (RelativeLogarithm.nonzero d hd z hz)).val :=
  60   LogarithmBranchCharts.derivative_reciprocal d hd (value p a) hF z hz
  61 
  62 theorem edge_segment (c d : Scalar) (hc : Nonzero c) (hcd : RelativeLogarithm.domain c hc d)
  63     (t : Rat) (ht : UniformPath.unitInterval t) :
  64     RelativeLogarithm.domain c hc (AffineSegment.point c d t) :=
  65   RelativeLogarithm.affine_mem c hc c d (RelativeLogarithm.center_mem c hc) hcd t ht
  66 
  67 theorem edge_germ (c d : Scalar) (hc : Nonzero c) (hd : Nonzero d)
  68     (hcd : RelativeLogarithm.domain c hc d) (a z : Scalar)
  69     (hz : Small (sub z.val d.val) (LogarithmBranchCharts.overlapRadius c d hc hd hcd).val) :
  70     ((LogarithmBranchCharts.function c hc a).eval z
  71       (LogarithmBranchCharts.overlapRadius_inside c d hc hd hcd z hz).1).val.Equiv
  72     ((LogarithmBranchCharts.function d hd ((LogarithmBranchCharts.function c hc a).eval d hcd)).eval z
  73       (LogarithmBranchCharts.overlapRadius_inside c d hc hd hcd z hz).2).val :=
  74   LogarithmBranchCharts.recenter_agreement c d hc hd a hcd z
  75     (LogarithmBranchCharts.overlapRadius_inside c d hc hd hcd z hz)

  77 def append {c d e : Scalar} {hc : Nonzero c} {hd : Nonzero d} {he : Nonzero e}
  78     (p : Chain c hc d hd) (q : Chain d hd e he) : Chain c hc e he :=
  79   match p with
  80   | .nil _ _ => q
  81   | .step hcd tail => .step hcd (append tail q)
  82 
  83 theorem value_append {c d e : Scalar} {hc : Nonzero c} {hd : Nonzero d} {he : Nonzero e}
  84     (p : Chain c hc d hd) (q : Chain d hd e he) (a : Scalar) :
  85     (value (append p q) a).val.Equiv (value q (value p a)).val := by
  86   induction p generalizing a with
  87   | nil => exact equiv_refl _ (value q a).property
  88   | step hcd tail ih => exact ih q _

  90 /-- An admissible subdivision has the same continued value. All three
  91 edge domain hypotheses are genuine local chart membership, not assumed
  92 agreement of the continued logarithms. -/
  93 theorem subdivision {c d e f : Scalar} {hc : Nonzero c} {hd : Nonzero d}
  94     {he : Nonzero e} {hf : Nonzero f} (hce : RelativeLogarithm.domain c hc e)
  95     (hcd : RelativeLogarithm.domain c hc d) (hde : RelativeLogarithm.domain d hd e)
  96     (tail : Chain e he f hf) (a : Scalar) :
  97     (value (.step hce tail) a).val.Equiv (value (.step hcd (.step hde tail)) a).val :=
  98   value_congr tail _ _ (LogarithmBranchCharts.recenter_agreement c d hc hd a hcd e ⟨hce,hde⟩)
  99 
 100 theorem terminal_agreement {c d : Scalar} {hc : Nonzero c} {hd : Nonzero d}
 101     (p q : Chain c hc d hd) (a b : Scalar) (h : (value p a).val.Equiv (value q b).val)
 102     (z w : Scalar) (hz : (terminalFunction p a).domain z) (hw : (terminalFunction q b).domain w)
 103     (hzw : z.val.Equiv w.val) :
 104     ((terminalFunction p a).eval z hz).val.Equiv ((terminalFunction q b).eval w hw).val :=
 105   LogarithmBranchCharts.value_congr d d hd hd _ _ (equiv_refl _ d.property) h z w hz hw hzw
