   8 /-- The local logarithm differs from its linear term by a quadratic error,
   9 for every valid represented complex input in its actual chart. -/
  10 theorem localLogarithm_linear_error (z : Scalar)
  11     (hz : interior LocalLogarithm.radius.val z) (H : Rat) (hH : 0 ≤ H)
  12     (hsmall : Small z.val H) :
  13     Small (sub (LocalLogarithm.function.eval z hz).val z.val) (16*H^2) := by
  14   let L := LocalLogarithm.function.eval z hz
  15   let a : Scalar := ⟨zero,ofQComplex_valid _⟩
  16   have h0 := LocalLogarithm.initial
  17   have hd := BoundedSeries.sumDerivative_zero LocalLogarithm.coefficient
  18     LocalLogarithm.coefficient_valid 1 1 LocalLogarithm.radius.val
  19     (by decide +kernel) (by decide +kernel) (by decide +kernel)
  20     LocalLogarithm.coefficient_bound (by decide +kernel)
  21   have h1 : (LocalLogarithm.coefficient 1).Equiv one := by
  22     change (scaleRat (1/1) one).Equiv one
  23     rw [show (1:Rat)/1=1 by decide +kernel]
  24     exact scaleRat_one_equiv one (ofQComplex_valid _)
  25   have he0 : (sub z.val zero).Equiv z.val := by
  26     apply ComplexRawQuotient.equiv_of_ofRaw_eq
  27       (hleft := sub_valid z.property (ofQComplex_valid _)) (hright := z.property)
  28     change ComplexRawQuotient.ofRaw z.val z.property-0=ComplexRawQuotient.ofRaw z.val z.property
  29     grind only
  30   have hb := BoundedSeries.sum_remainder_bound LocalLogarithm.coefficient zero z.val
  31     LocalLogarithm.coefficient_valid (ofQComplex_valid _) z.property
  32     1 1 LocalLogarithm.radius.val H
  33     (by decide +kernel) (by decide +kernel) (by decide +kernel) hH
  34     LocalLogarithm.coefficient_bound (Small.zero (by decide +kernel))
  35     (interior_bound _ z hz)
  36     (Small.congr z.property (sub_valid z.property (ofQComplex_valid _))
  37       (equiv_symm he0) hsmall) (by decide +kernel)
  38   have hD := equiv_trans
  39     (BoundedSeries.sumDerivative_valid LocalLogarithm.coefficient zero
  40       LocalLogarithm.coefficient_valid (ofQComplex_valid _) 1 1 LocalLogarithm.radius.val
  41       (by decide +kernel) (by decide +kernel) (by decide +kernel)
  42       LocalLogarithm.coefficient_bound (Small.zero (by decide +kernel)) (by decide +kernel))
  43     (LocalLogarithm.coefficient_valid 1) (ofQComplex_valid _) hd h1
  44   have hDv := BoundedSeries.sumDerivative_valid LocalLogarithm.coefficient zero
  45     LocalLogarithm.coefficient_valid (ofQComplex_valid _) 1 1 LocalLogarithm.radius.val

  50   have hh := FunctionTheory.sub_congr
  51     (FunctionTheory.sub_congr (equiv_refl _ L.property) h0) hm
  52   have he2 : (sub (sub L.val zero) (mul one z.val)).Equiv (sub L.val z.val) := by
  53     apply ComplexRawQuotient.equiv_of_ofRaw_eq
  54       (hleft := sub_valid (sub_valid L.property (ofQComplex_valid _))
  55         (mul_valid (ofQComplex_valid _) z.property)) (hright := sub_valid L.property z.property)
  56     change (ComplexRawQuotient.ofRaw L.val L.property-0)-1*ComplexRawQuotient.ofRaw z.val z.property=
  57       ComplexRawQuotient.ofRaw L.val L.property-ComplexRawQuotient.ofRaw z.val z.property
  58     grind only
  59   have hvalid := SeriesLimitLaws.remainder_valid _ _ _ _ L.property
  60     (LocalLogarithm.function.eval a (interior_zero _ LocalLogarithm.radius.property)).property
  61     (BoundedSeries.sumDerivative_valid _ _ LocalLogarithm.coefficient_valid (ofQComplex_valid _)
  62       1 1 LocalLogarithm.radius.val (by decide +kernel) (by decide +kernel)
  63       (by decide +kernel) LocalLogarithm.coefficient_bound (Small.zero (by decide +kernel))
  64       (by decide +kernel)) (sub_valid z.property (ofQComplex_valid QComplex.zero))
  65   have he := equiv_trans hvalid
  66     (sub_valid (sub_valid L.property (ofQComplex_valid _))
  67       (mul_valid (ofQComplex_valid _) z.property)) (sub_valid L.property z.property) hh he2
  68   have hh := Small.congr hvalid (sub_valid L.property z.property) he hb
  69   rw [show (1:Rat)^2=1 by decide +kernel] at hh
