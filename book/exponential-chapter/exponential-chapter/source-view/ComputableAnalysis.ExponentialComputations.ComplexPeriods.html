  11 /-- The represented value `2*pi*i*k`, using the geometric half-pi. -/
  12 def period (k : Int) : Scalar :=
  13   ModularForms.scalarProduct nomeSlope ⟨ofQComplex ⟨(k:Rat),0⟩,ofQComplex_valid _⟩
  14 
  15 theorem exp_period (k : Int) : (exp (period k)).val.Equiv one := by
  16   let o : Scalar := ⟨zero,ofQComplex_valid _⟩
  17   let c : Scalar := ⟨ofQComplex ⟨(k:Rat),0⟩,ofQComplex_valid _⟩
  18   have hc : (integerShiftScalar o k).val.Equiv c.val := by
  19     intro n
  20     apply (compareAt_overlap_iff _ _ n n).mpr
  21     simp [integerShiftScalar,integerAffine,o,c,zero,translate,scaleRat,ofQComplex,add,
  22       QBox.scaleRat,QBox.add,QComplex.add,QComplex.zero,QBox.Overlaps,QComplex.le_def,Rat.zero_add]
  23   have ho : (ModularForms.scalarProduct nomeSlope o).val.Equiv zero := mul_zero_equiv _ nomeSlope.property
  24   exact equiv_trans (exp (period k)).property (entireNomeMap.eval c (entireNomeMap_mem c)).property
  25     (ofQComplex_valid _) (equiv_symm (entireNomeMap_exponent c))
  26     (equiv_trans (entireNomeMap.eval c (entireNomeMap_mem c)).property
  27       (entireNomeMap.eval (integerShiftScalar o k) (entireNomeMap_mem _)).property (ofQComplex_valid _)
  28       (entireNomeMap.eval_congr _ _ _ _ (equiv_symm hc))
  29       (equiv_trans (entireNomeMap.eval (integerShiftScalar o k) (entireNomeMap_mem _)).property
  30         (entireNomeMap.eval o (entireNomeMap_mem o)).property (ofQComplex_valid _)
  31         (entireNomeMap_period_int o k)
  32         (equiv_trans (entireNomeMap.eval o (entireNomeMap_mem o)).property
  33           (exp (ModularForms.scalarProduct nomeSlope o)).property (ofQComplex_valid _) (entireNomeMap_exponent o)
  34           (equiv_trans (exp (ModularForms.scalarProduct nomeSlope o)).property (exp o).property (ofQComplex_valid _)
  35             (exp_congr _ _ ho) entireExponential_zero))))

  37 theorem exp_translate_period (z : Scalar) (k : Int) :
  38     (exp (scalarSum z (period k))).val.Equiv (exp z).val :=
  39   equiv_trans (exp (scalarSum z (period k))).property
  40     (mul_valid (exp z).property (exp (period k)).property) (exp z).property
  41     (equiv_symm (exp_add z (period k)))
  42     (equiv_trans (mul_valid (exp z).property (exp (period k)).property)
  43       (mul_valid (exp z).property (ofQComplex_valid _)) (exp z).property
  44       (mul_equiv (exp z).property (exp z).property (exp (period k)).property (ofQComplex_valid _)
  45         (equiv_refl _ (exp z).property) (exp_period k)) (mul_one_equiv _ (exp z).property))

  47 theorem exp_kernel (z : Scalar) (he : (exp z).val.Equiv one) :
  48     ∃ k : Int, z.val.Equiv (period k).val := by
  49   let w := ModularForms.scalarProduct nomeInverseSlope z
  50   have hi := ComplexRawQuotient.ofRaw_eq_ofRaw
  51     (hleft := mul_valid nomeSlope.property nomeInverseSlope.property) (hright := ofQComplex_valid _)
  52     (RepresentedReciprocal.mul_inverse nomeSlope nomeSlope_nonzero)
  53   have hw : (ModularForms.scalarProduct nomeSlope w).val.Equiv z.val := by
  54     apply ComplexRawQuotient.equiv_of_ofRaw_eq
  55       (hleft := (ModularForms.scalarProduct nomeSlope w).property) (hright := z.property)
  56     let S := scalarClass nomeSlope
  57     let I := scalarClass nomeInverseSlope
  58     let Z := scalarClass z
  59     change S*I=1 at hi
  60     change S*(I*Z)=Z
  61     grind only
  62   have hn := equiv_trans (entireNomeMap.eval w (entireNomeMap_mem w)).property
  63     (exp (ModularForms.scalarProduct nomeSlope w)).property (ofQComplex_valid _) (entireNomeMap_exponent w)
  64     (equiv_trans (exp (ModularForms.scalarProduct nomeSlope w)).property (exp z).property (ofQComplex_valid _)
  65       (exp_congr _ _ hw) he)
  66   obtain ⟨k,hk⟩ := entireNome_kernel_exists_integer w hn
  67   exact ⟨k,equiv_trans z.property (ModularForms.scalarProduct nomeSlope w).property (period k).property
  68     (equiv_symm hw) (mul_equiv nomeSlope.property nomeSlope.property w.property
  69       (ofQComplex_valid _) (equiv_refl _ nomeSlope.property) hk)⟩
  70 
  71 theorem exp_fiber (z w : Scalar) (h : (exp z).val.Equiv (exp w).val) :
  72     ∃ k : Int, (sub z.val w.val).Equiv (period k).val :=
  73   exp_kernel ⟨sub z.val w.val,sub_valid z.property w.property⟩
  74     (entireExponential_fiber_difference z w h)

  76 /-- One integer works throughout the nonempty actual overlap. -/
  77 theorem LogSeed.overlap_period (s t : LogSeed) (p : Scalar)
  78     (hp : s.function.domain p ∧ t.function.domain p) :
  79     ∃ k : Int, ∀ z : Scalar, ∀ hz : s.function.domain z ∧ t.function.domain z,
  80       (sub (s.function.eval z hz.1).val (t.function.eval z hz.2).val).Equiv (period k).val := by
  81   have he := equiv_trans (exp (s.function.eval p hp.1)).property p.property
  82     (exp (t.function.eval p hp.2)).property (s.exponential_eval p hp.1)
  83     (equiv_symm (t.exponential_eval p hp.2))
  84   obtain ⟨k,hk⟩ := exp_fiber _ _ he
  85   refine ⟨k,fun z hz => ?_⟩
  86   exact equiv_trans (sub_valid (s.function.eval z hz.1).property (t.function.eval z hz.2).property)
  87     (sub_valid (s.function.eval p hp.1).property (t.function.eval p hp.2).property) (period k).property
  88     (s.overlap_difference t p z hp hz) hk

  90 /-- The branch is the inverse on the neighborhood of its chosen seed.
  91 The chart hypothesis concerns the exponential image; proximity selects
  92 which logarithm of that image is meant. -/
  93 theorem LogSeed.log_exp (s : LogSeed) (w : Scalar)
  94     (hw : s.function.domain (exp w)) (hnear : Small (sub w.val s.value.val) 1) :
  95     (s.function.eval (exp w) hw).val.Equiv w.val := by
  96   let u := RelativeLogarithm.point s.center s.nonzero (exp w)
  97   let l := LocalLogarithm.function.eval u hw
  98   let o : Scalar := ⟨zero,ofQComplex_valid _⟩
  99   have hzero := LocalLogarithm.initial
 100   have hu := interior_bound LocalLogarithm.radius.val u hw
 101   have hdiff : (sub u.val o.val).Equiv u.val := by
 102     apply ComplexRawQuotient.equiv_of_ofRaw_eq
 103       (hleft := sub_valid u.property o.property) (hright := u.property)
 104     change scalarClass u-0=scalarClass u
 105     grind only
 106   have hb := LocalLogarithm.lipschitz o u (interior_zero _ LocalLogarithm.radius.property) hw
 107     LocalLogarithm.radius (Small.congr u.property (sub_valid u.property o.property) (equiv_symm hdiff) hu)
 108   have hL : (sub l.val (LocalLogarithm.function.eval o (interior_zero _ LocalLogarithm.radius.property)).val).Equiv l.val := by
 109     have hi := ComplexRawQuotient.ofRaw_eq_ofRaw
 110       (hleft := (LocalLogarithm.function.eval o (interior_zero _ LocalLogarithm.radius.property)).property)
 111       (hright := ofQComplex_valid _) hzero
 112     apply ComplexRawQuotient.equiv_of_ofRaw_eq
 113       (hleft := sub_valid l.property (LocalLogarithm.function.eval o (interior_zero _ LocalLogarithm.radius.property)).property)
 114       (hright := l.property)
 115     let L := scalarClass l
 116     let O := scalarClass (LocalLogarithm.function.eval o (interior_zero _ LocalLogarithm.radius.property))
 117     change O=0 at hi
 118     change L-O=L
 119     grind only
 120   have hl : Small l.val 1 := (Small.congr
 121     (sub_valid l.property (LocalLogarithm.function.eval o (interior_zero _ LocalLogarithm.radius.property)).property)
 122     l.property hL hb).mono (by decide +kernel)
 123   let a := s.function.eval (exp w) hw
 124   let d : Scalar := ⟨sub a.val w.val,sub_valid a.property w.property⟩
 125   have hd : d.val.Equiv (sub l.val (sub w.val s.value.val)) := by
 126     apply ComplexRawQuotient.equiv_of_ofRaw_eq
 127       (hleft := d.property) (hright := sub_valid l.property (sub_valid w.property s.value.property))
 128     let A := scalarClass s.value
 129     let L := scalarClass l
 130     let W := scalarClass w
 131     change (A+L)-W=L-(W-A)
 132     grind only
 133   have hbound := Small.congr (sub_valid l.property (sub_valid w.property s.value.property)) d.property
 134     (equiv_symm hd) (SeriesLimitLaws.small_sub hl hnear)
 135   have hb4 := hbound.mono (show (1+1:Rat)≤4 by decide +kernel)
 136   have hz := entireExponential_kernel_four_strip d hb4.2.2.1 hb4.2.2.2
 137     (entireExponential_fiber_difference a w (s.exponential_eval (exp w) hw))
 138   have he := ComplexRawQuotient.ofRaw_eq_ofRaw (hleft := d.property) (hright := ofQComplex_valid _) hz
 139   apply ComplexRawQuotient.equiv_of_ofRaw_eq (hleft := a.property) (hright := w.property)
 140   let A := scalarClass a
 141   let W := scalarClass w
 142   change A-W=0 at he
 143   change A=W
 144   grind only

 146 /-- The positive real logarithm extends as a full complex value throughout
 147 the real part of its normalized chart, not merely in its real coordinate. -/
 148 theorem realLogarithmAt_agrees (x y : PositiveInput)
 149     (hy : (realLogarithmAt x).function.domain (realAxis y.val)) :
 150     ((realLogarithmAt x).function.eval (realAxis y.val) hy).val.Equiv (realAxis (log y)).val := by
 151   let s := realLogarithmAt x
 152   let a := s.function.eval (realAxis y.val) hy
 153   let b := realAxis (log y)
 154   let d : Scalar := ⟨sub a.val b.val,sub_valid a.property b.property⟩
 155   have he := equiv_trans (exp a).property (realAxis y.val).property (exp b).property
 156     (s.exponential_eval _ hy) (equiv_symm (realLogarithmAt y).exponential)
 157   have hb := RelativeLogarithm.value_bound (realAxis x.val)
 158     (positive_real_nonzero x.val x.property) (realAxis y.val) hy
 159   have hl : (RealRaw.ofRat (-4)).Le d.val.imagPart := by
 160     intro n m
 161     have h := hb.2.2.1 n m
 162     simpa only [d,a,b,s,LogSeed.function,realLogarithmAt,LogarithmBranchCharts.function,
 163       MatrixExponential.propagatedBranch,DomainFunctions.scalarSum,realAxis,ofRealRaw,sub,add,neg,imagPart,
 164       QBox.add,QBox.neg,QComplex.add,QComplex.neg,Rat.zero_add,Rat.neg_zero,Rat.add_zero] using h
 165   have hu : d.val.imagPart.Le (RealRaw.ofRat 4) := by
 166     intro n m
 167     have h := hb.2.2.2 n m
 168     simpa only [d,a,b,s,LogSeed.function,realLogarithmAt,LogarithmBranchCharts.function,
 169       MatrixExponential.propagatedBranch,DomainFunctions.scalarSum,realAxis,ofRealRaw,sub,add,neg,imagPart,
 170       QBox.add,QBox.neg,QComplex.add,QComplex.neg,Rat.zero_add,Rat.neg_zero,Rat.add_zero] using h
 171   have hz := entireExponential_kernel_four_strip d hl hu (entireExponential_fiber_difference a b he)
 172   have hd := ComplexRawQuotient.ofRaw_eq_ofRaw (hleft := d.property) (hright := ofQComplex_valid _) hz
 173   apply ComplexRawQuotient.equiv_of_ofRaw_eq (hleft := a.property) (hright := b.property)
 174   let A := scalarClass a
 175   let B := scalarClass b
 176   change A-B=0 at hd
 177   change A=B
 178   grind only
 179 end ComputableAnalysis.ExponentialComputations.Complex
