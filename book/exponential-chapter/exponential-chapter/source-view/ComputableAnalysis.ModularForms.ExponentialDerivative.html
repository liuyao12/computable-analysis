   7  theorem exponentialCoefficient_shift (n : Nat) :
   8     (scaleRat ((n+1 : Nat) : Rat) (exponentialCoefficients (n+1))).Equiv
   9       (exponentialCoefficients n) := by
  10   have he := congrFun FormalPowerSeries.expCoeff_derivative n
  11   change ((n+1 : Nat) : Rat)*(1/factorialRat (n+1))=1/factorialRat n at he
  12   have hp : 0 ≤ ((n+1 : Nat) : Rat) := Rat.le_of_lt (Rat.natCast_pos.mpr (Nat.succ_pos n))
  13   intro k
  14   apply (compareAt_overlap_iff _ _ k k).mpr
  15   simp only [scaleRat, exponentialCoefficients, ofQComplex, QBox.scaleRat, if_pos hp,
  16     QBox.Overlaps, QComplex.le_def, he, Rat.mul_zero]
  17   exact ⟨⟨Rat.le_refl,Rat.le_refl⟩,⟨Rat.le_refl,Rat.le_refl⟩⟩

  19 theorem exponentialDerivativeTerm (z : Scalar) (n : Nat) :
  20     (BoundedSeries.derivativeTerm exponentialCoefficients z.val n).Equiv
  21       (LocalODE.seriesTerm exponentialCoefficients z.val n) := by
  22   have h := scaleRat_mul_equiv ((n+1 : Nat) : Rat) (exponentialCoefficients (n+1))
  23     (LocalODE.power z.val n) (exponentialCoefficients_valid _) (LocalODE.power_valid _ z.property _)
  24   exact equiv_trans
  25     (BoundedSeries.derivativeTerm_valid _ _ exponentialCoefficients_valid z.property n)
  26     (mul_valid (scaleRat_valid (exponentialCoefficients_valid _)) (LocalODE.power_valid _ z.property _))
  27     (LocalODE.seriesTerm_valid _ _ exponentialCoefficients_valid z.property n) h
  28     (mul_equiv (scaleRat_valid (exponentialCoefficients_valid _)) (exponentialCoefficients_valid _)
  29       (LocalODE.power_valid _ z.property _) (LocalODE.power_valid _ z.property _)
  30       (exponentialCoefficient_shift n) (equiv_refl _ (LocalODE.power_valid _ z.property _)))

  32 /-- The independently constructed sum of derivative terms equals the value sum. -/
  33 theorem exponentialChart_derivative_value (R : QPos) (z : Scalar)
  34     (hz : (exponentialChart R).domain z) :
  35     ((exponentialChart_holomorphic R).derivative z).Equiv ((exponentialChart R).eval z) := by
  36   let K := exponentialRatio R
  37   let C := exponentialBudget K
  38   have hK : 0 ≤ K := Rat.le_of_lt (exponentialRatio_positive R)
  39   have hC : 0 ≤ C := exponentialBudget_nonnegative K (exponentialRatio_positive R)
  40   have hR : 0 ≤ R.val := Rat.le_of_lt R.property
  41   have hzB := LocalODE.interior_bound R.val z hz
  42   have hcB := exponentialCoefficients_small K (exponentialRatio_positive R)
  43   have h8 := exponentialRatio_local R
  44   have hq : 0 ≤ 4*K*R.val := Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hK) hR
  45   have hr : 0 ≤ 2*K*R.val := Rat.mul_nonneg (Rat.mul_nonneg (by decide +kernel) hK) hR
  46   have h4 : 4*K*R.val ≤ (1 : Rat)/2 := by change 8*K*R.val ≤ (1 : Rat)/2 at h8; grind only
  47   have h2 : 2*K*R.val ≤ (1 : Rat)/2 := by grind only
  48   apply ScalarSeries.value_congr
  49     (BoundedSeries.derivativeTerm exponentialCoefficients z.val)
  50     (LocalODE.seriesTerm exponentialCoefficients z.val)
  51     (BoundedSeries.derivativeTerm_valid _ _ exponentialCoefficients_valid z.property)
  52     (LocalODE.seriesTerm_valid _ _ exponentialCoefficients_valid z.property)
  53     (C*K) (4*K*R.val) C (2*K*R.val) (Rat.mul_nonneg hC hK) hq hC hr h4 h2
  54   · intro n
  55     simpa only [Rat.mul_assoc] using BoundedSeries.derivativeTerm_majorant
  56       exponentialCoefficients z.val exponentialCoefficients_valid z.property C K R.val hC hK hR hcB hzB n
  57   · exact LocalODE.seriesTerm_bound exponentialCoefficients z.val exponentialCoefficients_valid
  58       z.property C K R.val hC hK hR hcB hzB
  59   · exact exponentialDerivativeTerm z

  61 /-- The actual entire exponential has derivative equal to its value. -/

  70 /-- The entire factorial exponential is normalized to one at zero. -/
  71 theorem entireExponential_zero :
  72     (entireExponentialValue ⟨zero,ofQComplex_valid _⟩).val.Equiv (ofQComplex QComplex.one) := by
  73   let R : QPos := ⟨1,by decide +kernel⟩
  74   let z : Scalar := ⟨zero,ofQComplex_valid _⟩
  75   have hz : (exponentialChart R).domain z :=
  76     ⟨0,Rat.le_refl,R.property,Small.zero Rat.le_refl⟩
  77   have hinit := BoundedSeries.seriesMap_initial exponentialCoefficients exponentialCoefficients_valid
  78     (exponentialBudget (exponentialRatio R)) (exponentialRatio R) R.val
  79     (exponentialBudget_nonnegative _ (exponentialRatio_positive R))
  80     (Rat.le_of_lt (exponentialRatio_positive R)) (Rat.le_of_lt R.property)
  81     (exponentialCoefficients_small _ (exponentialRatio_positive R)) (exponentialRatio_local R)
  82   have hc : (exponentialCoefficients 0).Equiv (ofQComplex QComplex.one) := by
  83     intro k
  84     apply (compareAt_overlap_iff _ _ k k).mpr
  85     change ((1 : Rat)/((1 : Nat) : Rat) ≤ 1 ∧ (0 : Rat) ≤ 0) ∧
  86       ((1 : Rat) ≤ (1 : Rat)/((1 : Nat) : Rat) ∧ (0 : Rat) ≤ 0)
  87     decide +kernel
  88   exact equiv_trans (entireExponentialValue z).property
  89     ((exponentialChart R).valid z hz) (ofQComplex_valid _)
  90     (equiv_symm (entireExponential_chart R z hz))
  91     (equiv_trans ((exponentialChart R).valid z hz) (exponentialCoefficients_valid 0)
  92       (ofQComplex_valid _) hinit hc)
