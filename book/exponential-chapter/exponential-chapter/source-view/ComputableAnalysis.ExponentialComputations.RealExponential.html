  10 theorem real_embedding_congr {x y : RealRaw} (hx : x.Valid) (hy : y.Valid) (h : x.Equiv y) :
  11     (ofRealRaw x).Equiv (ofRealRaw y) := ofRealRaw_equiv_of_equiv hx hy h

  13 abbrev RealInput := {x : RealRaw // x.Valid}
  14 def realAxis (x : RealInput) : Scalar := ⟨ofRealRaw x.val,ofRealRaw_valid _ x.property⟩
  15 def exp (x : RealInput) : RealInput :=
  16   ⟨(entireExponentialValue (realAxis x)).val.realPart,realPart_valid (entireExponentialValue (realAxis x)).property⟩
  17 
  18 theorem exp_congr (x y : RealInput) (h : x.val.Equiv y.val) : (exp x).val.Equiv (exp y).val :=
  19   realPart_equiv (entireExponentialValue_congr (realAxis x) (realAxis y) (real_embedding_congr x.property y.property h))

  21 theorem exp_real_embedding (x : RealInput) :
  22     (entireExponentialValue (realAxis x)).val.Equiv (realAxis (exp x)).val :=
  23   entireExponential_real_embedding x.val x.property

  25 /-- A strictly positive real name cannot represent zero. -/
  26 theorem positive_real_nonzero (x : RealInput) (hx : x.val.Pos) : NonzeroBoxSearch.Nonzero (realAxis x) := by
  27   intro hzero
  28   have h := realPart_equiv hzero
  29   obtain ⟨k,hk⟩ := hx
  30   have hb := (RealRaw.compareAt_overlap_iff _ _ k k).mp (h k)
  31   change (x.val.compute k).lo≤(0:Rat) ∧ (0:Rat)≤(x.val.compute k).hi at hb
  32   change 0<(x.val.compute k).lo at hk
  33   grind only

  36 /-- Exponential positivity on the entire represented real axis. -/
  37 theorem exp_positive (x : RealInput) : (exp x).val.Pos := by
  38   let h : Scalar := realAxis ⟨RealRaw.scaleRat (1/2) x.val,RealRaw.scaleRat_valid x.property⟩
  39   have he : (ofRealRaw (RealRaw.scaleRat 2 h.val.realPart)).Equiv (realAxis x).val := by
  40     intro n
  41     apply (compareAt_overlap_iff _ _ n n).mpr
  42     have ho := RealRaw.interval_order_of_valid x.val x.property n
  43     simp only [h,realAxis,ofRealRaw,realPart,RealRaw.scaleRat,RealRaw.scaleRatCompute,
  44       if_pos (show (0:Rat)≤2 by decide +kernel),if_pos (show (0:Rat)≤1/2 by decide +kernel)]
  45     constructor <;> constructor <;> grind [Rat.div_def]
  46   have hmag := entireExponential_squared_magnitude_real h
  47   have hnonneg := scalar_squared_magnitude_nonnegative (entireExponentialValue h)
  48   have hv := realPart_valid (mul_valid (entireExponentialValue h).property (conj_valid _ (entireExponentialValue h).property))
  49   let r : Scalar := ⟨ofRealRaw (RealRaw.scaleRat 2 h.val.realPart),
  50     ofRealRaw_valid _ (RealRaw.scaleRat_valid (realPart_valid h.property))⟩
  51   have hvalue := realPart_equiv (entireExponentialValue_congr r (realAxis x) he)
  52   have h0 : (RealRaw.ofRat 0).Le (exp x).val :=
  53     RealRaw.le_trans hv hnonneg
  54       (RealRaw.le_of_equiv hv (exp x).property
  55         (RealRaw.equiv_trans hv (realPart_valid (entireExponentialValue _).property) (exp x).property hmag hvalue))
  56   apply Classical.byContradiction
  57   intro hp
  58   have hle : (exp x).val.Le (RealRaw.ofRat 0) := by
  59     intro n m
  60     have hn : ¬0<((exp x).val.compute n).lo := fun hn => hp ⟨n,hn⟩
  61     change ((exp x).val.compute n).lo≤(0:Rat)
  62     grind only
  63   have hz := RealRaw.le_antisymm hle h0
  64   have hzero := real_embedding_congr (x := (exp x).val) (y := RealRaw.ofRat 0)
  65     (exp x).property (RealRaw.ofRat_valid 0) hz
  66   have hzembed : (ofRealRaw (RealRaw.ofRat 0)).Equiv zero := by
  67     intro n
  68     apply (compareAt_overlap_iff _ _ n n).mpr
  69     constructor <;> exact QComplex.le_refl _
  70   have hc := equiv_trans (entireExponentialValue (realAxis x)).property
  71     (realAxis (exp x)).property (ofQComplex_valid QComplex.zero)
  72     (exp_real_embedding x) (equiv_trans (realAxis (exp x)).property
  73       (ofRealRaw_valid (RealRaw.ofRat 0) (RealRaw.ofRat_valid 0)) (ofQComplex_valid _) hzero hzembed)
  74   exact entireExponential_ne_zero (realAxis x) hc

  76 /-- Real-axis injectivity, derived from the exact complex exponential fiber law. -/
  77 theorem exp_injective (x y : RealInput) (h : (exp x).val.Equiv (exp y).val) : x.val.Equiv y.val := by
  78   have he := equiv_trans (entireExponentialValue (realAxis x)).property
  79     (realAxis (exp x)).property (entireExponentialValue (realAxis y)).property
  80     (exp_real_embedding x)
  81     (equiv_trans (realAxis (exp x)).property (realAxis (exp y)).property
  82       (entireExponentialValue (realAxis y)).property (real_embedding_congr (exp x).property (exp y).property h) (equiv_symm (exp_real_embedding y)))
  83   exact entireExponential_fiber_realPart (realAxis x) (realAxis y) he
